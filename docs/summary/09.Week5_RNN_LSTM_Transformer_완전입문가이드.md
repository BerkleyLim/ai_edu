## 🔄 2. RNN (Recurrent Neural Network) 완전 이해

### 2.1 RNN의 핵심 아이디어

**순환 구조:**
```
h_t = f(h_{t-1}, x_t)
y_t = g(h_t)

h_t: 시점 t의 은닉 상태 (메모리 역할)
x_t: 시점 t의 입력
y_t: 시점 t의 출력
```

**RNN의 메모리 메커니즘:**

### 2.2 바닐라 RNN 완전 구현

```python
class VanillaRNN:
    def __init__(self, input_size, hidden_size, output_size, learning_rate=0.01):
        """
        바닐라 RNN 완전 구현
        
        Args:
            input_size: 입력 차원
            hidden_size: 은닉 상태 차원
            output_size: 출력 차원
            learning_rate: 학습률
        """
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.learning_rate = learning_rate
        
        # 가중치 초기화 (Xavier 초기화)
        self.Wxh = np.random.randn(hidden_size, input_size) * np.sqrt(2.0 / input_size)
        self.Whh = np.random.randn(hidden_size, hidden_size) * np.sqrt(2.0 / hidden_size)
        self.Why = np.random.randn(output_size, hidden_size) * np.sqrt(2.0 / hidden_size)
        
        # 편향
        self.bh = np.zeros((hidden_size, 1))
        self.by = np.zeros((output_size, 1))
        
        # 학습 기록
        self.losses = []
    
    def forward(self, inputs, initial_hidden=None):
        """
        순전파
        
        Args:
            inputs: 입력 시퀀스 [(input_size,), (input_size,), ...]
            initial_hidden: 초기 은닉 상태
        
        Returns:
            outputs: 출력 시퀀스
            hidden_states: 은닉 상태들
        """
        seq_len = len(inputs)
        
        if initial_hidden is None:
            h = np.zeros((self.hidden_size, 1))
        else:
            h = initial_hidden.copy()
        
        outputs = []
        hidden_states = [h.copy()]
        
        for t in range(seq_len):
            x = inputs[t].reshape(-1, 1)
            
            # 은닉 상태 업데이트: h_t = tanh(W_xh * x_t + W_hh * h_{t-1} + b_h)
            h = np.tanh(np.dot(self.Wxh, x) + np.dot(self.Whh, h) + self.bh)
            
            # 출력 계산: y_t = W_hy * h_t + b_y
            y = np.dot(self.Why, h) + self.by
            
            outputs.append(y)
            hidden_states.append(h.copy())
        
        return outputs, hidden_states[1:]  # 초기 상태 제외
    
    def backward(self, inputs, targets, outputs, hidden_states):
        """
        역전파 (BPTT - Backpropagation Through Time)
        
        Args:
            inputs: 입력 시퀀스
            targets: 타겟 시퀀스
            outputs: 순전파 출력
            hidden_states: 은닉 상태들
        """
        seq_len = len(inputs)
        
        # 기울기 초기화
        dWxh = np.zeros_like(self.Wxh)
        dWhh = np.zeros_like(self.Whh)
        dWhy = np.zeros_like(self.Why)
        dbh = np.zeros_like(self.bh)
        dby = np.zeros_like(self.by)
        
        dh_next = np.zeros((self.hidden_size, 1))
        
        # 시간을 거슬러 올라가며 기울기 계산
        for t in reversed(range(seq_len)):
            x = inputs[t].reshape(-1, 1)
            target = targets[t].reshape(-1, 1) if targets[t] is not None else None
            y = outputs[t]
            h = hidden_states[t]
            h_prev = hidden_states[t-1] if t > 0 else np.zeros((self.hidden_size, 1))
            
            # 출력층 기울기
            if target is not None:
                dy = y - target  # MSE의 기울기
                dWhy += np.dot(dy, h.T)
                dby += dy
                
                # 은닉층으로 역전파
                dh = np.dot(self.Why.T, dy) + dh_next
            else:
                dh = dh_next
            
            # tanh 미분: dh_raw = (1 - h²) * dh
            dh_raw = (1 - h * h) * dh
            
            # 가중치 기울기
            dWxh += np.dot(dh_raw, x.T)
            dWhh += np.dot(dh_raw, h_prev.T)
            dbh += dh_raw
            
            # 다음 시간 단계로 전파
            dh_next = np.dot(self.Whh.T, dh_raw)
        
        # 기울기 클리핑 (기울기 폭발 방지)
        max_norm = 5.0
        for grad in [dWxh, dWhh, dWhy, dbh, dby]:
            np.clip(grad, -max_norm, max_norm, out=grad)
        
        return dWxh, dWhh, dWhy, dbh, dby
    
    def train_step(self, inputs, targets):
        """한 스텝 학습"""
        # 순전파
        outputs, hidden_states = self.forward(inputs)
        
        # 손실 계산
        loss = 0
        valid_targets = [t for t in targets if t is not None]
        valid_outputs = [outputs[i] for i, t in enumerate(targets) if t is not None]
        
        if valid_targets:
            for out, tar in zip(valid_outputs, valid_targets):
                loss += np.mean((out - tar.reshape(-1, 1)) ** 2)
            loss /= len(valid_targets)
        
        # 역전파
        gradients = self.backward(inputs, targets, outputs, hidden_states)
        dWxh, dWhh, dWhy, dbh, dby = gradients
        
        # 매개변수 업데이트
        self.Wxh -= self.learning_rate * dWxh
        self.Whh -= self.learning_rate * dWhh
        self.Why -= self.learning_rate * dWhy
        self.bh -= self.learning_rate * dbh
        self.by -= self.learning_rate * dby
        
        return loss
    
    def generate_text(self, seed_inputs, length=10):
        """텍스트 생성 (간단한 예시)"""
        h = np.zeros((self.hidden_size, 1))
        generated = []
        
        # Seed로 상태 초기화
        for x in seed_inputs:
            h = np.tanh(np.dot(self.Wxh, x.reshape(-1, 1)) + np.dot(self.Whh, h) + self.bh)
        
        # 텍스트 생성
        for _ in range(length):
            y = np.dot(self.Why, h) + self.by
            # 간단한 확률적 샘플링 (실제로는 softmax 사용)
            next_input = y + np.random.randn(*y.shape) * 0.1
            generated.append(next_input.flatten())
            
            # 다음 상태 계산
            h = np.tanh(np.dot(self.Wxh, next_input) + np.dot(self.Whh, h) + self.bh)
        
        return generated

# RNN으로 간단한 시퀀스 학습
def train_simple_sequence():
    """간단한 패턴을 RNN으로 학습"""
    print("=== RNN으로 사인파 학습 ===")
    
    # 사인파 데이터 생성
    t = np.linspace(0, 4*np.pi, 100)
    sine_wave = np.sin(t)
    
    # 시퀀스 데이터 준비 (과거 5개 값으로 다음 값 예측)
    sequence_length = 5
    X_sequences = []
    y_targets = []
    
    for i in range(len(sine_wave) - sequence_length):
        X_sequences.append(sine_wave[i:i+sequence_length])
        y_targets.append(sine_wave[i+sequence_length])
    
    # RNN 생성
    rnn = VanillaRNN(input_size=1, hidden_size=10, output_size=1, learning_rate=0.01)
    
    # 학습
    print("RNN 학습 중...")
    for epoch in range(200):
        total_loss = 0
        
        for seq_x, target_y in zip(X_sequences[:50], y_targets[:50]):  # 일부만 사용
            # 입력 준비
            inputs = [np.array([x]) for x in seq_x]
            targets = [None] * (len(inputs) - 1) + [np.array([target_y])]  # 마지막에만 타겟
            
            loss = rnn.train_step(inputs, targets)
            total_loss += loss
        
        if epoch % 50 == 0:
            avg_loss = total_loss / 50
            print(f"Epoch {epoch}: Loss = {avg_loss:.6f}")
            rnn.losses.append(avg_loss)
    
    # 예측 테스트
    print("\n예측 테스트:")
    test_sequence = sine_wave[50:55]
    test_inputs = [np.array([x]) for x in test_sequence]
    
    outputs, _ = rnn.forward(test_inputs)
    predicted = outputs[-1].flatten()[0]
    actual = sine_wave[55]
    
    print(f"입력 시퀀스: {test_sequence}")
    print(f"실제 다음 값: {actual:.4f}")
    print(f"예측된 값: {predicted:.4f}")
    print(f"오차: {abs(actual - predicted):.4f}")
    
    # 시각화
    plt.figure(figsize=(12, 8))
    
    plt.subplot(2, 2, 1)
    plt.plot(t, sine_wave, 'b-', label='사인파')
    plt.axvline(x=t[50], color='r', linestyle='--', label='테스트 시작')
    plt.axvline(x=t[55], color='g', linestyle='--', label='예측 지점')
    plt.title('원본 사인파 데이터')
    plt.legend()
    
    plt.subplot(2, 2, 2)
    plt.plot(rnn.losses, 'r-o')
    plt.title('학습 손실')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    
    plt.subplot(2, 2, 3)
    plt.plot(test_sequence, 'b-o', label='입력 시퀀스')
    plt.axhline(y=actual, color='g', linestyle='--', label=f'실제값: {actual:.3f}')
    plt.axhline(y=predicted, color='r', linestyle='--', label=f'예측값: {predicted:.3f}')
    plt.title('예측 결과')
    plt.legend()
    
    plt.tight_layout()
    plt.show()
    
    return rnn

# 실행
trained_rnn = train_simple_sequence()
```

```python
def demonstrate_gradient_vanishing():
    """기울기 소실 문제 시연"""
    print("=== 기울기 소실 문제 시연 ===")
    
    # tanh의 미분값은 최대 1
    # 연쇄 법칙에 의해 기울기가 계속 곱해짐
    
    sequence_lengths = [5, 10, 20, 50, 100]
    
    print("시퀀스 길이별 기울기 변화:")
    print(f"{'길이':<6} {'초기기울기':<12} {'최종기울기':<12} {'비율':<10}")
    print("-" * 45)
    
    for seq_len in sequence_lengths:
        initial_gradient = 1.0
        current_gradient = initial_gradient
        
        # tanh 미분의 평균값 약 0.4로 가정
        tanh_derivative_avg = 0.4
        weight_avg = 0.5  # 가중치 평균값
        
        # 기울기가 시간을 거슬러 올라가며 감소
        for t in range(seq_len):
            current_gradient *= (tanh_derivative_avg * weight_avg)
        
        ratio = current_gradient / initial_gradient
        print(f"{seq_len:<6} {initial_gradient:<12.1f} {current_gradient:<12.8f} {ratio:<10.2e}")
    
    print("\n관찰:")
    print("1. 시퀀스가 길어질수록 기울기가 기하급수적으로 감소")
    print("2. 긴 의존성을 학습하기 어려움")
```

```python
def demonstrate_gradient_exploding():
    """기울기 폭발 문제 시연"""
    print("\n=== 기울기 폭발 문제 시연 ===")
    
    sequence_lengths = [5, 10, 20, 30]
    
    print("큰 가중치로 인한 기울기 폭발:")
    print(f"{'길이':<6} {'초기기울기':<12} {'최종기울기':<15} {'상태':<10}")
    print("-" * 50)
    
    for seq_len in sequence_lengths:
        initial_gradient = 1.0
        current_gradient = initial_gradient
        
        # 큰 가중치와 활성화 함수 미분
        large_weight = 1.2  # 1보다 큰 가중치
        activation_derivative = 1.0  # ReLU 미분값이라고 가정
        
        for t in range(seq_len):

        if current_gradient > 1000:
            status = "폭발!"

            status = "위험"
        else:
            status = "정상"
        
        print(f"{seq_len:<6} {initial_gradient:<12.1f} {current_gradient:<15.2f} {status:<10}")
    
    print("\n해결책:")
    print("1. 기울기 클리핑 (Gradient Clipping)")
    print("2. 더 나은 초기화")
    print("3. LSTM/GRU 사용")

demonstrate_gradient_exploding()
```

## 🧠 3. LSTM (Long Short-Term Memory) 완전 정복

### 3.1 LSTM의 핵심 아이디어

**문제점과 해결책:**
```
RNN 문제: 장기 의존성을 기억하지 못함
LSTM 해결: 셀 상태(Cell State)라는 별도의 메모리 라인 추가
```

**LSTM의 구조:**
```
1. 망각 게이트 (Forget Gate): 무엇을 잊을지 결정
2. 입력 게이트 (Input Gate): 새로운 정보 중 무엇을 저장할지 결정
3. 셀 상태 업데이트: 이전 기억 + 새로운 기억
4. 출력 게이트 (Output Gate): 무엇을 출력할지 결정
```

```python
class LSTMEducational:
    """교육용 LSTM - 각 단계를 자세히 설명"""
    
    def __init__(self, input_size, hidden_size):
        self.input_size = input_size
        self.hidden_size = hidden_size
        
        # LSTM은 4개의 게이트를 위한 가중치가 필요
        # forget, input, candidate, output gate
        
        # 입력에 대한 가중치
        self.Wf = np.random.randn(hidden_size, input_size) * 0.1  # forget gate
        self.Wi = np.random.randn(hidden_size, input_size) * 0.1  # input gate
        self.Wc = np.random.randn(hidden_size, input_size) * 0.1  # candidate
        self.Wo = np.random.randn(hidden_size, input_size) * 0.1  # output gate
        
        # 이전 은닉상태에 대한 가중치
        self.Uf = np.random.randn(hidden_size, hidden_size) * 0.1
        self.Ui = np.random.randn(hidden_size, hidden_size) * 0.1
        self.Uc = np.random.randn(hidden_size, hidden_size) * 0.1
        self.Uo = np.random.randn(hidden_size, hidden_size) * 0.1
        
        # 편향
        self.bf = np.zeros((hidden_size, 1))
        self.bi = np.zeros((hidden_size, 1))
        self.bc = np.zeros((hidden_size, 1))
        self.bo = np.zeros((hidden_size, 1))
    
    def sigmoid(self, x):
        """시그모이드 함수 (0~1 사이 출력)"""
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
    
    def tanh(self, x):
        """tanh 함수 (-1~1 사이 출력)"""
        return np.tanh(np.clip(x, -500, 500))
    
    def lstm_cell(self, x, h_prev, c_prev):
        """
        LSTM 셀의 순전파
        
        Args:
            x: 현재 입력 (input_size, 1)
            h_prev: 이전 은닉상태 (hidden_size, 1)
            c_prev: 이전 셀상태 (hidden_size, 1)
        
        Returns:
            h: 새로운 은닉상태
            c: 새로운 셀상태
            gates: 각 게이트 값들 (디버깅용)
        """
        
        # 1. 망각 게이트 (Forget Gate)
        # "이전 셀 상태에서 무엇을 잊을까?"
        forget_gate = self.sigmoid(
            np.dot(self.Wf, x) + np.dot(self.Uf, h_prev) + self.bf
        )
        
        # 2. 입력 게이트 (Input Gate) 
        # "새로운 정보 중에서 무엇을 저장할까?"
        input_gate = self.sigmoid(
            np.dot(self.Wi, x) + np.dot(self.Ui, h_prev) + self.bi
        )
        
        # 3. 후보 값 (Candidate Values)
        # "새로운 정보는 무엇인가?"
        candidate_values = self.tanh(
            np.dot(self.Wc, x) + np.dot(self.Uc, h_prev) + self.bc
        )
        
        # 4. 셀 상태 업데이트
        # 셀 상태 = (이전 기억 × 망각 게이트) + (새 정보 × 입력 게이트)
        cell_state = forget_gate * c_prev + input_gate * candidate_values
        
        # 5. 출력 게이트 (Output Gate)
        # "업데이트된 셀 상태에서 무엇을 출력할까?"
        output_gate = self.sigmoid(
            np.dot(self.Wo, x) + np.dot(self.Uo, h_prev) + self.bo
        )
        
        # 6. 은닉 상태 계산
        # 은닉 상태 = 출력 게이트 × tanh(셀 상태)
        hidden_state = output_gate * self.tanh(cell_state)
        
        # 디버깅을 위한 게이트 값들
        gates = {
            'forget': forget_gate,
            'input': input_gate,
            'candidate': candidate_values,
            'output': output_gate
        }
        
        return hidden_state, cell_state, gates
    
    def forward_with_explanation(self, sequence):
        """설명과 함께 순전파 실행"""
        seq_len = len(sequence)
        h = np.zeros((self.hidden_size, 1))
        c = np.zeros((self.hidden_size, 1))
        
        print("=== LSTM 순전파 과정 ===")
        print(f"{'시점':<4} {'입력':<8} {'망각게이트':<10} {'입력게이트':<10} {'출력게이트':<10} {'셀상태변화':<12}")
        print("-" * 70)
        
        all_states = []
        
        for t, x_val in enumerate(sequence):
            x = np.array([[x_val]])
            
            h, c, gates = self.lstm_cell(x, h, c)
            
            # 평균값들 계산 (시각화용)
            forget_avg = np.mean(gates['forget'])
            input_avg = np.mean(gates['input'])
            output_avg = np.mean(gates['output'])
            cell_change = np.mean(np.abs(c - (gates['forget'] * c + gates['input'] * gates['candidate'])))
            
            print(f"{t:<4} {x_val:<8.3f} {forget_avg:<10.3f} {input_avg:<10.3f} "
                  f"{output_avg:<10.3f} {cell_change:<12.6f}")
            
            all_states.append({
                'h': h.copy(),
                'c': c.copy(),
                'gates': gates.copy()
            })
        
        return all_states
```

### 3.3 완전한 LSTM 구현

```python
class CompleteLSTM:
    """완전한 LSTM 구현"""
    
    def __init__(self, input_size, hidden_size, output_size, learning_rate=0.001):
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.learning_rate = learning_rate
        
        # Xavier 초기화
        def xavier_init(shape):
            return np.random.randn(*shape) * np.sqrt(2.0 / sum(shape))
        
        # LSTM 가중치 초기화
        self.params = {}
        
        # 입력 가중치
        self.params['Wf'] = xavier_init((hidden_size, input_size))
        self.params['Wi'] = xavier_init((hidden_size, input_size))
        self.params['Wc'] = xavier_init((hidden_size, input_size))
        self.params['Wo'] = xavier_init((hidden_size, input_size))
        
        # 은닉 상태 가중치
        self.params['Uf'] = xavier_init((hidden_size, hidden_size))
        self.params['Ui'] = xavier_init((hidden_size, hidden_size))
        self.params['Uc'] = xavier_init((hidden_size, hidden_size))
        self.params['Uo'] = xavier_init((hidden_size, hidden_size))
        
        # 편향 (forget gate는 1로 초기화 - 초기에는 기억하도록)
        self.params['bf'] = np.ones((hidden_size, 1))
        self.params['bi'] = np.zeros((hidden_size, 1))
        self.params['bc'] = np.zeros((hidden_size, 1))
        self.params['bo'] = np.zeros((hidden_size, 1))
        
        # 출력층
        self.params['Wy'] = xavier_init((output_size, hidden_size))
        self.params['by'] = np.zeros((output_size, 1))
        
        # 최적화를 위한 AdaGrad
        self.cache = {key: np.zeros_like(val) for key, val in self.params.items()}
        
    def sigmoid(self, x):
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
    
    def tanh(self, x):
        return np.tanh(np.clip(x, -500, 500))
    
    def forward(self, inputs, initial_states=None):
        """순전파"""
        seq_len = len(inputs)
        
        if initial_states is None:
            h = np.zeros((self.hidden_size, 1))
            c = np.zeros((self.hidden_size, 1))
        else:
            h, c = initial_states
        
        # 중간 값들 저장 (역전파용)
        cache = {
            'inputs': inputs,
            'h_states': [h.copy()],
            'c_states': [c.copy()],
            'gates': []
        }
        
        outputs = []
        
        for t in range(seq_len):
            x = inputs[t].reshape(-1, 1)
            
            # LSTM 셀
            # 1. 망각 게이트
            f = self.sigmoid(np.dot(self.params['Wf'], x) + 
                           np.dot(self.params['Uf'], h) + self.params['bf'])
            
            # 2. 입력 게이트
            i = self.sigmoid(np.dot(self.params['Wi'], x) + 
                           np.dot(self.params['Ui'], h) + self.params['bi'])
            
            # 3. 후보값
            c_tilde = self.tanh(np.dot(self.params['Wc'], x) + 
                               np.dot(self.params['Uc'], h) + self.params['bc'])
            
            # 4. 셀 상태 업데이트
            c = f * c + i * c_tilde
            
            # 5. 출력 게이트
            o = self.sigmoid(np.dot(self.params['Wo'], x) + 
                           np.dot(self.params['Uo'], h) + self.params['bo'])
            
            # 6. 은닉 상태 업데이트
            h = o * self.tanh(c)
            
            # 7. 출력 계산
            y = np.dot(self.params['Wy'], h) + self.params['by']
            
            # 캐시에 저장
            cache['h_states'].append(h.copy())
            cache['c_states'].append(c.copy())
            cache['gates'].append({
                'f': f, 'i': i, 'c_tilde': c_tilde, 'o': o,
                'c': c.copy(), 'h': h.copy(), 'y': y.copy()
            })
            
            outputs.append(y)
        
        return outputs, cache
    
    def backward(self, targets, cache):
        """역전파"""
        inputs = cache['inputs']
        h_states = cache['h_states']
        c_states = cache['c_states']
        gates = cache['gates']
        
        seq_len = len(inputs)
        
        # 기울기 초기화
        grads = {key: np.zeros_like(val) for key, val in self.params.items()}
        
        dh_next = np.zeros((self.hidden_size, 1))
        dc_next = np.zeros((self.hidden_size, 1))
        
        # BPTT
        for t in reversed(range(seq_len)):
            x = inputs[t].reshape(-1, 1)
            target = targets[t].reshape(-1, 1) if targets[t] is not None else None
            
            gate = gates[t]
            h_prev = h_states[t]
            c_prev = c_states[t]
            
            # 출력층 기울기
            if target is not None:
                dy = gate['y'] - target
                grads['Wy'] += np.dot(dy, gate['h'].T)
                grads['by'] += dy
                dh = np.dot(self.params['Wy'].T, dy) + dh_next
            else:
                dh = dh_next
            
            # 출력 게이트
            do = dh * self.tanh(gate['c'])
            do_input = do * gate['o'] * (1 - gate['o'])
            
            # 셀 상태
            dc = dh * gate['o'] * (1 - self.tanh(gate['c'])**2) + dc_next
            
            # 망각 게이트
            df = dc * c_prev
            df_input = df * gate['f'] * (1 - gate['f'])
            
            # 입력 게이트
            di = dc * gate['c_tilde']
            di_input = di * gate['i'] * (1 - gate['i'])
            
            # 후보값
            dc_tilde = dc * gate['i']
            dc_tilde_input = dc_tilde * (1 - gate['c_tilde']**2)
            
            # 가중치 기울기 누적
            grads['Wf'] += np.dot(df_input, x.T)
            grads['Wi'] += np.dot(di_input, x.T)
            grads['Wc'] += np.dot(dc_tilde_input, x.T)
            grads['Wo'] += np.dot(do_input, x.T)
            
            grads['Uf'] += np.dot(df_input, h_prev.T)
            grads['Ui'] += np.dot(di_input, h_prev.T)
            grads['Uc'] += np.dot(dc_tilde_input, h_prev.T)
            grads['Uo'] += np.dot(do_input, h_prev.T)
            
            grads['bf'] += df_input
            grads['bi'] += di_input
            grads['bc'] += dc_tilde_input
            grads['bo'] += do_input
            
            # 다음 시간 단계로 전파
            dh_next = (np.dot(self.params['Uf'].T, df_input) + 
                      np.dot(self.params['Ui'].T, di_input) + 
                      np.dot(self.params['Uc'].T, dc_tilde_input) + 
                      np.dot(self.params['Uo'].T, do_input))
            
            dc_next = dc * gate['f']
        
        # 기울기 클리핑
        max_norm = 5.0
        for key in grads:
            np.clip(grads[key], -max_norm, max_norm, out=grads[key])
        
        return grads
    
    def train_step(self, inputs, targets):
        """한 스텝 훈련"""
        # 순전파
        outputs, cache = self.forward(inputs)
        
        # 손실 계산
        loss = 0
        count = 0
        for i, target in enumerate(targets):
            if target is not None:
                loss += np.mean((outputs[i] - target.reshape(-1, 1))**2)
                count += 1
        
        if count > 0:
            loss /= count
        
        # 역전파
        grads = self.backward(targets, cache)
        
        # AdaGrad 업데이트
        eps = 1e-8
        for key in self.params:
            self.cache[key] += grads[key]**2
            self.params[key] -= (self.learning_rate / 
                               (np.sqrt(self.cache[key]) + eps)) * grads[key]
        
        return loss
    
    def predict(self, inputs):
        """예측"""
        outputs, _ = self.forward(inputs)
        return outputs

def test_long_term_dependency():

            sequences.append(seq)
            targets.append(target)

    correct = 0

        predicted = 1 if outputs[-1][0, 0] > 0 else -1
        
        if predicted == target:
            correct += 1

    print(f"\nLSTM 정확도: {accuracy:.1f}%")
    
    # 시각화
```

## 🎯 4. Attention 메커니즘과 Transformer

### 4.1 Attention의 핵심 아이디어

**문제: RNN/LSTM의 정보 병목**
```
긴 문장을 처리할 때 모든 정보를 마지막 은닉상태에 압축
→ 정보 손실 발생
→ 번역에서 처음 단어들이 잊혀짐
```

**해결: Attention 메커니즘**
```
"번역할 때 어떤 단어에 집중할지 동적으로 결정"
모든 입력 위치의 정보를 직접 참조 가능
```

### 4.2 Attention 메커니즘 구현

```python
class AttentionMechanism:
    """Attention 메커니즘 교육용 구현"""
    
    def __init__(self, hidden_size):
        self.hidden_size = hidden_size
        
        # Attention 가중치
        self.W_attention = np.random.randn(hidden_size, hidden_size) * 0.1
        self.v_attention = np.random.randn(hidden_size, 1) * 0.1
    
    def softmax(self, x):
        """안정한 softmax"""
        exp_x = np.exp(x - np.max(x))
        return exp_x / np.sum(exp_x, keepdims=True)
    
    def compute_attention(self, query, keys, values):
        """
        Attention 가중치 계산
        
        Args:
            query: 현재 상태 (hidden_size, 1)
            keys: 모든 입력 상태들 [(hidden_size, 1), ...]
            values: 모든 입력 상태들 (보통 keys와 같음)
        
        Returns:
            context_vector: 가중합된 컨텍스트 벡터
            attention_weights: attention 가중치들
        """
        seq_len = len(keys)
        
        # 1. 각 키에 대한 점수 계산
        scores = []
        for key in keys:
            # score = v^T * tanh(W * (query + key))
            combined = query + key
            score = np.dot(self.v_attention.T, 
                          np.tanh(np.dot(self.W_attention, combined)))
            scores.append(score[0, 0])
        
        scores = np.array(scores)
        
        # 2. Softmax로 확률 분포 변환
        attention_weights = self.softmax(scores)
        
        # 3. 가중합으로 컨텍스트 벡터 생성
        context_vector = np.zeros_like(values[0])
        for i, (weight, value) in enumerate(zip(attention_weights, values)):
            context_vector += weight * value
```

```python
class SelfAttention:
    """Self-Attention 메커니즘"""
    
    def __init__(self, embed_dim, num_heads=1):
        self.embed_dim = embed_dim
        self.num_heads = num_heads
        self.head_dim = embed_dim // num_heads
        
        assert embed_dim % num_heads == 0
        
        # Query, Key, Value 변환 행렬
        self.W_q = np.random.randn(embed_dim, embed_dim) * 0.1
        self.W_k = np.random.randn(embed_dim, embed_dim) * 0.1
        self.W_v = np.random.randn(embed_dim, embed_dim) * 0.1
        self.W_o = np.random.randn(embed_dim, embed_dim) * 0.1
    
    def softmax(self, x, axis=-1):
        """안정한 softmax"""
        x_max = np.max(x, axis=axis, keepdims=True)
        exp_x = np.exp(x - x_max)
        return exp_x / np.sum(exp_x, axis=axis, keepdims=True)
    
    def scaled_dot_product_attention(self, Q, K, V):
        """
        Scaled Dot-Product Attention
        
        Attention(Q,K,V) = softmax(QK^T / √d_k)V
        """
        # Q, K의 내적 계산
        scores = np.dot(Q, K.T)  # (seq_len, seq_len)
        
        # √d_k로 스케일링
        scores = scores / np.sqrt(self.head_dim)
        
        # Softmax 적용
        attention_weights = self.softmax(scores, axis=-1)
        
        # Value와 가중합
        output = np.dot(attention_weights, V)
        
        return output, attention_weights
    
    def forward(self, X):
        """
        Self-Attention 순전파
        
        Args:
            X: 입력 시퀀스 (seq_len, embed_dim)
        
        Returns:
            output: attention 출력 (seq_len, embed_dim)
            attention_weights: attention 가중치
        """
        seq_len, embed_dim = X.shape
        
        # Query, Key, Value 생성
        Q = np.dot(X, self.W_q)  # (seq_len, embed_dim)
        K = np.dot(X, self.W_k)  # (seq_len, embed_dim)
        V = np.dot(X, self.W_v)  # (seq_len, embed_dim)
        
        if self.num_heads == 1:
            # Single-head attention
            output, attention_weights = self.scaled_dot_product_attention(Q, K, V)
        else:
            # Multi-head attention
            outputs = []
            all_attention_weights = []
            
            for head in range(self.num_heads):
                # 헤드별로 분할
                start_idx = head * self.head_dim
                end_idx = start_idx + self.head_dim
                
                Q_head = Q[:, start_idx:end_idx]
                K_head = K[:, start_idx:end_idx]
                V_head = V[:, start_idx:end_idx]
                
                output_head, weights_head = self.scaled_dot_product_attention(
                    Q_head, K_head, V_head
                )
                
                outputs.append(output_head)
                all_attention_weights.append(weights_head)
            
            # 헤드들을 연결
            output = np.concatenate(outputs, axis=1)
            attention_weights = np.mean(all_attention_weights, axis=0)
        
        # 출력 변환
        output = np.dot(output, self.W_o)
        
        return output, attention_weights
    
    def visualize_self_attention(self, sentence, attention_weights):
        """Self-Attention 히트맵 시각화"""
        words = sentence.split()
        
        plt.figure(figsize=(10, 8))
        
        # Attention 히트맵
        im = plt.imshow(attention_weights, cmap='Blues', aspect='auto')
        plt.xticks(range(len(words)), words, rotation=45)
        plt.yticks(range(len(words)), words)
        plt.xlabel('Keys (attending to)')
        plt.ylabel('Queries (attending from)')
        plt.title('Self-Attention Heatmap')
        
        # 각 셀에 값 표시
        for i in range(len(words)):
            for j in range(len(words)):
                plt.text(j, i, f'{attention_weights[i, j]:.2f}', 
                        ha='center', va='center',
                        color='white' if attention_weights[i, j] > 0.5 else 'black')
        
        plt.colorbar(im)
        plt.tight_layout()
        plt.show()
```

```python
class SimpleTransformer:
    """간단한 Transformer 구현"""
    
    def __init__(self, vocab_size, embed_dim, num_heads, ff_dim, num_layers):
        self.vocab_size = vocab_size
        self.embed_dim = embed_dim
        self.num_heads = num_heads
        self.ff_dim = ff_dim
        self.num_layers = num_layers
        
        # 임베딩 레이어
        self.embedding = np.random.randn(vocab_size, embed_dim) * 0.1
        
        # 위치 인코딩
        self.positional_encoding = self._create_positional_encoding(1000, embed_dim)
        
        # Transformer 레이어들
        self.attention_layers = []
        self.ff_layers = []
        self.layer_norms1 = []
        self.layer_norms2 = []
        
        for _ in range(num_layers):
            self.attention_layers.append(SelfAttention(embed_dim, num_heads))
            
            # Feed-Forward 네트워크
            ff_layer = {
                'W1': np.random.randn(embed_dim, ff_dim) * 0.1,
                'b1': np.zeros((1, ff_dim)),
                'W2': np.random.randn(ff_dim, embed_dim) * 0.1,
                'b2': np.zeros((1, embed_dim))
            }
            self.ff_layers.append(ff_layer)
            
            # Layer Normalization (간단한 버전)
            self.layer_norms1.append({'gamma': np.ones(embed_dim), 'beta': np.zeros(embed_dim)})
            self.layer_norms2.append({'gamma': np.ones(embed_dim), 'beta': np.zeros(embed_dim)})
    
    def _create_positional_encoding(self, max_len, embed_dim):
        """사인/코사인 위치 인코딩 생성"""
        pe = np.zeros((max_len, embed_dim))
        position = np.arange(0, max_len).reshape(-1, 1)
        
        div_term = np.exp(np.arange(0, embed_dim, 2) * 
                         -(np.log(10000.0) / embed_dim))
        
        pe[:, 0::2] = np.sin(position * div_term)
        pe[:, 1::2] = np.cos(position * div_term)
        
        return pe
    
    def layer_norm(self, x, gamma, beta, eps=1e-8):
        """Layer Normalization"""
        mean = np.mean(x, axis=-1, keepdims=True)
        var = np.var(x, axis=-1, keepdims=True)
        normalized = (x - mean) / np.sqrt(var + eps)
        return gamma * normalized + beta
    
    def feed_forward(self, x, ff_layer):
        """Feed-Forward 네트워크"""
        # 첫 번째 선형 변환 + ReLU
        hidden = np.maximum(0, np.dot(x, ff_layer['W1']) + ff_layer['b1'])
        
        # 두 번째 선형 변환
        output = np.dot(hidden, ff_layer['W2']) + ff_layer['b2']
        
        return output
    
    def forward(self, token_ids):
        """Transformer 순전파"""
        seq_len = len(token_ids)
        
        # 1. 토큰 임베딩
        embedded = self.embedding[token_ids]  # (seq_len, embed_dim)
        
        # 2. 위치 인코딩 추가
        x = embedded + self.positional_encoding[:seq_len]
        
        print(f"임베딩 후 크기: {x.shape}")
        
        # 3. Transformer 레이어들 통과
        attention_maps = []
        
        for layer in range(self.num_layers):
            print(f"\n--- Transformer Layer {layer + 1} ---")
            
            # Self-Attention
            attn_output, attn_weights = self.attention_layers[layer].forward(x)
            attention_maps.append(attn_weights)
            
            # Residual connection + Layer Norm
            x = self.layer_norm(x + attn_output, 
                               self.layer_norms1[layer]['gamma'], 
                               self.layer_norms1[layer]['beta'])
            
            # Feed-Forward
            ff_output = self.feed_forward(x, self.ff_layers[layer])
            
            # Residual connection + Layer Norm
            x = self.layer_norm(x + ff_output,
                               self.layer_norms2[layer]['gamma'],
                               self.layer_norms2[layer]['beta'])
            
            print(f"레이어 {layer + 1} 출력 크기: {x.shape}")
        
        return x, attention_maps
    
    def visualize_positional_encoding(self):
        """위치 인코딩 시각화"""
        plt.figure(figsize=(12, 8))
        
        # 위치 인코딩 히트맵
        pe_sample = self.positional_encoding[:50, :self.embed_dim]
        
        plt.subplot(2, 1, 1)
        plt.imshow(pe_sample.T, cmap='RdBu', aspect='auto')
        plt.xlabel('Position')
        plt.ylabel('Embedding Dimension')
        plt.title('Positional Encoding')
        plt.colorbar()
        
        # 특정 차원들의 위치별 변화
        plt.subplot(2, 1, 2)
        positions = np.arange(50)
        for dim in [0, 1, 2, 3]:
            plt.plot(positions, pe_sample[:, dim], label=f'Dim {dim}')
        
        plt.xlabel('Position')
        plt.ylabel('Encoding Value')
        plt.title('Positional Encoding by Dimension')
        plt.legend()
        plt.grid(True, alpha=0.3)
        
        plt.tight_layout()
        plt.show()
```

```python
class StockPredictionRNN:
    """RNN을 사용한 주식 가격 예측"""

        self.scaler_min = None
        self.scaler_max = None
    
    def normalize_data(self, data):
        """데이터 정규화"""
        self.scaler_min = np.min(data)
        self.scaler_max = np.max(data)
        return (data - self.scaler_min) / (self.scaler_max - self.scaler_min)
    
    def denormalize_data(self, normalized_data):
        """정규화 해제"""
        return normalized_data * (self.scaler_max - self.scaler_min) + self.scaler_min
    
    def create_stock_data(self):
        """가상의 주식 데이터 생성"""
        np.random.seed(42)
        
        # 기본 트렌드 + 노이즈 + 계절성
        days = 200
        t = np.arange(days)
        
        trend = 0.02 * t  # 상승 트렌드
        seasonality = 5 * np.sin(2 * np.pi * t / 30)  # 30일 주기
        noise = np.random.randn(days) * 2
        
        stock_prices = 100 + trend + seasonality + noise
        
        return stock_prices
    
    def prepare_sequences(self, data, sequence_length):
        """시퀀스 데이터 준비"""
        X, y = [], []
        
        for i in range(len(data) - sequence_length):
            X.append(data[i:i+sequence_length])
            y.append(data[i+sequence_length])
        
        return np.array(X), np.array(y)

        print(f"시퀀스 길이: {sequence_length}")
        print(f"훈련 시퀀스: {len(X)}개")
        
        # 훈련/테스트 분할

        y_train, y_test = y[:train_size], y[train_size:]
        
        # 학습

        for epoch in range(200):

            for seq_x, target_y in zip(X_train, y_train):
                inputs = [np.array([x]) for x in seq_x]
                targets = [None] * (len(inputs) - 1) + [np.array([target_y])]
                
                loss = self.rnn.train_step(inputs, targets)
                total_loss += loss
            
            avg_loss = total_loss / len(X_train)
            losses.append(avg_loss)
            
            if epoch % 50 == 0:
                print(f"Epoch {epoch}: Loss = {avg_loss:.6f}")
        
        # 예측 테스트
        print("\n=== 주식 예측 테스트 ===")
        predictions = []
        actuals = []
        
        for seq_x, actual_y in zip(X_test[:10], y_test[:10]):  # 처음 10개만 테스트
            inputs = [np.array([x]) for x in seq_x]
            outputs, _ = self.rnn.forward(inputs)
            pred_y = outputs[-1].flatten()[0]
            
            predictions.append(pred_y)
            actuals.append(actual_y)
        
        # 정규화 해제
        pred_prices = self.denormalize_data(np.array(predictions))
        actual_prices = self.denormalize_data(np.array(actuals))
        
        # 결과 출력
        print(f"{'실제가격':<8} {'예측가격':<8} {'오차':<8}")
        print("-" * 25)
        for actual, pred in zip(actual_prices, pred_prices):
            error = abs(actual - pred)
            print(f"{actual:<8.2f} {pred:<8.2f} {error:<8.2f}")
        
        mse = np.mean((actual_prices - pred_prices)**2)
        print(f"\nMSE: {mse:.2f}")
        
        # 시각화
        plt.figure(figsize=(15, 10))
        
        # 전체 주식 데이터
        plt.subplot(2, 2, 1)
        plt.plot(stock_data)
        plt.axvline(x=train_size + sequence_length, color='r', linestyle='--', label='테스트 시작')
        plt.title('전체 주식 가격 데이터')
        plt.xlabel('일')
        plt.ylabel('가격')
        plt.legend()
        
        # 학습 곡선
        plt.subplot(2, 2, 2)
        plt.plot(losses)
        plt.title('학습 곡선')
        plt.xlabel('Epoch')
        plt.ylabel('Loss')
        
        # 예측 vs 실제
        plt.subplot(2, 2, 3)
        plt.plot(actual_prices, 'b-o', label='실제 가격')
        plt.plot(pred_prices, 'r-s', label='예측 가격')
        plt.title('예측 vs 실제 (테스트 데이터)')
        plt.xlabel('테스트 샘플')
        plt.ylabel('가격')
        plt.legend()
        
        # 오차 분포
        plt.subplot(2, 2, 4)
        errors = actual_prices - pred_prices
        plt.hist(errors, bins=10, alpha=0.7)
        plt.title('예측 오차 분포')
        plt.xlabel('오차 (실제 - 예측)')
        plt.ylabel('빈도')
        
        plt.tight_layout()
        plt.show()
        
        return stock_data, predictions, actuals
```
