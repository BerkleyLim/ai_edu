### 6.2 MDP (Markov Decision Process) 완전 이해

**MDP의 4가지 구성요소:**
```
S: 상태 집합 (States)
A: 행동 집합 (Actions) 
R: 보상 함수 (Rewards)
P: 전이 확률 (Transition Probabilities)
```

```python
class GridWorldMDP:
    """그리드 월드 MDP 구현"""
    
    def __init__(self, width=4, height=3):
        self.width = width
        self.height = height
        self.states = [(i, j) for i in range(height) for j in range(width)]
        self.actions = ['up', 'down', 'left', 'right']
        
        # 특수 상태들
        self.goal_state = (0, 3)      # 목표 (+10 보상)
        self.pit_state = (1, 3)       # 함정 (-10 보상)
        self.wall_state = (1, 1)      # 벽 (이동 불가)
        
        # 보상 설정
        self.rewards = {}
        for state in self.states:
            if state == self.goal_state:
                self.rewards[state] = 10
            elif state == self.pit_state:
                self.rewards[state] = -10
            elif state == self.wall_state:
                self.rewards[state] = 0  # 벽은 접근 불가
            else:
                self.rewards[state] = -1  # 기본적으로 -1 (빨리 끝내도록 유도)
    
    def get_next_state(self, state, action):
        """행동에 따른 다음 상태 계산"""
        if state in [self.goal_state, self.pit_state]:
            return state  # 종료 상태
        
        row, col = state
        
        # 행동에 따른 이동
        if action == 'up':
            next_state = (max(0, row - 1), col)
        elif action == 'down':
            next_state = (min(self.height - 1, row + 1), col)
        elif action == 'left':
            next_state = (row, max(0, col - 1))
        elif action == 'right':
            next_state = (row, min(self.width - 1, col + 1))
        
        # 벽으로 이동하려고 하면 제자리
        if next_state == self.wall_state:
            next_state = state
        
        return next_state
    
    def get_reward(self, state, action, next_state):
        """보상 계산"""
        return self.rewards[next_state]
    
    def is_terminal(self, state):
        """종료 상태인지 확인"""
        return state in [self.goal_state, self.pit_state]
    
    def visualize_grid(self, values=None, policy=None):
        """그리드 월드 시각화"""
        fig, ax = plt.subplots(figsize=(8, 6))
        
        # 그리드 그리기
        for i in range(self.height + 1):
            ax.axhline(i, color='black', linewidth=1)
        for j in range(self.width + 1):
            ax.axvline(j, color='black', linewidth=1)
        
        # 각 셀 색칠 및 정보 표시
        for state in self.states:
            row, col = state
            
            # 셀 색칠
            if state == self.goal_state:
                ax.fill_between([col, col+1], [self.height-row-1, self.height-row-1], 
                               [self.height-row, self.height-row], color='green', alpha=0.7)
                ax.text(col+0.5, self.height-row-0.5, 'GOAL\n+10', 
                       ha='center', va='center', fontsize=10, weight='bold')
            elif state == self.pit_state:
                ax.fill_between([col, col+1], [self.height-row-1, self.height-row-1],
                               [self.height-row, self.height-row], color='red', alpha=0.7)
                ax.text(col+0.5, self.height-row-0.5, 'PIT\n-10', 
                       ha='center', va='center', fontsize=10, weight='bold')
            elif state == self.wall_state:
                ax.fill_between([col, col+1], [self.height-row-1, self.height-row-1],
                               [self.height-row, self.height-row], color='gray', alpha=0.9)
                ax.text(col+0.5, self.height-row-0.5, 'WALL', 
                       ha='center', va='center', fontsize=10, weight='bold')
            else:
                # 가치함수 표시
                if values is not None and state in values:
                    value = values[state]
                    color_intensity = min(abs(value) / 10, 1)
                    color = 'lightblue' if value >= 0 else 'lightyellow'
                    ax.fill_between([col, col+1], [self.height-row-1, self.height-row-1],
                                   [self.height-row, self.height-row], color=color, alpha=color_intensity)
                    ax.text(col+0.5, self.height-row-0.3, f'{value:.1f}', 
                           ha='center', va='center', fontsize=8)
                
                # 정책 표시 (화살표)
                if policy is not None and state in policy:
                    action = policy[state]
                    arrow_dict = {
                        'up': '↑', 'down': '↓', 'left': '←', 'right': '→'
                    }
                    ax.text(col+0.5, self.height-row-0.7, arrow_dict[action], 
                           ha='center', va='center', fontsize=12, weight='bold', color='red')
        
        ax.set_xlim(0, self.width)
        ax.set_ylim(0, self.height)
        ax.set_aspect('equal')
        ax.set_title('Grid World MDP')
        ax.set_xticks([])
        ax.set_yticks([])
        
        plt.tight_layout()
        plt.show()
```

### 6.3 가치 반복 (Value Iteration) 구현

**벨만 방정식:**
```
V(s) = max_a Σ_{s'} P(s'|s,a)[R(s,a,s') + γV(s')]

γ: 할인인자 (discount factor)
```

```python
class ValueIteration:
    """가치 반복 알고리즘 구현"""
    
    def __init__(self, mdp, gamma=0.9, theta=1e-6):
        self.mdp = mdp
        self.gamma = gamma  # 할인인자
        self.theta = theta  # 수렴 임계값
        self.values = {state: 0.0 for state in mdp.states}  # 가치함수 초기화
    
    def value_iteration(self, max_iterations=1000):
        """가치 반복 알고리즘"""
        print("💰 가치 반복 알고리즘 실행")
        
        iteration_values = []  # 각 반복의 가치함수 저장
        
        for iteration in range(max_iterations):
            old_values = self.values.copy()
            max_delta = 0
            
            # 모든 상태에 대해 가치 업데이트
            for state in self.mdp.states:
                if self.mdp.is_terminal(state):
                    continue  # 종료 상태는 업데이트하지 않음
                
                # 모든 행동에 대해 Q값 계산
                action_values = []
                for action in self.mdp.actions:
                    next_state = self.mdp.get_next_state(state, action)
                    reward = self.mdp.get_reward(state, action, next_state)
                    
                    # 벨만 방정식: Q(s,a) = R + γV(s')
                    q_value = reward + self.gamma * old_values[next_state]
                    action_values.append(q_value)
                
                # 최대 Q값으로 가치 업데이트
                new_value = max(action_values)
                self.values[state] = new_value
                
                # 수렴 체크를 위한 델타 계산
                delta = abs(new_value - old_values[state])
                max_delta = max(max_delta, delta)
            
            iteration_values.append(self.values.copy())
            
            # 진행 상황 출력
            if iteration % 10 == 0 or iteration < 5:
                print(f"반복 {iteration}: 최대 변화량 = {max_delta:.6f}")
            
            # 수렴 체크
            if max_delta < self.theta:
                print(f"수렴 완료: {iteration + 1} 반복")
                break
        
        return iteration_values
    
    def extract_policy(self):
        """가치함수로부터 최적 정책 추출"""
        policy = {}
        
        for state in self.mdp.states:
            if self.mdp.is_terminal(state):
                continue
            
            best_action = None
            best_value = float('-inf')
            
            # 모든 행동 중 가장 높은 Q값을 가진 행동 선택
            for action in self.mdp.actions:
                next_state = self.mdp.get_next_state(state, action)
                reward = self.mdp.get_reward(state, action, next_state)
                q_value = reward + self.gamma * self.values[next_state]
                
                if q_value > best_value:
                    best_value = q_value
                    best_action = action
            
            policy[state] = best_action
        
        return policy
    
    def visualize_convergence(self, iteration_values):
        """가치함수 수렴 과정 시각화"""
        # 특정 상태들의 가치 변화 추적
        sample_states = [(0, 0), (2, 0), (0, 2), (2, 2)]
        
        plt.figure(figsize=(12, 8))
        
        # 가치함수 수렴 그래프
        plt.subplot(2, 2, 1)
        for state in sample_states:
            if state in iteration_values[0]:
                values_over_time = [iteration_values[i][state] for i in range(len(iteration_values))]
                plt.plot(values_over_time, label=f'State {state}', marker='o', markersize=3)
        
        plt.xlabel('반복 횟수')
        plt.ylabel('가치')
        plt.title('가치함수 수렴 과정')
        plt.legend()
        plt.grid(True, alpha=0.3)
        
        # 최종 가치함수 히트맵
        plt.subplot(2, 2, 2)
        value_matrix = np.zeros((self.mdp.height, self.mdp.width))
        for (row, col), value in self.values.items():
            value_matrix[row, col] = value
        
        im = plt.imshow(value_matrix, cmap='RdYlBu', aspect='auto')
        plt.title('최종 가치함수')
        plt.colorbar(im)
        
        # 정책 시각화를 위한 준비
        policy = self.extract_policy()
        
        plt.tight_layout()
        plt.show()
        
        return policy

# 가치 반복 실행
value_iter = ValueIteration(gridworld, gamma=0.9)
iteration_values = value_iter.value_iteration()

# 결과 시각화
policy = value_iter.visualize_convergence(iteration_values)

# 최종 결과 출력
print("\n=== 최적 가치함수 ===")
for state in gridworld.states:
    if not gridworld.is_terminal(state) and state != gridworld.wall_state:
        print(f"상태 {state}: {value_iter.values[state]:.2f}")

print("\n=== 최적 정책 ===")
for state, action in policy.items():
    print(f"상태 {state}: {action}")

# 최종 그리드 시각화
gridworld.visualize_grid(values=value_iter.values, policy=policy)
```

### 6.4 Q-Learning 알고리즘 구현

**Q-Learning 업데이트 규칙:**
```
Q(s,a) ← Q(s,a) + α[r + γ max Q(s',a') - Q(s,a)]

α: 학습률 (learning rate)
γ: 할인인자 (discount factor)
```

```python
class QLearningAgent:
    """Q-Learning 에이전트 구현"""
    
    def __init__(self, mdp, alpha=0.1, gamma=0.9, epsilon=0.1):
        self.mdp = mdp
        self.alpha = alpha      # 학습률
        self.gamma = gamma      # 할인인자
        self.epsilon = epsilon  # 탐험 확률
        
        # Q-테이블 초기화
        self.q_table = {}
        for state in mdp.states:
            self.q_table[state] = {}
            for action in mdp.actions:
                self.q_table[state][action] = 0.0
        
        # 학습 기록
        self.episode_rewards = []
        self.episode_lengths = []
    
    def get_action(self, state, training=True):
        """ε-greedy 정책으로 행동 선택"""
        if self.mdp.is_terminal(state):
            return None
        
        # 탐험 vs 활용
        if training and np.random.random() < self.epsilon:
            # 탐험: 랜덤 행동
            return np.random.choice(self.mdp.actions)
        else:
            # 활용: 가장 높은 Q값의 행동
            q_values = self.q_table[state]
            max_q = max(q_values.values())
            best_actions = [action for action, q in q_values.items() if q == max_q]
            return np.random.choice(best_actions)
    
    def update_q_value(self, state, action, reward, next_state):
        """Q값 업데이트"""
        if self.mdp.is_terminal(next_state):
            # 종료 상태의 경우 미래 가치는 0
            max_next_q = 0
        else:
            # 다음 상태에서 최대 Q값
            max_next_q = max(self.q_table[next_state].values())
        
        # Q-Learning 업데이트 규칙
        old_q = self.q_table[state][action]
        self.q_table[state][action] = old_q + self.alpha * (
            reward + self.gamma * max_next_q - old_q
        )
    
    def train(self, episodes=1000, start_state=(2, 0)):
        """Q-Learning 학습"""
        print("🎮 Q-Learning 학습 시작")
        
        for episode in range(episodes):
            state = start_state
            total_reward = 0
            steps = 0
            
            while not self.mdp.is_terminal(state) and steps < 100:  # 무한 루프 방지
                # 행동 선택
                action = self.get_action(state, training=True)
                
                # 환경과 상호작용
                next_state = self.mdp.get_next_state(state, action)
                reward = self.mdp.get_reward(state, action, next_state)
                
                # Q값 업데이트
                self.update_q_value(state, action, reward, next_state)
                
                # 다음 상태로 이동
                state = next_state
                total_reward += reward
                steps += 1
            
            # 에피소드 기록
            self.episode_rewards.append(total_reward)
            self.episode_lengths.append(steps)
            
            # 진행 상황 출력
            if episode % 100 == 0:
                avg_reward = np.mean(self.episode_rewards[-100:])
                avg_length = np.mean(self.episode_lengths[-100:])
                print(f"에피소드 {episode}: 평균 보상 = {avg_reward:.2f}, 평균 길이 = {avg_length:.1f}")
        
        print("학습 완료!")
    
    def extract_policy_from_q(self):
        """Q-테이블에서 정책 추출"""
        policy = {}
        
        for state in self.mdp.states:
            if self.mdp.is_terminal(state):
                continue
            
            q_values = self.q_table[state]
            best_action = max(q_values, key=q_values.get)
            policy[state] = best_action
        
        return policy
    
    def visualize_learning(self):
        """학습 과정 시각화"""
        plt.figure(figsize=(15, 10))
        
        # 에피소드별 보상
        plt.subplot(2, 3, 1)
        plt.plot(self.episode_rewards, alpha=0.3, color='blue')
        # 이동 평균
        window = 50
        if len(self.episode_rewards) >= window:
            moving_avg = np.convolve(self.episode_rewards, np.ones(window)/window, mode='valid')
            plt.plot(range(window-1, len(self.episode_rewards)), moving_avg, color='red', linewidth=2)
        plt.xlabel('에피소드')
        plt.ylabel('총 보상')
        plt.title('학습 진행: 에피소드별 보상')
        plt.grid(True, alpha=0.3)
        
        # 에피소드 길이
        plt.subplot(2, 3, 2)
        plt.plot(self.episode_lengths, alpha=0.3, color='green')
        if len(self.episode_lengths) >= window:
            moving_avg_length = np.convolve(self.episode_lengths, np.ones(window)/window, mode='valid')
            plt.plot(range(window-1, len(self.episode_lengths)), moving_avg_length, color='orange', linewidth=2)
        plt.xlabel('에피소드')
        plt.ylabel('스텝 수')
        plt.title('학습 진행: 에피소드 길이')
        plt.grid(True, alpha=0.3)
        
        # Q-테이블 히트맵 (특정 행동에 대해)
        for i, action in enumerate(self.mdp.actions[:4]):  # 최대 4개 행동만
            plt.subplot(2, 3, 3+i)
            
            q_matrix = np.zeros((self.mdp.height, self.mdp.width))
            for (row, col) in self.mdp.states:
                q_matrix[row, col] = self.q_table[(row, col)][action]
            
            im = plt.imshow(q_matrix, cmap='RdYlBu', aspect='auto')
            plt.title(f'Q-값: {action}')
            plt.colorbar(im, shrink=0.8)
        
        plt.tight_layout()
        plt.show()
    
    def test_agent(self, start_state=(2, 0), max_steps=20):
        """학습된 에이전트 테스트"""
        print(f"\n🧪 학습된 에이전트 테스트 (시작: {start_state})")
        
        state = start_state
        path = [state]
        total_reward = 0
        
        for step in range(max_steps):
            if self.mdp.is_terminal(state):
                break
                
            action = self.get_action(state, training=False)  # 탐험 없이
            next_state = self.mdp.get_next_state(state, action)
            reward = self.mdp.get_reward(state, action, next_state)
```

### 6.5 Deep Q-Network (DQN) 구현

```python
class DQN:
    """Deep Q-Network 구현"""
    
    def __init__(self, state_size, action_size, hidden_size=64, learning_rate=0.001):
        self.state_size = state_size
        self.action_size = action_size
        self.learning_rate = learning_rate
        
        # 신경망 파라미터 (간단한 2층 네트워크)
        self.W1 = np.random.randn(state_size, hidden_size) * np.sqrt(2.0 / state_size)
        self.b1 = np.zeros((1, hidden_size))
        self.W2 = np.random.randn(hidden_size, hidden_size) * np.sqrt(2.0 / hidden_size)
        self.b2 = np.zeros((1, hidden_size))
        self.W3 = np.random.randn(hidden_size, action_size) * np.sqrt(2.0 / hidden_size)
        self.b3 = np.zeros((1, action_size))
        
        # 경험 재생 버퍼
        self.memory = []
        self.memory_size = 10000
        self.batch_size = 32
        
        # DQN 하이퍼파라미터
        self.gamma = 0.95
        self.epsilon = 1.0
        self.epsilon_min = 0.01
        self.epsilon_decay = 0.995
    
    def relu(self, x):
        return np.maximum(0, x)
    
    def relu_derivative(self, x):
        return (x > 0).astype(float)
    
    def forward(self, state):
        """신경망 순전파"""
        self.z1 = np.dot(state, self.W1) + self.b1
        self.a1 = self.relu(self.z1)
        self.z2 = np.dot(self.a1, self.W2) + self.b2
        self.a2 = self.relu(self.z2)
        self.z3 = np.dot(self.a2, self.W3) + self.b3
        return self.z3  # Q-값들
    
    def predict(self, state):
        """Q값 예측"""
        if len(state.shape) == 1:
            state = state.reshape(1, -1)
        return self.forward(state)
    
    def get_action(self, state, training=True):
        """ε-greedy 행동 선택"""
        if training and np.random.random() < self.epsilon:
            return np.random.randint(self.action_size)
        else:
            q_values = self.predict(state)
            return np.argmax(q_values[0])
    
    def remember(self, state, action, reward, next_state, done):
        """경험을 메모리에 저장"""
        if len(self.memory) >= self.memory_size:
            self.memory.pop(0)
        
        self.memory.append((state, action, reward, next_state, done))
    
    def train_step(self, state, action, target_q):
        """한 스텝 학습"""
        # 순전파
        q_values = self.forward(state)
        
        # 손실 계산 (MSE)
        target = q_values.copy()
        target[0, action] = target_q
        
        loss = 0.5 * np.sum((q_values - target) ** 2)
        
        # 역전파
        dq = q_values - target
        
        # 출력층 기울기
        dW3 = np.dot(self.a2.T, dq)
        db3 = np.sum(dq, axis=0, keepdims=True)
        
        # 은닉층 2 기울기
        da2 = np.dot(dq, self.W3.T)
        dz2 = da2 * self.relu_derivative(self.z2)
        dW2 = np.dot(self.a1.T, dz2)
        db2 = np.sum(dz2, axis=0, keepdims=True)
        
        # 은닉층 1 기울기
        da1 = np.dot(dz2, self.W2.T)
        dz1 = da1 * self.relu_derivative(self.z1)
        dW1 = np.dot(state.T, dz1)
        db1 = np.sum(dz1, axis=0, keepdims=True)
        
        # 파라미터 업데이트
        self.W3 -= self.learning_rate * dW3
        self.b3 -= self.learning_rate * db3
        self.W2 -= self.learning_rate * dW2
        self.b2 -= self.learning_rate * db2
        self.W1 -= self.learning_rate * dW1
        self.b1 -= self.learning_rate * db1
        
        return loss
    
    def replay(self):
        """경험 재생 학습"""
        if len(self.memory) < self.batch_size:
            return
        
        # 배치 샘플링
        batch = np.random.choice(len(self.memory), self.batch_size, replace=False)
        
        total_loss = 0
        for i in batch:
            state, action, reward, next_state, done = self.memory[i]
            
            if done:
                target_q = reward
            else:
                target_q = reward + self.gamma * np.max(self.predict(next_state))
            
            loss = self.train_step(state.reshape(1, -1), action, target_q)
            total_loss += loss
        
        # ε 감소
        if self.epsilon > self.epsilon_min:
            self.epsilon *= self.epsilon_decay
        
        return total_loss / self.batch_size

class DQNGridWorldAgent:

        self.episode_lengths = []
        self.losses = []

        """상태를 벡터로 변환"""
        row, col = state
        return np.array([row / self.gridworld.height, col / self.gridworld.width])
    
    def train_dqn(self, episodes=1000, start_state=(2, 0)):
        """DQN 학습"""
        print("🧠 Deep Q-Network 학습 시작")
        
        for episode in range(episodes):
            state = start_state
            state_vec = self.state_to_vector(state)
            total_reward = 0
            steps = 0
            
            while not self.gridworld.is_terminal(state) and steps < 100:
                # 행동 선택
                action_idx = self.dqn.get_action(state_vec)
                action = self.action_map[action_idx]
                
                # 환경과 상호작용
                next_state = self.gridworld.get_next_state(state, action)
                reward = self.gridworld.get_reward(state, action, next_state)
                done = self.gridworld.is_terminal(next_state)
                
                next_state_vec = self.state_to_vector(next_state)
                
                # 경험 저장
                self.dqn.remember(state_vec, action_idx, reward, next_state_vec, done)
                
                # 학습
                if len(self.dqn.memory) > self.dqn.batch_size:
                    loss = self.dqn.replay()
                    if loss is not None:
                        self.losses.append(loss)
                
                state = next_state
                state_vec = next_state_vec
                total_reward += reward
                steps += 1
            
            self.episode_rewards.append(total_reward)
            self.episode_lengths.append(steps)
            
            # 진행 상황 출력
            if episode % 100 == 0:
                avg_reward = np.mean(self.episode_rewards[-100:])
                avg_length = np.mean(self.episode_lengths[-100:])
```

### 7.2 VAE (Variational Autoencoder) 구현

**VAE의 핵심 아이디어:**
```
1. 데이터를 잠재 공간(latent space)으로 인코딩
2. 잠재 공간에서 샘플링
3. 샘플을 다시 데이터로 디코딩
4. KL divergence로 잠재 분포를 정규화
```

```python
class VAE:
    """Variational Autoencoder 구현"""
    
    def __init__(self, input_dim, latent_dim, hidden_dim=128):
        self.input_dim = input_dim
        self.latent_dim = latent_dim
        self.hidden_dim = hidden_dim
        
        # 인코더 파라미터
        self.encoder_W1 = np.random.randn(input_dim, hidden_dim) * 0.1
        self.encoder_b1 = np.zeros((1, hidden_dim))
        
        # 평균과 분산을 위한 층
        self.mu_W = np.random.randn(hidden_dim, latent_dim) * 0.1
        self.mu_b = np.zeros((1, latent_dim))
        self.logvar_W = np.random.randn(hidden_dim, latent_dim) * 0.1
        self.logvar_b = np.zeros((1, latent_dim))
        
        # 디코더 파라미터
        self.decoder_W1 = np.random.randn(latent_dim, hidden_dim) * 0.1
        self.decoder_b1 = np.zeros((1, hidden_dim))
        self.decoder_W2 = np.random.randn(hidden_dim, input_dim) * 0.1
        self.decoder_b2 = np.zeros((1, input_dim))
        
        self.learning_rate = 0.001
    
    def relu(self, x):
        return np.maximum(0, x)
    
    def sigmoid(self, x):
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
    
    def encode(self, x):
        """인코더: x -> (mu, logvar)"""
        h = self.relu(np.dot(x, self.encoder_W1) + self.encoder_b1)
        mu = np.dot(h, self.mu_W) + self.mu_b
        logvar = np.dot(h, self.logvar_W) + self.logvar_b
        return mu, logvar
    
    def reparameterize(self, mu, logvar):
        """재매개화 트릭: mu + std * epsilon"""
        std = np.exp(0.5 * logvar)
        epsilon = np.random.normal(0, 1, mu.shape)
        return mu + std * epsilon
    
    def decode(self, z):
        """디코더: z -> x"""
        h = self.relu(np.dot(z, self.decoder_W1) + self.decoder_b1)
        reconstructed = self.sigmoid(np.dot(h, self.decoder_W2) + self.decoder_b2)
        return reconstructed
    
    def forward(self, x):
        """순전파"""
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        reconstructed = self.decode(z)
        return reconstructed, mu, logvar, z
    
    def compute_loss(self, x, reconstructed, mu, logvar):
        """VAE 손실: 재구성 손실 + KL divergence"""
        batch_size = x.shape[0]
        
        # 재구성 손실 (Binary Cross Entropy)
        reconstruction_loss = -np.sum(
            x * np.log(reconstructed + 1e-8) + (1 - x) * np.log(1 - reconstructed + 1e-8)
        ) / batch_size
        
        # KL divergence: KL(q(z|x) || p(z))
        # p(z) = N(0,I), q(z|x) = N(mu, exp(logvar))
        kl_loss = -0.5 * np.sum(1 + logvar - mu**2 - np.exp(logvar)) / batch_size
        
        total_loss = reconstruction_loss + kl_loss
        
        return total_loss, reconstruction_loss, kl_loss
    
    def generate(self, num_samples):
        """새로운 샘플 생성"""
        # 표준 정규분포에서 샘플링
        z = np.random.normal(0, 1, (num_samples, self.latent_dim))
        generated = self.decode(z)
        return generated
    
    def visualize_latent_space(self, data, labels=None):
        """2D 잠재 공간 시각화 (latent_dim=2인 경우)"""
        if self.latent_dim != 2:
            print("잠재 차원이 2가 아니어서 시각화할 수 없습니다.")
            return
        
        mu, _ = self.encode(data)
        
        plt.figure(figsize=(10, 8))
        if labels is not None:
            scatter = plt.scatter(mu[:, 0], mu[:, 1], c=labels, cmap='tab10', alpha=0.7)
            plt.colorbar(scatter)
        else:
            plt.scatter(mu[:, 0], mu[:, 1], alpha=0.7)
        
        plt.xlabel('Latent Dimension 1')
        plt.ylabel('Latent Dimension 2')
        plt.title('VAE Latent Space')
        plt.grid(True, alpha=0.3)
        plt.show()

# 간단한 2D 데이터로 VAE 테스트
def test_vae_2d():
    """2D 데이터로 VAE 테스트"""
    print("🎭 VAE 2D 데이터 테스트")
    
    # 간단한 2D 데이터 생성 (두 개의 클러스터)
    np.random.seed(42)
    
    cluster1 = np.random.normal([2, 2], 0.5, (200, 2))
    cluster2 = np.random.normal([6, 6], 0.5, (200, 2))
    data = np.vstack([cluster1, cluster2])
    labels = np.array([0]*200 + [1]*200)
    
    # 데이터 정규화
    data = (data - np.min(data)) / (np.max(data) - np.min(data))
    
    # VAE 생성 (2D -> 2D latent)
    vae = VAE(input_dim=2, latent_dim=2, hidden_dim=16)
    
    # 학습
    print("VAE 학습 중...")
    losses = []
    recon_losses = []
    kl_losses = []
    
    for epoch in range(1000):
        # 미니배치 (전체 데이터 사용)
        reconstructed, mu, logvar, z = vae.forward(data)
        
        total_loss, recon_loss, kl_loss = vae.compute_loss(data, reconstructed, mu, logvar)
        
        losses.append(total_loss)
```

### 7.3 GAN (Generative Adversarial Networks) 구현

**GAN의 핵심 아이디어:**
```
생성자(Generator) vs 판별자(Discriminator)
- 생성자: 가짜 데이터를 만들어서 판별자를 속이려 함
- 판별자: 진짜와 가짜를 구별하려 함
- 미니맥스 게임을 통해 서로 발전
```

```python
class SimpleGAN:
    """간단한 GAN 구현"""
    
    def __init__(self, noise_dim=10, data_dim=2, hidden_dim=16):
        self.noise_dim = noise_dim
        self.data_dim = data_dim
        self.hidden_dim = hidden_dim
        
        # 생성자 (Generator) 파라미터
        self.G_W1 = np.random.randn(noise_dim, hidden_dim) * 0.1
        self.G_b1 = np.zeros((1, hidden_dim))
        self.G_W2 = np.random.randn(hidden_dim, data_dim) * 0.1
        self.G_b2 = np.zeros((1, data_dim))
        
        # 판별자 (Discriminator) 파라미터
        self.D_W1 = np.random.randn(data_dim, hidden_dim) * 0.1
        self.D_b1 = np.zeros((1, hidden_dim))
        self.D_W2 = np.random.randn(hidden_dim, 1) * 0.1
        self.D_b2 = np.zeros((1, 1))
        
        self.learning_rate = 0.001
        
        # 학습 기록
        self.g_losses = []
        self.d_losses = []
    
    def relu(self, x):
        return np.maximum(0, x)
    
    def sigmoid(self, x):
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
    
    def tanh(self, x):
        return np.tanh(x)
    
    def generator(self, noise):
        """생성자: 노이즈 -> 가짜 데이터"""
        h = self.relu(np.dot(noise, self.G_W1) + self.G_b1)
        output = self.tanh(np.dot(h, self.G_W2) + self.G_b2)  # tanh로 [-1,1] 범위
        return output
    
    def discriminator(self, data):
        """판별자: 데이터 -> 진짜(1) 또는 가짜(0) 확률"""
        h = self.relu(np.dot(data, self.D_W1) + self.D_b1)
        output = self.sigmoid(np.dot(h, self.D_W2) + self.D_b2)
        return output
    
    def train_discriminator(self, real_data, fake_data):
        """판별자 학습"""
        batch_size = real_data.shape[0]
        
        # 판별자 순전파
        real_pred = self.discriminator(real_data)
        fake_pred = self.discriminator(fake_data)
        
        # 판별자 손실: 진짜는 1로, 가짜는 0으로 분류
        d_loss_real = -np.mean(np.log(real_pred + 1e-8))
        d_loss_fake = -np.mean(np.log(1 - fake_pred + 1e-8))
        d_loss = d_loss_real + d_loss_fake
        
        # 간단한 기울기 계산 (실제로는 자동미분 사용)
        # 여기서는 개념적 구현
        
        return d_loss
    
    def train_generator(self, noise):
        """생성자 학습"""
        # 생성자로 가짜 데이터 생성
        fake_data = self.generator(noise)
        
        # 판별자로 평가
        fake_pred = self.discriminator(fake_data)
        
        # 생성자 손실: 가짜를 진짜로 속이고 싶음
        g_loss = -np.mean(np.log(fake_pred + 1e-8))
        
        return g_loss
    
    def train_step(self, real_data):
        """한 스텝 학습"""
        batch_size = real_data.shape[0]
        
        # 노이즈 생성
        noise = np.random.normal(0, 1, (batch_size, self.noise_dim))
        
        # 가짜 데이터 생성
        fake_data = self.generator(noise)
        
        # 판별자 학습
        d_loss = self.train_discriminator(real_data, fake_data)
        
        # 생성자 학습
        g_loss = self.train_generator(noise)
        
        self.d_losses.append(d_loss)
        self.g_losses.append(g_loss)
        
        return d_loss, g_loss
    
    def generate_samples(self, num_samples):
        """샘플 생성"""
        noise = np.random.normal(0, 1, (num_samples, self.noise_dim))
        return self.generator(noise)
```

### 7.4 Diffusion Models 기본 개념

**Diffusion의 핵심 아이디어:**
```
1. Forward Process: 데이터에 점진적으로 노이즈 추가
2. Reverse Process: 노이즈에서 점진적으로 데이터 복원
3. 신경망이 reverse process를 학습
```

```python
class SimpleDiffusion:
    """단순화된 Diffusion Model"""
    
    def __init__(self, num_timesteps=100):
        self.num_timesteps = num_timesteps
        
        # 베타 스케줄 (노이즈 추가 정도)
        self.betas = np.linspace(0.0001, 0.02, num_timesteps)
        self.alphas = 1 - self.betas
        self.alpha_cumprod = np.cumprod(self.alphas)
        
        # 역과정을 위한 신경망 (매우 간단한 버전)
        self.denoising_net = self._create_simple_denoising_net()
    
    def _create_simple_denoising_net(self):
        """간단한 디노이징 네트워크"""
        # 실제로는 U-Net 같은 복잡한 구조 사용
        return {
            'W1': np.random.randn(3, 16) * 0.1,  # (data_dim + time_dim, hidden)
            'b1': np.zeros((1, 16)),
            'W2': np.random.randn(16, 16) * 0.1,
            'b2': np.zeros((1, 16)),
            'W3': np.random.randn(16, 2) * 0.1,   # (hidden, data_dim)
            'b3': np.zeros((1, 2))
        }
    
    def add_noise(self, x0, t):
        """순방향 과정: 데이터에 노이즈 추가"""
        # x_t = sqrt(alpha_cumprod_t) * x_0 + sqrt(1 - alpha_cumprod_t) * noise
        
        noise = np.random.normal(0, 1, x0.shape)
        alpha_t = self.alpha_cumprod[t]
        
        noisy_x = np.sqrt(alpha_t) * x0 + np.sqrt(1 - alpha_t) * noise
        
        return noisy_x, noise
    
    def predict_noise(self, xt, t):
        """신경망으로 노이즈 예측"""
        # 시간 정보를 포함한 입력
        t_embed = np.array([[t / self.num_timesteps]])  # 정규화된 시간
        
        # 입력 연결 (xt + time embedding)
        if len(xt.shape) == 1:
            xt = xt.reshape(1, -1)
        
        input_data = np.concatenate([xt, t_embed], axis=1)
        
        # 간단한 MLP
        h1 = np.tanh(np.dot(input_data, self.denoising_net['W1']) + self.denoising_net['b1'])
        h2 = np.tanh(np.dot(h1, self.denoising_net['W2']) + self.denoising_net['b2'])
        predicted_noise = np.dot(h2, self.denoising_net['W3']) + self.denoising_net['b3']
        
        return predicted_noise
    
    def denoise_step(self, xt, t):
        """한 스텝 디노이징"""
        # 노이즈 예측
        predicted_noise = self.predict_noise(xt, t)
        
        # 디노이징 공식 (단순화)
        alpha_t = self.alphas[t]
        alpha_cumprod_t = self.alpha_cumprod[t]
        
        if t > 0:
            alpha_cumprod_prev = self.alpha_cumprod[t-1]
        else:
            alpha_cumprod_prev = 1.0
        
        # x_{t-1} 계산
        coeff1 = 1 / np.sqrt(alpha_t)
        coeff2 = (1 - alpha_t) / np.sqrt(1 - alpha_cumprod_t)
        
        x_prev = coeff1 * (xt - coeff2 * predicted_noise)
        
        # 노이즈 추가 (t > 0인 경우)
        if t > 0:
            noise = np.random.normal(0, 1, xt.shape)
            sigma = np.sqrt((1 - alpha_cumprod_prev) / (1 - alpha_cumprod_t) * (1 - alpha_t))
            x_prev += sigma * noise
        
        return x_prev
    
    def sample(self, shape):
        """샘플 생성 (순수 노이즈에서 시작)"""
        # 순수 노이즈에서 시작
        xt = np.random.normal(0, 1, shape)
        
        # 역방향 과정
        for t in reversed(range(self.num_timesteps)):
            xt = self.denoise_step(xt, t)
        
        return xt
    
    def visualize_diffusion_process(self, x0):
        """디퓨전 과정 시각화"""
        timesteps_to_show = [0, 10, 25, 50, 75, 99]
        
        plt.figure(figsize=(15, 8))
        
        # 순방향 과정 (노이즈 추가)
        for i, t in enumerate(timesteps_to_show):
            plt.subplot(2, len(timesteps_to_show), i + 1)
            
            if t == 0:
                noisy_x = x0
            else:
                noisy_x, _ = self.add_noise(x0, t)
            
            if len(x0.shape) == 2:  # 2D 데이터
                plt.scatter(noisy_x[:, 0], noisy_x[:, 1], alpha=0.6, s=10)
            else:  # 1D 데이터
                plt.hist(noisy_x, bins=30, alpha=0.7)
            
            plt.title(f'Forward t={t}')
            if i == 0:
                plt.ylabel('Forward Process\n(노이즈 추가)')
        
        # 역방향 과정 시뮬레이션 (매우 간단한 버전)
        for i, t in enumerate(timesteps_to_show):
            plt.subplot(2, len(timesteps_to_show), len(timesteps_to_show) + i + 1)
            
            # 간단한 역방향 시뮬레이션
            if t == 99:
                # 순수 노이즈
                reverse_x = np.random.normal(0, 1, x0.shape)
            else:
                # 점진적 디노이징 (개념적)
                noise_level = t / 99.0
                reverse_x = x0 + np.random.normal(0, noise_level, x0.shape)
            
            if len(x0.shape) == 2:  # 2D 데이터
                plt.scatter(reverse_x[:, 0], reverse_x[:, 1], alpha=0.6, s=10, color='red')
            else:  # 1D 데이터
                plt.hist(reverse_x, bins=30, alpha=0.7, color='red')
            
            plt.title(f'Reverse t={99-t}')
            if i == 0:
                plt.ylabel('Reverse Process\n(디노이징)')
        
        plt.tight_layout()
        plt.show()
```

### 8.1 LoRA (Low-Rank Adaptation) 구현

**LoRA의 핵심 아이디어:**
```
원본 가중치 W를 고정하고, 작은 행렬 A, B로 업데이트
W' = W + BA
여기서 A는 r×d, B는 d×r (r << d)
```

```python
class LoRALayer:
    """LoRA (Low-Rank Adaptation) 레이어 구현"""
    
    def __init__(self, original_weight, rank=4, alpha=16):
        """
        LoRA 레이어 초기화
        
        Args:
            original_weight: 원본 가중치 행렬 (출력_dim x 입력_dim)
            rank: 저랭크 차원 (r)
            alpha: 스케일링 인수
        """
        self.original_weight = original_weight.copy()  # 원본은 고정
        self.rank = rank
        self.alpha = alpha
        
        out_dim, in_dim = original_weight.shape
        
        # LoRA 매개변수 A, B 초기화
        # A: Gaussian 초기화, B: 0 초기화
        self.lora_A = np.random.normal(0, 0.1, (rank, in_dim))
        self.lora_B = np.zeros((out_dim, rank))
        
        self.scale = alpha / rank
    
    def get_weight(self):
        """현재 가중치 계산: W + α/r * B * A"""
        delta_W = self.scale * np.dot(self.lora_B, self.lora_A)
        return self.original_weight + delta_W
    
    def forward(self, x):
        """순전파"""
        # 원본 경로
        original_output = np.dot(x, self.original_weight.T)
        
        # LoRA 경로
        lora_output = np.dot(x, self.lora_A.T)  # x * A^T
        lora_output = np.dot(lora_output, self.lora_B.T)  # * B^T
        lora_output = lora_output * self.scale
        
        return original_output + lora_output
    
    def get_trainable_params(self):
        """학습 가능한 매개변수 반환"""
        return {'lora_A': self.lora_A, 'lora_B': self.lora_B}
    
    def count_parameters(self):
        """매개변수 수 비교"""
        original_params = np.prod(self.original_weight.shape)
        lora_params = np.prod(self.lora_A.shape) + np.prod(self.lora_B.shape)
        
        return {
            'original': original_params,
            'lora': lora_params,
            'ratio': lora_params / original_params * 100
        }

class LoRAMLP:
    """LoRA를 적용한 MLP"""
    
    def __init__(self, layer_dims, lora_ranks=None, lora_alpha=16):
        self.layer_dims = layer_dims
        self.lora_ranks = lora_ranks or [4] * (len(layer_dims) - 1)
        self.lora_alpha = lora_alpha
        
        # 원본 네트워크 초기화
        self.layers = []
        for i in range(len(layer_dims) - 1):
            # 원본 가중치
            weight = np.random.randn(layer_dims[i+1], layer_dims[i]) * 0.1
            
            # LoRA 레이어 생성
            lora_layer = LoRALayer(weight, rank=self.lora_ranks[i], alpha=lora_alpha)
            self.layers.append(lora_layer)
        
        self.learning_rate = 0.001
    
    def forward(self, x):
        """순전파"""
        activations = [x]
        
        for i, layer in enumerate(self.layers):
            z = layer.forward(activations[-1])
            
            # 마지막 층이 아니면 ReLU 적용
            if i < len(self.layers) - 1:
                a = np.maximum(0, z)
            else:
                a = 1 / (1 + np.exp(-np.clip(z, -500, 500)))  # 시그모이드
            
            activations.append(a)
        
        return activations[-1], activations
    
    def get_parameter_efficiency(self):
        """매개변수 효율성 분석"""
        total_original = 0
        total_lora = 0
        
        print("📊 LoRA 매개변수 효율성 분석")
        print("=" * 50)
        
        for i, layer in enumerate(self.layers):
            params = layer.count_parameters()
            total_original += params['original']
            total_lora += params['lora']
            
            print(f"레이어 {i+1}:")
            print(f"  원본 매개변수: {params['original']:,}")
            print(f"  LoRA 매개변수: {params['lora']:,}")
            print(f"  비율: {params['ratio']:.2f}%")
            print()
        
        overall_ratio = total_lora / total_original * 100
        print(f"전체 비교:")
        print(f"  원본 총 매개변수: {total_original:,}")
        print(f"  LoRA 총 매개변수: {total_lora:,}")
        print(f"  전체 비율: {overall_ratio:.2f}%")
        print(f"  매개변수 감소: {100 - overall_ratio:.2f}%")
        
        return overall_ratio
    
    def visualize_weight_updates(self):
        """가중치 업데이트 시각화"""
```

```python
class LIME_Explainer:
    """LIME (Local Interpretable Model-agnostic Explanations) 구현"""
    
    def __init__(self, model, feature_names=None):
        self.model = model
        self.feature_names = feature_names or [f'Feature_{i}' for i in range(10)]
    
    def explain_instance(self, instance, num_samples=1000, num_features=5):
        """단일 인스턴스에 대한 설명 생성"""
        
        # 1. 인스턴스 주변에서 샘플링
        samples = self._generate_samples(instance, num_samples)
        
        # 2. 각 샘플에 대해 모델 예측
        predictions = []
        for sample in samples:
            pred = self.model(sample.reshape(1, -1))[0]
            predictions.append(pred)
        
        predictions = np.array(predictions)
        
        # 3. 거리 기반 가중치 계산
        distances = np.linalg.norm(samples - instance, axis=1)
        weights = np.exp(-distances**2 / (2 * 0.25**2))  # RBF 커널
        
        # 4. 선형 모델 학습
        X_weighted = samples * weights.reshape(-1, 1)
        y_weighted = predictions * weights
        
        # 가중 최소제곱법으로 선형 회귀
        coeffs = self._weighted_linear_regression(X_weighted, y_weighted, weights)
        
        # 5. 중요도가 높은 특성 선택
        importance_scores = np.abs(coeffs)
        top_features = np.argsort(importance_scores)[-num_features:]
        
        explanation = {
            'coefficients': coeffs,
            'top_features': top_features,
            'feature_importance': importance_scores,
            'instance': instance,
            'prediction': self.model(instance.reshape(1, -1))[0]
        }
        
        return explanation
    
    def _generate_samples(self, instance, num_samples):
        """인스턴스 주변 샘플 생성"""
        # 각 특성에 대해 노이즈 추가
        samples = []
        for _ in range(num_samples):
            noise = np.random.normal(0, 0.3, len(instance))
            sample = instance + noise
            samples.append(sample)
        
        return np.array(samples)
    
    def _weighted_linear_regression(self, X, y, weights):
        """가중 선형 회귀"""
        # W^T * W * β = W^T * y (가중 정규방정식)
        W = np.diag(weights)
        XtWX = X.T @ W @ X + 1e-6 * np.eye(X.shape[1])  # 정규화 추가
        XtWy = X.T @ W @ y
        
        try:
            coeffs = np.linalg.solve(XtWX, XtWy)
        except np.linalg.LinAlgError:
            coeffs = np.linalg.lstsq(XtWX, XtWy, rcond=None)[0]
        
        return coeffs
    
    def visualize_explanation(self, explanation):
        """설명 시각화"""
        coeffs = explanation['coefficients']
        top_features = explanation['top_features']
        instance = explanation['instance']
        
        plt.figure(figsize=(15, 8))
        
        # 특성 중요도
        plt.subplot(2, 3, 1)
        feature_names = [self.feature_names[i] for i in top_features]
        feature_coeffs = [coeffs[i] for i in top_features]
        
        colors = ['red' if c < 0 else 'blue' for c in feature_coeffs]
        plt.barh(range(len(feature_coeffs)), feature_coeffs, color=colors, alpha=0.7)
        plt.yticks(range(len(feature_coeffs)), feature_names)
        plt.xlabel('LIME 계수')
        plt.title('특성 중요도 (LIME)')
        plt.axvline(x=0, color='black', linestyle='-', alpha=0.3)
        
        # 인스턴스 값들
        plt.subplot(2, 3, 2)
        plt.bar(range(len(instance)), instance, alpha=0.7, color='green')
        plt.xlabel('특성 인덱스')
        plt.ylabel('특성 값')
        plt.title('설명 대상 인스턴스')
        
        # 전체 특성 중요도
        plt.subplot(2, 3, 3)
        plt.bar(range(len(coeffs)), np.abs(coeffs), alpha=0.7)
        plt.xlabel('특성 인덱스')
        plt.ylabel('절대 중요도')
        plt.title('모든 특성의 중요도')
        
        plt.tight_layout()
        plt.show()

class SHAPExplainer:
    """SHAP (SHapley Additive exPlanations) 간단한 구현"""
    
    def __init__(self, model, background_data):
        self.model = model
        self.background_data = background_data
        self.baseline = np.mean(model(background_data))
    
    def explain_instance(self, instance):
        """SHAP 값 계산 (샘플링 기반 근사)"""
        n_features = len(instance)
        shap_values = np.zeros(n_features)
        
        # 각 특성에 대해 SHAP 값 계산
        for i in range(n_features):
            shap_values[i] = self._calculate_shapley_value(instance, i)
        
        return {
            'shap_values': shap_values,
            'instance': instance,
            'baseline': self.baseline,
            'prediction': self.model(instance.reshape(1, -1))[0]
        }
    
    def _calculate_shapley_value(self, instance, feature_idx, num_samples=100):
        """특정 특성에 대한 Shapley 값 계산"""
        marginal_contributions = []
        
        for _ in range(num_samples):
            # 랜덤하게 다른 특성들의 부분집합 선택
            n_features = len(instance)
            subset = np.random.choice(n_features - 1, 
                                    size=np.random.randint(0, n_features - 1), 
                                    replace=False)
            
            # feature_idx를 포함한 집합과 포함하지 않은 집합 생성
            subset_with = np.sort(np.append(subset, feature_idx))
            subset_without = subset
            
            # 각 집합에 대해 예측값 계산
            pred_with = self._predict_with_subset(instance, subset_with)
            pred_without = self._predict_with_subset(instance, subset_without)
            
            # 한계 기여도 계산
            marginal_contributions.append(pred_with - pred_without)
        
        return np.mean(marginal_contributions)
    
    def _predict_with_subset(self, instance, feature_subset):
        """특성 부분집합을 사용한 예측"""
        # 배경 데이터의 평균을 사용해 나머지 특성 채우기
        masked_instance = np.mean(self.background_data, axis=0).copy()
        masked_instance[feature_subset] = instance[feature_subset]
        
        return self.model(masked_instance.reshape(1, -1))[0]
    
    def visualize_shap(self, explanation):
        """SHAP 설명 시각화"""
        shap_values = explanation['shap_values']
        instance = explanation['instance']
        baseline = explanation['baseline']
        prediction = explanation['prediction']
        
        plt.figure(figsize=(15, 8))
        
        # SHAP 값
        plt.subplot(2, 3, 1)
        colors = ['red' if s < 0 else 'blue' for s in shap_values]
        plt.barh(range(len(shap_values)), shap_values, color=colors, alpha=0.7)
        plt.xlabel('SHAP 값')
        plt.ylabel('특성 인덱스')
        plt.title('SHAP 특성 중요도')
        plt.axvline(x=0, color='black', linestyle='-', alpha=0.3)
        
        # 기여도 분해
        plt.subplot(2, 3, 2)
        cumulative = np.cumsum(np.concatenate([[baseline], shap_values]))
        plt.plot(range(len(cumulative)), cumulative, 'o-', linewidth=2)
        plt.axhline(y=baseline, color='red', linestyle='--', label=f'기준값: {baseline:.3f}')
        plt.axhline(y=prediction, color='green', linestyle='--', label=f'예측값: {prediction:.3f}')
        plt.xlabel('특성 추가 순서')
        plt.ylabel('예측값')
        plt.title('SHAP 기여도 분해')
        plt.legend()
        
        # 특성값 vs SHAP값
        plt.subplot(2, 3, 3)
        plt.scatter(instance, shap_values, alpha=0.7, s=50)
        plt.xlabel('특성 값')
        plt.ylabel('SHAP 값')
        plt.title('특성값 vs SHAP값')
        plt.axhline(y=0, color='black', linestyle='-', alpha=0.3)
        
        plt.tight_layout()
        plt.show()
```

```python
class SimpleCLIP:
    """CLIP 스타일 멀티모달 모델의 기본 개념 구현"""
    
    def __init__(self, image_dim=512, text_dim=512, embed_dim=256):
        self.image_dim = image_dim
        self.text_dim = text_dim
        self.embed_dim = embed_dim
        
        # 이미지 인코더 (간단한 MLP)
        self.image_encoder = {
            'W1': np.random.randn(image_dim, embed_dim) * 0.1,
            'b1': np.zeros((1, embed_dim)),
        }
        
        # 텍스트 인코더 (간단한 MLP)
        self.text_encoder = {
            'W1': np.random.randn(text_dim, embed_dim) * 0.1,
            'b1': np.zeros((1, embed_dim)),
        }
        
        # 온도 매개변수
        self.temperature = 0.07
    
    def encode_image(self, image_features):
        """이미지를 공통 임베딩 공간으로 인코딩"""
        # 간단한 선형 변환 + 정규화
        embeddings = np.dot(image_features, self.image_encoder['W1']) + self.image_encoder['b1']
        # L2 정규화
        embeddings = embeddings / (np.linalg.norm(embeddings, axis=1, keepdims=True) + 1e-8)
        return embeddings
    
    def encode_text(self, text_features):
        """텍스트를 공통 임베딩 공간으로 인코딩"""
        embeddings = np.dot(text_features, self.text_encoder['W1']) + self.text_encoder['b1']
        # L2 정규화
        embeddings = embeddings / (np.linalg.norm(embeddings, axis=1, keepdims=True) + 1e-8)
        return embeddings
    
    def compute_similarity(self, image_embeddings, text_embeddings):
        """이미지와 텍스트 임베딩 간 코사인 유사도 계산"""
        # 코사인 유사도 = 정규화된 벡터들의 내적
        similarity = np.dot(image_embeddings, text_embeddings.T)
        return similarity / self.temperature
    
    def contrastive_loss(self, image_features, text_features):
        """대조 학습 손실 계산"""
        batch_size = image_features.shape[0]
        
        # 임베딩 계산
        image_embeddings = self.encode_image(image_features)
        text_embeddings = self.encode_text(text_features)
        
        # 유사도 행렬 계산
        logits = self.compute_similarity(image_embeddings, text_embeddings)
        
        # 대각선이 정답 (i번째 이미지와 i번째 텍스트가 매칭)
        labels = np.arange(batch_size)
        
        # 이미지→텍스트 방향 손실
        image_to_text_loss = -np.mean([
            logits[i, labels[i]] - np.log(np.sum(np.exp(logits[i, :])))
            for i in range(batch_size)
        ])
        
        # 텍스트→이미지 방향 손실
        text_to_image_loss = -np.mean([
            logits[labels[j], j] - np.log(np.sum(np.exp(logits[:, j])))
            for j in range(batch_size)
        ])
        
        # 총 손실
        total_loss = (image_to_text_loss + text_to_image_loss) / 2
        
        return total_loss, logits
    
    def zero_shot_classification(self, image_features, text_prompts):
        """제로샷 분류"""
        # 이미지 임베딩
        image_embeddings = self.encode_image(image_features)
        
        # 텍스트 프롬프트들 임베딩
        text_embeddings = self.encode_text(text_prompts)
        
        # 유사도 계산
        similarities = self.compute_similarity(image_embeddings, text_embeddings)
        
        # 소프트맥스로 확률 변환
        probabilities = np.exp(similarities) / np.sum(np.exp(similarities), axis=1, keepdims=True)
        
        return probabilities
    
    def visualize_similarity_matrix(self, image_features, text_features, 
                                   image_labels=None, text_labels=None):
        """유사도 행렬 시각화"""
        image_embeddings = self.encode_image(image_features)
        text_embeddings = self.encode_text(text_features)
        
        similarity_matrix = np.dot(image_embeddings, text_embeddings.T)
        
        plt.figure(figsize=(10, 8))
        plt.imshow(similarity_matrix, cmap='RdYlBu', aspect='auto')
        plt.colorbar(label='코사인 유사도')
        
        if image_labels:
            plt.yticks(range(len(image_labels)), image_labels)
            plt.ylabel('이미지')
        
        if text_labels:
            plt.xticks(range(len(text_labels)), text_labels, rotation=45)
            plt.xlabel('텍스트')
        
        plt.title('CLIP 스타일 유사도 행렬')
        plt.tight_layout()
        plt.show()
        
        return similarity_matrix

    # 텍스트 특성 (가상의 텍스트 인코딩)
    text_features = np.random.randn(8, 512)   # 8개 텍스트
    text_labels = ['a cat', 'a dog', 'a car', 'an airplane', 'a bird', 'a horse', 'a ship', 'a truck']
    
    # 의도적으로 매칭되도록 특성 조정
    for i in range(len(image_features)):
        # 같은 인덱스끼리는 유사하게 만들기
        shared_component = np.random.randn(512) * 0.5
        image_features[i] += shared_component
        text_features[i] += shared_component
    
    # CLIP 모델 생성
    clip_model = SimpleCLIP(image_dim=512, text_dim=512, embed_dim=256)
    
    # 유사도 행렬 시각화
    print("\n📊 이미지-텍스트 유사도 행렬:")
    similarity_matrix = clip_model.visualize_similarity_matrix(
        image_features, text_features, image_labels, text_labels
    )
    
    # 제로샷 분류 데모
    print("\n🎯 제로샷 분류 데모:")
    test_image = image_features[0:1]  # 첫 번째 이미지 (고양이)
    
    # 다양한 클래스 프롬프트
    class_prompts = np.array([
        text_features[0],  # a cat
        text_features[1],  # a dog  
        text_features[2],  # a car
        text_features[4],  # a bird
    ])
    
    probabilities = clip_model.zero_shot_classification(test_image, class_prompts)
    
    class_names = ['cat', 'dog', 'car', 'bird']
    print(f"테스트 이미지: {image_labels[0]}")
    print("예측 확률:")
    for i, (class_name, prob) in enumerate(zip(class_names, probabilities[0])):
        print(f"  {class_name}: {prob:.3f}")
    
    # 대조 학습 손실 계산
    loss, logits = clip_model.contrastive_loss(image_features, text_features)
    print(f"\n📉 대조 학습 손실: {loss:.4f}")
    
    # 임베딩 공간 시각화 (2D 투영)
    print("\n🗺️ 임베딩 공간 시각화:")
    
    image_embeddings = clip_model.encode_image(image_features)
    text_embeddings = clip_model.encode_text(text_features)
    
    # PCA로 2D 투영 (간단한 구현)
    all_embeddings = np.vstack([image_embeddings, text_embeddings])
    
    # 공분산 행렬
    centered = all_embeddings - np.mean(all_embeddings, axis=0)
    cov_matrix = np.dot(centered.T, centered) / (len(all_embeddings) - 1)
    
    # 고유값 분해
    eigenvalues, eigenvectors = np.linalg.eigh(cov_matrix)
    
    # 상위 2개 주성분
    top_components = eigenvectors[:, -2:]
    projected = np.dot(all_embeddings, top_components)
    
    # 시각화
    plt.figure(figsize=(12, 8))
    
    # 이미지 임베딩
    img_projected = projected[:len(image_embeddings)]
    plt.scatter(img_projected[:, 0], img_projected[:, 1], 
               c='blue', marker='o', s=100, alpha=0.7, label='Images')
    
    # 텍스트 임베딩
    text_projected = projected[len(image_embeddings):]
    plt.scatter(text_projected[:, 0], text_projected[:, 1], 
               c='red', marker='s', s=100, alpha=0.7, label='Texts')
    
    # 매칭 라인
    for i in range(len(image_labels)):
        plt.plot([img_projected[i, 0], text_projected[i, 0]], 
                [img_projected[i, 1], text_projected[i, 1]], 
                'gray', alpha=0.5, linewidth=1)
        
        # 라벨
        plt.annotate(image_labels[i], img_projected[i], xytext=(5, 5), 
                    textcoords='offset points', fontsize=8, color='blue')
        plt.annotate(text_labels[i], text_projected[i], xytext=(5, -15), 
                    textcoords='offset points', fontsize=8, color='red')
    
    plt.xlabel('주성분 1')
    plt.ylabel('주성분 2') 
    plt.title('CLIP 임베딩 공간 (2D 투영)')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.show()
    
    print("\n✨ CLIP의 핵심 아이디어:")
    print("1. 이미지와 텍스트를 같은 임베딩 공간으로 매핑")
    print("2. 대조 학습으로 매칭되는 쌍은 가깝게, 매칭되지 않는 쌍은 멀게")
    print("3. 제로샷 분류: 새로운 클래스도 텍스트 설명만으로 분류 가능")
    print("4. 멀티모달 이해: 이미지와 텍스트의 의미적 연결 학습")
    
    return clip_model
```
