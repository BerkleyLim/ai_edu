### 1.1 벡터/행렬 연산 from scratch

```python
import math
import random

class Matrix:
    """순수 Python 행렬 클래스"""
    
    def __init__(self, data):
        if isinstance(data, list) and isinstance(data[0], list):
            self.data = data
            self.rows = len(data)
            self.cols = len(data[0])
        else:
            raise ValueError("2D 리스트가 필요합니다")
    
    def __getitem__(self, key):
        return self.data[key]
    
    def __setitem__(self, key, value):
        self.data[key] = value
    
    def dot(self, other):
        """행렬 곱셈: A × B"""
        if self.cols != other.rows:
            raise ValueError(f"행렬 차원 불일치: ({self.rows}×{self.cols}) × ({other.rows}×{other.cols})")
        
        # 결과 행렬 초기화
        result = [[0 for _ in range(other.cols)] for _ in range(self.rows)]
        
        # 삼중 반복문으로 행렬 곱셈
        for i in range(self.rows):
            for j in range(other.cols):
                for k in range(self.cols):
                    result[i][j] += self.data[i][k] * other.data[k][j]
        
        return Matrix(result)
    
    def transpose(self):
        """전치 행렬"""
        result = [[self.data[j][i] for j in range(self.rows)] for i in range(self.cols)]
        return Matrix(result)
    
    def add(self, other):
        """행렬 덧셈"""
        if self.rows != other.rows or self.cols != other.cols:
            raise ValueError("행렬 크기가 다릅니다")
        
        result = [[self.data[i][j] + other.data[i][j] 
                  for j in range(self.cols)] 
                 for i in range(self.rows)]
        return Matrix(result)
    
    def scalar_multiply(self, scalar):
        """스칼라 곱"""
        result = [[self.data[i][j] * scalar 
                  for j in range(self.cols)] 
                 for i in range(self.rows)]
        return Matrix(result)
    
    def to_vector(self):
        """1차원 벡터로 변환 (행렬이 1xN 또는 Nx1인 경우)"""
        if self.rows == 1:
            return self.data[0]
        elif self.cols == 1:
            return [row[0] for row in self.data]
        else:
            raise ValueError("벡터로 변환할 수 없습니다")

def vector_dot(a, b):
    """벡터 내적"""
    if len(a) != len(b):
        raise ValueError("벡터 길이가 다릅니다")
    return sum(a[i] * b[i] for i in range(len(a)))

def vector_norm(v):
    """벡터 크기 (L2 norm)"""
    return math.sqrt(sum(x*x for x in v))

def vector_normalize(v):
    """벡터 정규화"""
    norm = vector_norm(v)
    return [x/norm for x in v] if norm != 0 else v

# 테스트
print("=== 기본 행렬 연산 테스트 ===")
A = Matrix([[1, 2], [3, 4]])
B = Matrix([[5, 6], [7, 8]])

print("A × B:")
C = A.dot(B)
for row in C.data:
    print(row)

print(f"벡터 내적 [1,2,3] · [4,5,6] = {vector_dot([1,2,3], [4,5,6])}")
```

### 1.2 활성화 함수 수식과 구현

```python
def sigmoid(x):
    """
    시그모이드 함수
    σ(x) = 1 / (1 + e^(-x))
    
    도함수: σ'(x) = σ(x) × (1 - σ(x))
    """
    try:
        return 1 / (1 + math.exp(-x))
    except OverflowError:
        return 0 if x < 0 else 1

def sigmoid_derivative(x):
    """시그모이드 도함수"""
    s = sigmoid(x)
    return s * (1 - s)

def tanh(x):
    """
    하이퍼볼릭 탄젠트
    tanh(x) = (e^x - e^(-x)) / (e^x + e^(-x))
    
    도함수: tanh'(x) = 1 - tanh²(x)
    """
    try:
        exp_x = math.exp(x)
        exp_neg_x = math.exp(-x)
        return (exp_x - exp_neg_x) / (exp_x + exp_neg_x)
    except OverflowError:
        return 1 if x > 0 else -1

def tanh_derivative(x):
    """tanh 도함수"""
    t = tanh(x)
    return 1 - t*t

def relu(x):
    """
    ReLU 함수
    ReLU(x) = max(0, x)
    
    도함수: ReLU'(x) = 1 if x > 0 else 0
    """
    return max(0, x)

def relu_derivative(x):
    """ReLU 도함수"""
    return 1 if x > 0 else 0

def softmax(x_list):
    """
    소프트맥스 함수 (벡터용)
    softmax(x_i) = e^(x_i) / Σ(e^(x_j))
    """
    # 수치 안정성을 위해 최대값 빼기
    max_x = max(x_list)
    exp_x = [math.exp(x - max_x) for x in x_list]
    sum_exp = sum(exp_x)
    return [exp_val / sum_exp for exp_val in exp_x]

# 활성화 함수 시각화
```

```python
class Perceptron:
    """단일 퍼셉트론 - 선형 분류기"""
    
    def __init__(self, input_size, learning_rate=0.01):
        # 가중치 랜덤 초기화: w ~ N(0, 0.01)
        self.weights = [random.gauss(0, 0.01) for _ in range(input_size)]
        self.bias = random.gauss(0, 0.01)
        self.learning_rate = learning_rate
    
    def predict(self, x):
        """
        예측: ŷ = sign(w·x + b)
        
        where: w·x = Σ(w_i × x_i)
        """
        # 가중합 계산
        weighted_sum = vector_dot(self.weights, x) + self.bias
        
        # 활성화 (step function)
        return 1 if weighted_sum > 0 else 0
    
    def train_step(self, x, y):
        """
        퍼셉트론 학습 규칙:
        w_new = w_old + η × (y - ŷ) × x
        b_new = b_old + η × (y - ŷ)
        
        where:
        - η: 학습률
        - y: 실제 값
        - ŷ: 예측 값
        """
        prediction = self.predict(x)
        error = y - prediction
        
        # 가중치 업데이트
        for i in range(len(self.weights)):
            self.weights[i] += self.learning_rate * error * x[i]
        
        self.bias += self.learning_rate * error
        
        return abs(error)  # 오차 반환
    
    def train(self, X, y, epochs=100):
        """전체 데이터셋으로 학습"""
        for epoch in range(epochs):
            total_error = 0
            for i in range(len(X)):
                total_error += self.train_step(X[i], y[i])
            
            if total_error == 0:  # 완전 분류
                print(f"수렴 완료: {epoch+1} 에포크")
                break
            
            if (epoch + 1) % 10 == 0:
                print(f"에포크 {epoch+1}: 총 오차 = {total_error}")

# AND 게이트 학습
print("\n=== 퍼셉트론으로 AND 게이트 학습 ===")
X_and = [[0,0], [0,1], [1,0], [1,1]]

perceptron.train(X_and, y_and, epochs=20)
```

### 2.2 다층 퍼셉트론 (MLP) 완전 구현

```python
class MLP:
    """다층 퍼셉트론 - 역전파 알고리즘 포함"""
    
    def __init__(self, layer_sizes):
        """
        layer_sizes: [입력층, 은닉층1, 은닉층2, ..., 출력층]
        예: [2, 4, 3, 1] = 입력2개, 은닉층4개, 은닉층3개, 출력1개
        """
        self.layer_sizes = layer_sizes
        self.num_layers = len(layer_sizes)
        
        # 가중치 행렬 초기화 (Xavier 초기화)
        self.weights = []
        self.biases = []
        
        for i in range(self.num_layers - 1):
            # Xavier/Glorot 초기화: w ~ N(0, sqrt(2/(fan_in + fan_out)))
            fan_in = layer_sizes[i]
            fan_out = layer_sizes[i + 1]
            limit = math.sqrt(6 / (fan_in + fan_out))
            
            # 가중치 행렬: (이전층 × 다음층)
            weight_matrix = []
            for _ in range(fan_out):
                row = [random.uniform(-limit, limit) for _ in range(fan_in)]
                weight_matrix.append(row)
            
            self.weights.append(Matrix(weight_matrix))
            
            # 편향: 0으로 초기화
            bias_vector = [[0] for _ in range(fan_out)]
            self.biases.append(Matrix(bias_vector))
    
    def forward_pass(self, x):
        """
        순전파 (Forward Propagation)
        
        각 층에서: z^(l) = W^(l) × a^(l-1) + b^(l)
                  a^(l) = σ(z^(l))
        
        where:
        - z^(l): l번째 층의 가중합
        - a^(l): l번째 층의 활성화 값
        - W^(l): l번째 층의 가중치 행렬
        - b^(l): l번째 층의 편향
        - σ: 활성화 함수
        """
        # 입력을 행렬로 변환
        activation = Matrix([[val] for val in x])
        activations = [activation]  # 각 층의 활성화 저장
        zs = []  # 각 층의 가중합 저장
        
        for i in range(self.num_layers - 1):
            # z = W × a + b
            z = self.weights[i].dot(activation).add(self.biases[i])
            zs.append(z)
            
            # a = σ(z)
            activation_values = []
            for j in range(z.rows):
                if i == self.num_layers - 2:  # 출력층
                    activation_values.append([sigmoid(z[j][0])])
                else:  # 은닉층
                    activation_values.append([sigmoid(z[j][0])])
            
            activation = Matrix(activation_values)
            activations.append(activation)
        
        return activations, zs
    
    def backward_pass(self, x, y):
        """
        역전파 (Backpropagation)
        
        오차 계산: δ^(L) = (a^(L) - y) ⊙ σ'(z^(L))
        
        역전파: δ^(l) = ((W^(l+1))^T × δ^(l+1)) ⊙ σ'(z^(l))
        
        기울기: ∂C/∂W^(l) = δ^(l) × (a^(l-1))^T
               ∂C/∂b^(l) = δ^(l)
        """
        activations, zs = self.forward_pass(x)
        
        # 기울기 저장용
        nabla_w = [Matrix([[0 for _ in range(w.cols)] for _ in range(w.rows)]) 
                  for w in self.weights]
        nabla_b = [Matrix([[0] for _ in range(b.rows)]) 
                  for b in self.biases]
        
        # 출력층 오차 계산
        # δ^(L) = (a^(L) - y) ⊙ σ'(z^(L))
        last_activation = activations[-1]
        last_z = zs[-1]
        
        delta_values = []
        for i in range(last_activation.rows):
            output_error = last_activation[i][0] - y[i] if isinstance(y, list) else last_activation[i][0] - y
            derivative = sigmoid_derivative(last_z[i][0])
            delta_values.append([output_error * derivative])
        
        delta = Matrix(delta_values)
        
        # 출력층 기울기
        # ∂C/∂b = δ
        nabla_b[-1] = delta
        
        # ∂C/∂W = δ × a^(L-1)^T
        prev_activation_T = activations[-2].transpose()
        nabla_w[-1] = delta.dot(prev_activation_T)
        
        # 역전파로 이전 층들 계산
        for l in range(2, self.num_layers):
            z = zs[-l]
            
            # δ^(l) = ((W^(l+1))^T × δ^(l+1)) ⊙ σ'(z^(l))
            weight_T = self.weights[-l+1].transpose()
            delta_new_values = []
            
            for i in range(z.rows):
                # 가중치 전치와 델타의 곱
                weighted_error = 0
                for j in range(delta.rows):
                    weighted_error += weight_T[i][j] * delta[j][0]
                
                # 시그모이드 도함수와 곱
                sp = sigmoid_derivative(z[i][0])
                delta_new_values.append([weighted_error * sp])
            
            delta = Matrix(delta_new_values)
            
            nabla_b[-l] = delta
            
            if l < self.num_layers:
                prev_activation_T = activations[-l-1].transpose()
                nabla_w[-l] = delta.dot(prev_activation_T)
        
        return nabla_w, nabla_b
    
    def update_weights(self, nabla_w, nabla_b, learning_rate):
        """
        가중치 업데이트
        W_new = W_old - η × ∂C/∂W
        b_new = b_old - η × ∂C/∂b
        """
        for i in range(len(self.weights)):
            # W = W - η × ∇W
            for j in range(self.weights[i].rows):
                for k in range(self.weights[i].cols):
                    self.weights[i][j][k] -= learning_rate * nabla_w[i][j][k]
            
            # b = b - η × ∇b
            for j in range(self.biases[i].rows):
                self.biases[i][j][0] -= learning_rate * nabla_b[i][j][0]
    
    def train_step(self, x, y, learning_rate):
        """단일 훈련 스텝"""
        nabla_w, nabla_b = self.backward_pass(x, y)
        self.update_weights(nabla_w, nabla_b, learning_rate)
    
    def predict(self, x):
        """예측"""
        activations, _ = self.forward_pass(x)
        output = activations[-1]
        return [output[i][0] for i in range(output.rows)]
    
    def cost(self, X, y):
        """
        평균 제곱 오차 (MSE)
        C = (1/2n) × Σ||y - a^(L)||²
        """
        total_cost = 0
        for i in range(len(X)):
            prediction = self.predict(X[i])
            if isinstance(y[i], list):
                cost = sum((y[i][j] - prediction[j])**2 for j in range(len(y[i])))
            else:
                cost = (y[i] - prediction[0])**2
            total_cost += cost
        
        return total_cost / (2 * len(X))

# XOR 문제 해결
print("\n=== MLP로 XOR 문제 해결 ===")

y_xor = [0, 1, 1, 0]

# 2-4-1 네트워크 (입력2개, 은닉4개, 출력1개)

print("훈련 전 예측:")

    pred = mlp.predict(x)[0]
    print(f"{x[0]} XOR {x[1]} = {pred:.3f} (정답: {y_xor[i]})")

for epoch in range(epochs):
    for i in range(len(X_xor)):
        mlp.train_step(X_xor[i], y_xor[i], learning_rate)
    
    if (epoch + 1) % 200 == 0:
        cost = mlp.cost(X_xor, y_xor)
        print(f"에포크 {epoch+1}: 비용 = {cost:.4f}")

print("\n훈련 후 예측:")

    pred = mlp.predict(x)[0]
    print(f"{x[0]} XOR {x[1]} = {pred:.3f} (정답: {y_xor[i]})")

print(f"최종 비용: {mlp.cost(X_xor, y_xor):.4f}")
```

## 🎲 3. 확률론적 알고리즘 구현

### 3.1 나이브 베이즈 분류기

```python
class NaiveBayes:
    """
    나이브 베이즈 분류기
    
    베이즈 정리:
    P(C|X) = P(X|C) × P(C) / P(X)
    
    나이브 가정: P(X|C) = ∏ P(x_i|C)
    """
    
    def __init__(self):
        self.classes = []
        self.class_probabilities = {}  # P(C)
        self.feature_probabilities = {}  # P(x_i|C)
        self.feature_stats = {}  # 가우시안 분포용 통계
    
    def fit(self, X, y):
        """학습: 각 클래스와 특성의 확률 계산"""
        n_samples = len(X)
        n_features = len(X[0])
        
        # 클래스별 개수 계산
        class_counts = {}
        for label in y:
            if label not in class_counts:
                class_counts[label] = 0
            class_counts[label] += 1
        
        self.classes = list(class_counts.keys())
        
        # 사전 확률 P(C) 계산
        for cls in self.classes:
            self.class_probabilities[cls] = class_counts[cls] / n_samples
        
        # 각 클래스별로 특성 통계 계산
        for cls in self.classes:
            self.feature_stats[cls] = []
            
            # 해당 클래스의 샘플들만 선택
            class_samples = []
            for i in range(n_samples):
                if y[i] == cls:
                    class_samples.append(X[i])
            
            # 각 특성별로 평균과 분산 계산
            for feature_idx in range(n_features):
                feature_values = [sample[feature_idx] for sample in class_samples]
                
                # 평균: μ = (1/n) × Σx_i
                mean = sum(feature_values) / len(feature_values)
                
                # 분산: σ² = (1/n) × Σ(x_i - μ)²
                variance = sum((x - mean)**2 for x in feature_values) / len(feature_values)
                
                # 수치 안정성을 위해 최소 분산 보장
                variance = max(variance, 1e-6)
                
                self.feature_stats[cls].append({
                    'mean': mean,
                    'variance': variance,
                    'std': math.sqrt(variance)
                })
    
    def gaussian_probability(self, x, mean, variance):
        """
        가우시안 확률 밀도 함수
        P(x) = (1/√(2πσ²)) × exp(-((x-μ)²)/(2σ²))
        """
        coefficient = 1 / math.sqrt(2 * math.pi * variance)
        exponent = math.exp(-((x - mean) ** 2) / (2 * variance))
        return coefficient * exponent
    
    def predict_proba(self, x):
        """각 클래스에 대한 확률 계산"""
        probabilities = {}
        
        for cls in self.classes:
            # 사전 확률 P(C)
            prob = self.class_probabilities[cls]
            
            # 우도 P(X|C) = ∏ P(x_i|C) 계산
            for i in range(len(x)):
                feature_stats = self.feature_stats[cls][i]
                feature_prob = self.gaussian_probability(
                    x[i], 
                    feature_stats['mean'], 
                    feature_stats['variance']
                )
                prob *= feature_prob
            
            probabilities[cls] = prob
        
        # 정규화 (사후 확률로 변환)
        total_prob = sum(probabilities.values())
        if total_prob > 0:
            for cls in probabilities:
                probabilities[cls] /= total_prob
        
        return probabilities
    
    def predict(self, x):
        """가장 확률이 높은 클래스 반환"""
        probabilities = self.predict_proba(x)
        return max(probabilities.keys(), key=lambda k: probabilities[k])

# 간단한 2차원 데이터 생성
X_iris = [
```

```python
class KMeans:
    """
    k-평균 클러스터링
    
    목적 함수 (최소화):
    J = Σ Σ ||x_i - μ_k||²
    
    where:
    - x_i: i번째 데이터 포인트
    - μ_k: k번째 클러스터 중심
    """
    
    def __init__(self, k, max_iters=100, tol=1e-4):
        self.k = k
        self.max_iters = max_iters
        self.tol = tol
        self.centroids = []
        self.labels = []
    
    def euclidean_distance(self, point1, point2):
        """유클리드 거리: d = √(Σ(x_i - y_i)²)"""
        return math.sqrt(sum((a - b)**2 for a, b in zip(point1, point2)))
    
    def fit(self, X):
        """k-means 클러스터링 실행"""
        n_samples = len(X)
        n_features = len(X[0])
        
        # 1. 중심점 랜덤 초기화
        self.centroids = []
        for _ in range(self.k):
            # 데이터 범위 내에서 랜덤하게 초기화
            centroid = []
            for j in range(n_features):
                feature_values = [X[i][j] for i in range(n_samples)]
                min_val = min(feature_values)
                max_val = max(feature_values)
                centroid.append(random.uniform(min_val, max_val))
            self.centroids.append(centroid)
        
        print(f"초기 중심점: {[[f'{c:.2f}' for c in centroid] for centroid in self.centroids]}")
        
        for iteration in range(self.max_iters):
            # 2. 클러스터 할당 (E-step)
            new_labels = []
            for i in range(n_samples):
                distances = []
                for k in range(self.k):
                    dist = self.euclidean_distance(X[i], self.centroids[k])
                    distances.append(dist)
                
                # 가장 가까운 중심점의 클러스터에 할당
                closest_cluster = distances.index(min(distances))
                new_labels.append(closest_cluster)
            
            # 3. 중심점 업데이트 (M-step)
            new_centroids = []
            for k in range(self.k):
                # k번째 클러스터에 속한 점들 찾기
                cluster_points = []
                for i in range(n_samples):
                    if new_labels[i] == k:
                        cluster_points.append(X[i])
                
                if len(cluster_points) > 0:
                    # 중심점 = 클러스터 내 점들의 평균
                    # μ_k = (1/|C_k|) × Σ x_i  (x_i ∈ C_k)
                    new_centroid = []
                    for j in range(n_features):
                        feature_sum = sum(point[j] for point in cluster_points)
                        mean_value = feature_sum / len(cluster_points)
                        new_centroid.append(mean_value)
                    new_centroids.append(new_centroid)
                else:
                    # 빈 클러스터인 경우 기존 중심점 유지
                    new_centroids.append(self.centroids[k][:])
            
            # 4. 수렴 확인
            max_centroid_shift = 0
            for k in range(self.k):
                shift = self.euclidean_distance(self.centroids[k], new_centroids[k])
                max_centroid_shift = max(max_centroid_shift, shift)
            
            self.centroids = new_centroids
            self.labels = new_labels
            
            print(f"반복 {iteration+1}: 중심점 이동 = {max_centroid_shift:.4f}")
            
            if max_centroid_shift < self.tol:
                print(f"수렴 완료: {iteration+1} 반복")
                break
        
        return self
    
    def predict(self, X):
        """새로운 데이터 포인트의 클러스터 예측"""
        predictions = []
        for point in X:
            distances = []
            for k in range(self.k):
                dist = self.euclidean_distance(point, self.centroids[k])
                distances.append(dist)
            
            closest_cluster = distances.index(min(distances))
            predictions.append(closest_cluster)
        
        return predictions
    
    def inertia(self, X):
        """
        관성 (Inertia) 계산 - 클러스터 내 제곱합
        WCSS = Σ Σ ||x_i - μ_k||²
        """
        total_inertia = 0
        for i, point in enumerate(X):
            cluster = self.labels[i]
            centroid = self.centroids[cluster]
            distance_squared = sum((a - b)**2 for a, b in zip(point, centroid))
            total_inertia += distance_squared
        
        return total_inertia

# 클러스터링 테스트
print("\n=== k-평균 클러스터링 ===")
```

```python
class GradientDescent:
    """다양한 경사하강법 구현"""
    
    @staticmethod
    def vanilla_gd(f, grad_f, x0, learning_rate=0.01, max_iters=1000, tol=1e-6):
        """
        바닐라 경사하강법
        
        x_{t+1} = x_t - η × ∇f(x_t)
        
        where:
        - η: 학습률
        - ∇f(x_t): x_t에서의 기울기
        """
        x = x0[:]  # 복사
        history = [x[:]]
        
        for i in range(max_iters):
            # 기울기 계산
            grad = grad_f(x)
            
            # 파라미터 업데이트
            x_new = []
            for j in range(len(x)):
                x_new.append(x[j] - learning_rate * grad[j])
            
            history.append(x_new[:])
            
            # 수렴 확인
            if vector_norm([x_new[j] - x[j] for j in range(len(x))]) < tol:
                print(f"수렴: {i+1} 반복")
                break
            
            x = x_new
        
        return x, history
    
    @staticmethod
    def sgd_with_momentum(f, grad_f, x0, learning_rate=0.01, momentum=0.9, 
                         max_iters=1000, tol=1e-6):
        """
        모멘텀 SGD
        
        v_{t+1} = γ × v_t + η × ∇f(x_t)
        x_{t+1} = x_t - v_{t+1}
        
        where:
        - γ: 모멘텀 계수 (보통 0.9)
        - v_t: 속도 벡터
        """
        x = x0[:]
        v = [0] * len(x0)  # 속도 벡터 초기화
        history = [x[:]]
        
        for i in range(max_iters):
            grad = grad_f(x)
            
            # 속도 업데이트: v = γv + η∇f
            for j in range(len(x)):
                v[j] = momentum * v[j] + learning_rate * grad[j]
            
            # 파라미터 업데이트: x = x - v
            x_new = []
            for j in range(len(x)):
                x_new.append(x[j] - v[j])
            
            history.append(x_new[:])
            
            # 수렴 확인
            if vector_norm([x_new[j] - x[j] for j in range(len(x))]) < tol:
                print(f"모멘텀 SGD 수렴: {i+1} 반복")
                break
            
            x = x_new
        
        return x, history
    
    @staticmethod
    def adam(f, grad_f, x0, learning_rate=0.001, beta1=0.9, beta2=0.999, 
             epsilon=1e-8, max_iters=1000, tol=1e-6):
        """
        Adam 최적화
        
        m_{t+1} = β₁ × m_t + (1-β₁) × ∇f(x_t)     # 1차 모멘트
        v_{t+1} = β₂ × v_t + (1-β₂) × (∇f(x_t))²  # 2차 모멘트
        
        m̂_{t+1} = m_{t+1} / (1 - β₁^{t+1})        # 편향 보정
        v̂_{t+1} = v_{t+1} / (1 - β₂^{t+1})        # 편향 보정
        
        x_{t+1} = x_t - η × m̂_{t+1} / (√v̂_{t+1} + ε)
        """
        x = x0[:]
        m = [0] * len(x0)  # 1차 모멘트
        v = [0] * len(x0)  # 2차 모멘트
        history = [x[:]]
        
        for t in range(1, max_iters + 1):
            grad = grad_f(x)
            
            # 모멘트 업데이트
            for j in range(len(x)):
                # 1차 모멘트: m = β₁m + (1-β₁)g
                m[j] = beta1 * m[j] + (1 - beta1) * grad[j]
                
                # 2차 모멘트: v = β₂v + (1-β₂)g²
                v[j] = beta2 * v[j] + (1 - beta2) * (grad[j] ** 2)
            
            # 편향 보정
            m_corrected = [m[j] / (1 - beta1**t) for j in range(len(x))]
            v_corrected = [v[j] / (1 - beta2**t) for j in range(len(x))]
            
            # 파라미터 업데이트
            x_new = []
            for j in range(len(x)):
                update = learning_rate * m_corrected[j] / (math.sqrt(v_corrected[j]) + epsilon)
                x_new.append(x[j] - update)
            
            history.append(x_new[:])
            
            # 수렴 확인
            if vector_norm([x_new[j] - x[j] for j in range(len(x))]) < tol:
                print(f"Adam 수렴: {t} 반복")
                break
            
            x = x_new
        
        return x, history

# 테스트: Rosenbrock 함수 최적화
def rosenbrock(x):
    """
    로젠브록 함수: f(x,y) = (a-x)² + b(y-x²)²
    전역 최솟값: (a,a²) = (1,1)에서 f=0
    """
    a, b = 1, 100
    return (a - x[0])**2 + b * (x[1] - x[0]**2)**2

def rosenbrock_grad(x):
    """로젠브록 함수의 기울기"""
    a, b = 1, 100
    df_dx = -2*(a - x[0]) - 4*b*x[0]*(x[1] - x[0]**2)
    df_dy = 2*b*(x[1] - x[0]**2)
    return [df_dx, df_dy]
```

```python
class ConvolutionalLayer:
    """합성곱 층 완전 구현"""
    
    def __init__(self, input_shape, num_filters, filter_size, stride=1, padding=0):
        """
        input_shape: (height, width, channels)
        num_filters: 필터 개수
        filter_size: 필터 크기 (정사각형)
        """
        self.input_shape = input_shape
        self.num_filters = num_filters
        self.filter_size = filter_size
        self.stride = stride
        self.padding = padding
        
        # 출력 크기 계산
        self.output_height = (input_shape[0] + 2*padding - filter_size) // stride + 1
        self.output_width = (input_shape[1] + 2*padding - filter_size) // stride + 1
        
        # 필터 초기화 (Xavier 초기화)
        fan_in = filter_size * filter_size * input_shape[2]
        fan_out = filter_size * filter_size * num_filters
        limit = math.sqrt(6 / (fan_in + fan_out))
        
        self.filters = []
        for _ in range(num_filters):
            filter_weights = []
            for _ in range(filter_size):
                row = []
                for _ in range(filter_size):
                    channel = []
                    for _ in range(input_shape[2]):
                        weight = random.uniform(-limit, limit)
                        channel.append(weight)
                    row.append(channel)
                filter_weights.append(row)
            self.filters.append(filter_weights)
        
        # 편향
        self.biases = [random.uniform(-0.1, 0.1) for _ in range(num_filters)]
    
    def add_padding(self, input_data):
        """패딩 추가"""
        if self.padding == 0:
            return input_data
        
        height, width, channels = len(input_data), len(input_data[0]), len(input_data[0][0])
        padded_height = height + 2 * self.padding
        padded_width = width + 2 * self.padding
        
        # 0으로 패딩된 배열 생성
        padded = [[[0 for _ in range(channels)] 
                  for _ in range(padded_width)] 
                 for _ in range(padded_height)]
        
        # 원본 데이터 복사
        for h in range(height):
            for w in range(width):
                for c in range(channels):
                    padded[h + self.padding][w + self.padding][c] = input_data[h][w][c]
        
        return padded
    
    def convolution(self, input_data, filter_weights, bias):
        """
        단일 필터와의 합성곱 연산
        
        출력[i][j] = Σ Σ Σ 입력[i×s+di][j×s+dj][c] × 필터[di][dj][c] + 편향
        
        where:
        - s: 스트라이드
        - di, dj: 필터 내 위치
        - c: 채널 인덱스
        """
        padded_input = self.add_padding(input_data)
        
        output = [[0 for _ in range(self.output_width)] 
                 for _ in range(self.output_height)]
        
        for out_h in range(self.output_height):
            for out_w in range(self.output_width):
                # 현재 출력 위치에 해당하는 입력 영역
                start_h = out_h * self.stride
                start_w = out_w * self.stride
                
                conv_sum = 0
                
                # 필터와 입력의 원소별 곱셈 후 합산
                for fh in range(self.filter_size):
                    for fw in range(self.filter_size):
                        for c in range(self.input_shape[2]):
                            input_val = padded_input[start_h + fh][start_w + fw][c]
                            filter_val = filter_weights[fh][fw][c]
                            conv_sum += input_val * filter_val
                
                output[out_h][out_w] = conv_sum + bias
        
        return output
    
    def forward(self, input_data):
        """순전파: 모든 필터에 대해 합성곱 수행"""
        outputs = []
        
        for i in range(self.num_filters):
            filter_output = self.convolution(input_data, self.filters[i], self.biases[i])
            outputs.append(filter_output)
        
        return outputs

class MaxPoolingLayer:
    """최대 풀링 층"""
    
    def __init__(self, pool_size, stride=None):
        self.pool_size = pool_size
        self.stride = stride if stride else pool_size
    
    def forward(self, feature_maps):
        """
        최대 풀링
        출력[i][j] = max(입력[i×s:i×s+p][j×s:j×s+p])
        
        where:
        - s: 스트라이드
        - p: 풀링 크기
        """
        pooled_maps = []
        
        for feature_map in feature_maps:
            input_height = len(feature_map)
            input_width = len(feature_map[0])
            
            output_height = (input_height - self.pool_size) // self.stride + 1
            output_width = (input_width - self.pool_size) // self.stride + 1
            
            pooled = [[0 for _ in range(output_width)] 
                     for _ in range(output_height)]
            
            for out_h in range(output_height):
                for out_w in range(output_width):
                    start_h = out_h * self.stride
                    start_w = out_w * self.stride
                    
                    # 풀링 영역에서 최댓값 찾기
                    max_val = float('-inf')
                    for ph in range(self.pool_size):
                        for pw in range(self.pool_size):
                            val = feature_map[start_h + ph][start_w + pw]
                            max_val = max(max_val, val)
                    
                    pooled[out_h][out_w] = max_val
            
            pooled_maps.append(pooled)
        
        return pooled_maps
```

**Low Level 개발의 핵심은:**

1. **수식 완전 이해** - 모든 알고리즘을 수학 공식으로 이해
2. **순수 구현** - 라이브러리 없이 처음부터 끝까지 직접 구현  
3. **메모리 최적화** - 불필요한 메모리 사용 최소화
4. **수치 안정성** - 오버플로우/언더플로우 방지
5. **디버깅 능력** - 각 단계별 수치 확인 가능

이제 **라이브러리 없이도** AI 알고리즘의 내부 동작을 완벽히 이해하고 구현할 수 있습니다! 🚀
