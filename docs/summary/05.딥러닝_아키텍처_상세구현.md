# 딥러닝 아키텍처 상세 구현 및 수식

## 1. 다층 퍼셉트론 (Multi-Layer Perceptron)

### 1.1 수학적 기초

**순전파 (Forward Propagation)**
```
z^[l] = W^[l]a^[l-1] + b^[l]
a^[l] = σ(z^[l])
```

**역전파 (Backpropagation)**
```
dZ^[l] = dA^[l] * σ'(Z^[l])
dW^[l] = (1/m) * dZ^[l] * A^[l-1]ᵀ
db^[l] = (1/m) * Σ dZ^[l]
dA^[l-1] = W^[l]ᵀ * dZ^[l]
```

**활성화 함수들**
```
ReLU: f(x) = max(0, x)
Sigmoid: f(x) = 1 / (1 + e^(-x))
Tanh: f(x) = (e^x - e^(-x)) / (e^x + e^(-x))
```

### 1.2 NumPy로 구현한 신경망

```python
import numpy as np
import matplotlib.pyplot as plt

class NeuralNetwork:
    def __init__(self, layer_dims, activation='relu'):
        """
        신경망 초기화
        
        Args:
            layer_dims: 각 층의 뉴런 수 리스트 [input_size, hidden1, hidden2, ..., output_size]
            activation: 활성화 함수 ('relu', 'sigmoid', 'tanh')
        """
        self.layer_dims = layer_dims
        self.activation = activation
        self.parameters = {}
        self.costs = []
        self._initialize_parameters()
    
    def _initialize_parameters(self):
        """가중치 초기화 (He 초기화)"""
        np.random.seed(42)
        
        for l in range(1, len(self.layer_dims)):
            self.parameters[f'W{l}'] = np.random.randn(self.layer_dims[l], self.layer_dims[l-1]) * np.sqrt(2.0 / self.layer_dims[l-1])
            self.parameters[f'b{l}'] = np.zeros((self.layer_dims[l], 1))
    
    def _activation_function(self, Z, activation):
        """활성화 함수"""
        if activation == 'relu':
            return np.maximum(0, Z)
        elif activation == 'sigmoid':
            Z = np.clip(Z, -500, 500)  # 수치적 안정성
            return 1 / (1 + np.exp(-Z))
        elif activation == 'tanh':
            return np.tanh(Z)
        else:
            raise ValueError(f"지원하지 않는 활성화 함수: {activation}")
    
    def _activation_derivative(self, Z, activation):
        """활성화 함수 미분"""
        if activation == 'relu':
            return (Z > 0).astype(float)
        elif activation == 'sigmoid':
            s = self._activation_function(Z, 'sigmoid')
            return s * (1 - s)
        elif activation == 'tanh':
            return 1 - np.tanh(Z) ** 2
        else:
            raise ValueError(f"지원하지 않는 활성화 함수: {activation}")
    
    def _forward_propagation(self, X):
        """순전파"""
        caches = {}
        A = X
        L = len(self.layer_dims) - 1  # 층의 수
        
        # 은닉층들 (ReLU 사용)
        for l in range(1, L):
            A_prev = A
            W = self.parameters[f'W{l}']
            b = self.parameters[f'b{l}']
            
            Z = np.dot(W, A_prev) + b
            A = self._activation_function(Z, self.activation)
            
            caches[f'A{l-1}'] = A_prev
            caches[f'Z{l}'] = Z
            caches[f'A{l}'] = A
        
        # 출력층 (Sigmoid 사용)
        A_prev = A
        W = self.parameters[f'W{L}']
        b = self.parameters[f'b{L}']
        
        Z = np.dot(W, A_prev) + b
        AL = self._activation_function(Z, 'sigmoid')
        
        caches[f'A{L-1}'] = A_prev
        caches[f'Z{L}'] = Z
        caches[f'A{L}'] = AL
        
        return AL, caches
    
    def _compute_cost(self, AL, Y):
        """교차 엔트로피 비용 함수"""
        m = Y.shape[1]
        
        # 수치적 안정성을 위한 클리핑
        AL = np.clip(AL, 1e-15, 1 - 1e-15)
        
        cost = -1/m * np.sum(Y * np.log(AL) + (1 - Y) * np.log(1 - AL))
        return cost
    
    def _backward_propagation(self, AL, Y, caches):
        """역전파"""
        gradients = {}
        L = len(self.layer_dims) - 1
        m = AL.shape[1]
        
        # 출력층 기울기
        dZL = AL - Y
        gradients[f'dW{L}'] = 1/m * np.dot(dZL, caches[f'A{L-1}'].T)
        gradients[f'db{L}'] = 1/m * np.sum(dZL, axis=1, keepdims=True)
        gradients[f'dA{L-1}'] = np.dot(self.parameters[f'W{L}'].T, dZL)
        
        # 은닉층들 역전파
        for l in reversed(range(1, L)):
            dA = gradients[f'dA{l}']
            dZ = dA * self._activation_derivative(caches[f'Z{l}'], self.activation)
            
            gradients[f'dW{l}'] = 1/m * np.dot(dZ, caches[f'A{l-1}'].T)
            gradients[f'db{l}'] = 1/m * np.sum(dZ, axis=1, keepdims=True)
            
            if l > 1:
                gradients[f'dA{l-1}'] = np.dot(self.parameters[f'W{l}'].T, dZ)
        
        return gradients
    
    def _update_parameters(self, gradients, learning_rate):
        """매개변수 업데이트"""
        L = len(self.layer_dims) - 1
        
        for l in range(1, L + 1):
            self.parameters[f'W{l}'] -= learning_rate * gradients[f'dW{l}']
            self.parameters[f'b{l}'] -= learning_rate * gradients[f'db{l}']
    
    def fit(self, X, Y, learning_rate=0.01, num_iterations=1000, print_cost=True):
        """신경망 학습"""
        for i in range(num_iterations):
            # 순전파
            AL, caches = self._forward_propagation(X)
            
            # 비용 계산
            cost = self._compute_cost(AL, Y)
            self.costs.append(cost)
            
            # 역전파
            gradients = self._backward_propagation(AL, Y, caches)
            
            # 매개변수 업데이트
            self._update_parameters(gradients, learning_rate)
            
            # 진행 상황 출력
            if print_cost and i % 100 == 0:
                print(f"반복 {i}: 비용 = {cost:.6f}")
    
    def predict(self, X):
        """예측"""
        AL, _ = self._forward_propagation(X)
        predictions = (AL > 0.5).astype(int)
        return predictions
    
    def predict_proba(self, X):
        """확률 예측"""
        AL, _ = self._forward_propagation(X)
        return AL
    
    def plot_cost(self):
        """비용 함수 그래프"""
        plt.figure(figsize=(10, 6))
        plt.plot(self.costs)
        plt.title('Cost Function During Training')
        plt.xlabel('Iterations (per 100)')
        plt.ylabel('Cost')
        plt.grid(True)
        plt.show()

# 사용 예시
if __name__ == "__main__":
    # 데이터 생성
    np.random.seed(42)
    m = 400  # 샘플 수
    X = np.random.randn(2, m)
    Y = ((X[0, :] ** 2 + X[1, :] ** 2) < 1).astype(int).reshape(1, m)  # 원형 경계
    
    # 신경망 생성 및 학습
    nn = NeuralNetwork([2, 10, 5, 1], activation='relu')
    nn.fit(X, Y, learning_rate=0.1, num_iterations=2000)

    print(f"훈련 정확도: {accuracy:.2f}%")
    
    nn.plot_cost()
```

## 2. 합성곱 신경망 (CNN) 상세 구현

### 2.1 수학적 기초

**합성곱 연산 (Convolution)**
```
(f * g)(t) = Σᵢ f(i) * g(t - i)
```

**2D 합성곱**
```
Y(i,j) = Σₘ Σₙ X(i+m, j+n) * K(m,n)
```

**패딩과 스트라이드**
```
출력 크기 = (입력 크기 + 2*패딩 - 필터 크기) / 스트라이드 + 1
```

### 2.2 NumPy CNN 구현

```python
class ConvolutionalLayer:
    def __init__(self, input_channels, output_channels, kernel_size, stride=1, padding=0):
        """
        합성곱 층 초기화
        
        Args:
            input_channels: 입력 채널 수
            output_channels: 출력 채널 수 (필터 수)
            kernel_size: 커널 크기
            stride: 스트라이드
            padding: 패딩
        """
        self.input_channels = input_channels
        self.output_channels = output_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # Xavier 초기화
        self.weights = np.random.randn(output_channels, input_channels, kernel_size, kernel_size) * \
                      np.sqrt(2.0 / (input_channels * kernel_size * kernel_size))
        self.bias = np.zeros((output_channels, 1))
        
        # 캐시
        self.cache = None
    
    def _add_padding(self, X):
        """패딩 추가"""
        if self.padding > 0:
            return np.pad(X, ((0, 0), (0, 0), (self.padding, self.padding), (self.padding, self.padding)), 
                         mode='constant', constant_values=0)
        return X
    
    def forward(self, X):
        """
        순전파
        
        Args:
            X: 입력 (batch_size, input_channels, height, width)
        
        Returns:
            output: 출력 (batch_size, output_channels, out_height, out_width)
        """
        batch_size, _, input_height, input_width = X.shape
        
        # 패딩 적용
        X_padded = self._add_padding(X)
        
        # 출력 크기 계산
        out_height = (input_height + 2 * self.padding - self.kernel_size) // self.stride + 1
        out_width = (input_width + 2 * self.padding - self.kernel_size) // self.stride + 1
        
        # 출력 초기화
        output = np.zeros((batch_size, self.output_channels, out_height, out_width))
        
        # 합성곱 연산
        for n in range(batch_size):  # 배치
            for f in range(self.output_channels):  # 필터
                for h in range(out_height):  # 높이
                    for w in range(out_width):  # 너비
                        h_start = h * self.stride
                        h_end = h_start + self.kernel_size
                        w_start = w * self.stride
                        w_end = w_start + self.kernel_size
                        
                        # 합성곱 계산
                        output[n, f, h, w] = np.sum(
                            X_padded[n, :, h_start:h_end, w_start:w_end] * self.weights[f]
                        ) + self.bias[f]
        
        # 역전파를 위한 캐시 저장
        self.cache = X_padded
        
        return output
    
    def backward(self, dout):
        """
        역전파
        
        Args:
            dout: 출력에 대한 기울기
        
        Returns:
            dx: 입력에 대한 기울기
        """
        X_padded = self.cache
        batch_size, _, input_height, input_width = X_padded.shape
        _, _, out_height, out_width = dout.shape
        
        # 기울기 초기화
        dX_padded = np.zeros_like(X_padded)
        dW = np.zeros_like(self.weights)
        db = np.zeros_like(self.bias)
        
        # 역전파 계산
        for n in range(batch_size):
            for f in range(self.output_channels):
                for h in range(out_height):
                    for w in range(out_width):
                        h_start = h * self.stride
                        h_end = h_start + self.kernel_size
                        w_start = w * self.stride
                        w_end = w_start + self.kernel_size
                        
                        # 가중치 기울기
                        dW[f] += X_padded[n, :, h_start:h_end, w_start:w_end] * dout[n, f, h, w]
                        
                        # 입력 기울기
                        dX_padded[n, :, h_start:h_end, w_start:w_end] += \
                            self.weights[f] * dout[n, f, h, w]
                
                # 편향 기울기
                db[f] = np.sum(dout[:, f, :, :])
        
        # 패딩 제거
        if self.padding > 0:
            dX = dX_padded[:, :, self.padding:-self.padding, self.padding:-self.padding]
        else:
            dX = dX_padded
        
        # 가중치 업데이트 (여기서는 기울기만 반환)
        self.dW = dW
        self.db = db
        
        return dX

class MaxPoolingLayer:
    def __init__(self, pool_size=2, stride=2):
        """
        최대 풀링 층 초기화
        
        Args:
            pool_size: 풀링 윈도우 크기
            stride: 스트라이드
        """
        self.pool_size = pool_size
        self.stride = stride
        self.cache = None
    
    def forward(self, X):
        """순전파"""
        batch_size, channels, input_height, input_width = X.shape
        
        out_height = (input_height - self.pool_size) // self.stride + 1
        out_width = (input_width - self.pool_size) // self.stride + 1
        
        output = np.zeros((batch_size, channels, out_height, out_width))
        
        for n in range(batch_size):
            for c in range(channels):
                for h in range(out_height):
                    for w in range(out_width):
                        h_start = h * self.stride
                        h_end = h_start + self.pool_size
                        w_start = w * self.stride
                        w_end = w_start + self.pool_size
                        
                        # 최대값 선택
                        output[n, c, h, w] = np.max(X[n, c, h_start:h_end, w_start:w_end])
        
        self.cache = X
        return output
    
    def backward(self, dout):
        """역전파"""
        X = self.cache
        batch_size, channels, input_height, input_width = X.shape
        _, _, out_height, out_width = dout.shape
        
        dX = np.zeros_like(X)
        
        for n in range(batch_size):
            for c in range(channels):
                for h in range(out_height):
                    for w in range(out_width):
                        h_start = h * self.stride
                        h_end = h_start + self.pool_size
                        w_start = w * self.stride
                        w_end = w_start + self.pool_size
                        
                        # 최대값 위치 찾기
                        window = X[n, c, h_start:h_end, w_start:w_end]
                        mask = (window == np.max(window))
                        dX[n, c, h_start:h_end, w_start:w_end] += mask * dout[n, c, h, w]
        
        return dX

class SimpleCNN:
    def __init__(self):
        """간단한 CNN 모델"""
        # 층 정의
        self.conv1 = ConvolutionalLayer(1, 16, 3, padding=1)  # 28x28 -> 28x28
        self.pool1 = MaxPoolingLayer(2, 2)  # 28x28 -> 14x14
        self.conv2 = ConvolutionalLayer(16, 32, 3, padding=1)  # 14x14 -> 14x14
        self.pool2 = MaxPoolingLayer(2, 2)  # 14x14 -> 7x7
        
        # 완전연결층 (7x7x32 = 1568 -> 10)
        self.fc_input_size = 7 * 7 * 32
        self.fc_weights = np.random.randn(10, self.fc_input_size) * 0.01
        self.fc_bias = np.zeros((10, 1))
    
    def relu(self, x):
        """ReLU 활성화 함수"""
        return np.maximum(0, x)
    
    def relu_derivative(self, x):
        """ReLU 미분"""
        return (x > 0).astype(float)
    
    def softmax(self, x):
        """소프트맥스 함수"""
        exp_x = np.exp(x - np.max(x, axis=0, keepdims=True))  # 수치적 안정성
        return exp_x / np.sum(exp_x, axis=0, keepdims=True)
    
    def forward(self, X):
        """순전파"""
        # 합성곱층 1
        conv1_out = self.conv1.forward(X)
        relu1_out = self.relu(conv1_out)
        pool1_out = self.pool1.forward(relu1_out)
        
        # 합성곱층 2
        conv2_out = self.conv2.forward(pool1_out)
        relu2_out = self.relu(conv2_out)
        pool2_out = self.pool2.forward(relu2_out)
        
        # 평탄화
        batch_size = X.shape[0]
        flattened = pool2_out.reshape(batch_size, -1).T  # (features, batch_size)
        
        # 완전연결층
        fc_out = np.dot(self.fc_weights, flattened) + self.fc_bias
        output = self.softmax(fc_out)
        
        # 캐시 저장
        self.cache = {
            'conv1_out': conv1_out,
            'relu1_out': relu1_out,
            'pool1_out': pool1_out,
            'conv2_out': conv2_out,
            'relu2_out': relu2_out,
            'pool2_out': pool2_out,
            'flattened': flattened,
            'fc_out': fc_out
        }
        
        return output
    
    def compute_loss(self, predictions, labels):
        """교차 엔트로피 손실"""
        batch_size = predictions.shape[1]
        loss = -np.sum(labels * np.log(predictions + 1e-15)) / batch_size
        return loss
    
    def backward(self, predictions, labels):
        """역전파"""
        batch_size = predictions.shape[1]
        
        # 출력층 기울기
        dfc_out = predictions - labels
        
        # 완전연결층 기울기
        dfc_weights = np.dot(dfc_out, self.cache['flattened'].T) / batch_size
        dfc_bias = np.sum(dfc_out, axis=1, keepdims=True) / batch_size
        dflattened = np.dot(self.fc_weights.T, dfc_out)
        
        # 평탄화 역연산
        dpool2_out = dflattened.T.reshape(self.cache['pool2_out'].shape)
        
        # 풀링층 2 역전파
        drelu2_out = self.pool2.backward(dpool2_out)
        
        # ReLU 2 역전파
        dconv2_out = drelu2_out * self.relu_derivative(self.cache['conv2_out'])
        
        # 합성곱층 2 역전파
        dpool1_out = self.conv2.backward(dconv2_out)
        
        # 풀링층 1 역전파
        drelu1_out = self.pool1.backward(dpool1_out)
        
        # ReLU 1 역전파
        dconv1_out = drelu1_out * self.relu_derivative(self.cache['conv1_out'])
        
        # 합성곱층 1 역전파
        dx = self.conv1.backward(dconv1_out)
        
        return dfc_weights, dfc_bias
    
    def update_parameters(self, dfc_weights, dfc_bias, learning_rate=0.001):
        """매개변수 업데이트"""
        # 완전연결층 업데이트
        self.fc_weights -= learning_rate * dfc_weights
        self.fc_bias -= learning_rate * dfc_bias
        
        # 합성곱층 업데이트
        self.conv1.weights -= learning_rate * self.conv1.dW
        self.conv1.bias -= learning_rate * self.conv1.db
        self.conv2.weights -= learning_rate * self.conv2.dW
        self.conv2.bias -= learning_rate * self.conv2.db
```

## 3. 순환 신경망 (RNN) 구현

### 3.1 수학적 기초

**바닐라 RNN**
```
h_t = tanh(W_hh * h_{t-1} + W_xh * x_t + b_h)
y_t = W_hy * h_t + b_y
```

**LSTM 셀**
```
f_t = σ(W_f · [h_{t-1}, x_t] + b_f)  # 망각 게이트
i_t = σ(W_i · [h_{t-1}, x_t] + b_i)  # 입력 게이트
C̃_t = tanh(W_C · [h_{t-1}, x_t] + b_C)  # 후보 값
C_t = f_t * C_{t-1} + i_t * C̃_t  # 셀 상태
o_t = σ(W_o · [h_{t-1}, x_t] + b_o)  # 출력 게이트
h_t = o_t * tanh(C_t)  # 은닉 상태
```

### 3.2 Python 구현

```python
class VanillaRNN:
    def __init__(self, input_size, hidden_size, output_size):
        """
        바닐라 RNN 초기화
        
        Args:
            input_size: 입력 차원
            hidden_size: 은닉 상태 차원
            output_size: 출력 차원
        """
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        
        # 가중치 초기화
        self.Wxh = np.random.randn(hidden_size, input_size) * 0.01
        self.Whh = np.random.randn(hidden_size, hidden_size) * 0.01
        self.Why = np.random.randn(output_size, hidden_size) * 0.01
        self.bh = np.zeros((hidden_size, 1))
        self.by = np.zeros((output_size, 1))
    
    def forward(self, inputs, h_prev):
        """
        순전파
        
        Args:
            inputs: 입력 시퀀스 (seq_len, input_size)
            h_prev: 이전 은닉 상태 (hidden_size, 1)
        
        Returns:
            outputs: 출력 시퀀스
            hidden_states: 은닉 상태들
        """
        seq_len = len(inputs)
        outputs = []
        hidden_states = [h_prev]
        
        h = h_prev
        for t in range(seq_len):
            x = inputs[t].reshape(-1, 1)
            
            # 은닉 상태 계산
            h = np.tanh(np.dot(self.Wxh, x) + np.dot(self.Whh, h) + self.bh)
            
            # 출력 계산
            y = np.dot(self.Why, h) + self.by
            
            outputs.append(y)
            hidden_states.append(h)
        
        return outputs, hidden_states
    
    def backward(self, inputs, outputs, targets, hidden_states):
        """
        BPTT (Backpropagation Through Time)
        
        Args:
            inputs: 입력 시퀀스
            outputs: 출력 시퀀스
            targets: 타겟 시퀀스
            hidden_states: 은닉 상태들
        
        Returns:
            gradients: 매개변수 기울기들
        """
        seq_len = len(inputs)
        
        # 기울기 초기화
        dWxh = np.zeros_like(self.Wxh)
        dWhh = np.zeros_like(self.Whh)
        dWhy = np.zeros_like(self.Why)
        dbh = np.zeros_like(self.bh)
        dby = np.zeros_like(self.by)
        dh_next = np.zeros_like(hidden_states[0])
        
        # 역방향으로 전파
        for t in reversed(range(seq_len)):
            x = inputs[t].reshape(-1, 1)
            h = hidden_states[t + 1]
            h_prev = hidden_states[t]
            y = outputs[t]
            target = targets[t].reshape(-1, 1)
            
            # 출력층 기울기
            dy = y - target  # 평균 제곱 오차의 기울기
            dWhy += np.dot(dy, h.T)
            dby += dy
            
            # 은닉층 기울기
            dh = np.dot(self.Why.T, dy) + dh_next
            dh_raw = (1 - h * h) * dh  # tanh 미분
            
            # 매개변수 기울기
            dWxh += np.dot(dh_raw, x.T)
            dWhh += np.dot(dh_raw, h_prev.T)
            dbh += dh_raw
            
            # 다음 시간 단계로 전파
            dh_next = np.dot(self.Whh.T, dh_raw)
        
        # 기울기 클리핑 (기울기 폭발 방지)
        for gradient in [dWxh, dWhh, dWhy, dbh, dby]:
            np.clip(gradient, -5, 5, out=gradient)
        
        return dWxh, dWhh, dWhy, dbh, dby
    
    def update_parameters(self, gradients, learning_rate=0.001):
        """매개변수 업데이트"""
        dWxh, dWhh, dWhy, dbh, dby = gradients
        
        self.Wxh -= learning_rate * dWxh
        self.Whh -= learning_rate * dWhh
        self.Why -= learning_rate * dWhy
        self.bh -= learning_rate * dbh
        self.by -= learning_rate * dby

class LSTMCell:
    def __init__(self, input_size, hidden_size):
        """LSTM 셀 초기화"""
        self.input_size = input_size
        self.hidden_size = hidden_size
        
        # 가중치 초기화 (Xavier 초기화)
        std = 1.0 / np.sqrt(input_size + hidden_size)
        
        self.Wf = np.random.normal(0, std, (hidden_size, input_size + hidden_size))
        self.Wi = np.random.normal(0, std, (hidden_size, input_size + hidden_size))
        self.Wo = np.random.normal(0, std, (hidden_size, input_size + hidden_size))
        self.Wg = np.random.normal(0, std, (hidden_size, input_size + hidden_size))
        
        self.bf = np.zeros((hidden_size, 1))
        self.bi = np.zeros((hidden_size, 1))
        self.bo = np.zeros((hidden_size, 1))
        self.bg = np.zeros((hidden_size, 1))
    
    def sigmoid(self, x):
        """시그모이드 함수"""
        x = np.clip(x, -500, 500)
        return 1 / (1 + np.exp(-x))
    
    def forward(self, x, h_prev, C_prev):
        """
        LSTM 순전파
        
        Args:
            x: 입력 (input_size, 1)
            h_prev: 이전 은닉 상태 (hidden_size, 1)
            C_prev: 이전 셀 상태 (hidden_size, 1)
        
        Returns:
            h: 현재 은닉 상태
            C: 현재 셀 상태
            cache: 역전파를 위한 중간값들
        """
        # 입력과 이전 은닉 상태 연결
        concat = np.vstack([x, h_prev])
        
        # 게이트 계산
        f = self.sigmoid(np.dot(self.Wf, concat) + self.bf)  # 망각 게이트
        i = self.sigmoid(np.dot(self.Wi, concat) + self.bi)  # 입력 게이트
        o = self.sigmoid(np.dot(self.Wo, concat) + self.bo)  # 출력 게이트
        g = np.tanh(np.dot(self.Wg, concat) + self.bg)       # 후보 값
        
        # 셀 상태와 은닉 상태 업데이트
        C = f * C_prev + i * g
        h = o * np.tanh(C)
        
        # 역전파를 위한 캐시
        cache = {
            'x': x, 'h_prev': h_prev, 'C_prev': C_prev,
            'f': f, 'i': i, 'o': o, 'g': g,
            'C': C, 'concat': concat
        }
        
        return h, C, cache
    
    def backward(self, dh, dC, cache):
        """LSTM 역전파"""
        x, h_prev, C_prev, f, i, o, g, C, concat = cache.values()
        
        # 은닉 상태 기울기로부터 셀 상태 기울기 계산
        dC = dC + dh * o * (1 - np.tanh(C) ** 2)
        
        # 게이트 기울기
        df = dC * C_prev * f * (1 - f)
        di = dC * g * i * (1 - i)
        do = dh * np.tanh(C) * o * (1 - o)
        dg = dC * i * (1 - g ** 2)
        
        # 가중치 기울기
        dWf = np.dot(df, concat.T)
        dWi = np.dot(di, concat.T)
        dWo = np.dot(do, concat.T)
        dWg = np.dot(dg, concat.T)
        
        # 편향 기울기
        dbf = df
        dbi = di
        dbo = do
        dbg = dg
        
        # 입력 기울기
        dconcat = (np.dot(self.Wf.T, df) + np.dot(self.Wi.T, di) + 
                   np.dot(self.Wo.T, do) + np.dot(self.Wg.T, dg))
        
        dx = dconcat[:self.input_size]
        dh_prev = dconcat[self.input_size:]
        dC_prev = dC * f
        
        gradients = {
            'dWf': dWf, 'dWi': dWi, 'dWo': dWo, 'dWg': dWg,
            'dbf': dbf, 'dbi': dbi, 'dbo': dbo, 'dbg': dbg
        }
        
        return dx, dh_prev, dC_prev, gradients

class LSTM:
    def __init__(self, input_size, hidden_size, output_size, num_layers=1):
        """
        LSTM 네트워크
        
        Args:
            input_size: 입력 차원
            hidden_size: 은닉 상태 차원
            output_size: 출력 차원
            num_layers: LSTM 층 수
        """
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size
        self.num_layers = num_layers
        
        # LSTM 셀들 초기화
        self.lstm_cells = []
        for layer in range(num_layers):
            if layer == 0:
                cell_input_size = input_size
            else:
                cell_input_size = hidden_size
            
            self.lstm_cells.append(LSTMCell(cell_input_size, hidden_size))
        
        # 출력층
        self.Wy = np.random.randn(output_size, hidden_size) * 0.01
        self.by = np.zeros((output_size, 1))
    
    def forward(self, inputs):
        """
        LSTM 순전파
        
        Args:
            inputs: 입력 시퀀스 (seq_len, input_size)
        
        Returns:
            outputs: 출력 시퀀스
            caches: 역전파를 위한 캐시들
        """
        seq_len = len(inputs)
        outputs = []
        caches = []
        
        # 초기 은닉/셀 상태
        h = [np.zeros((self.hidden_size, 1)) for _ in range(self.num_layers)]
        C = [np.zeros((self.hidden_size, 1)) for _ in range(self.num_layers)]
        
        for t in range(seq_len):
            x = inputs[t].reshape(-1, 1)
            layer_caches = []
            
            # 각 LSTM 층 통과
            for layer in range(self.num_layers):
                h[layer], C[layer], cache = self.lstm_cells[layer].forward(x, h[layer], C[layer])
                layer_caches.append(cache)
                x = h[layer]  # 다음 층의 입력
            
            # 출력층
            y = np.dot(self.Wy, h[-1]) + self.by
            outputs.append(y)
            caches.append(layer_caches)
        
        return outputs, caches
    
    def compute_loss(self, outputs, targets):
        """평균 제곱 오차 손실"""
        loss = 0
        for t in range(len(outputs)):
            loss += np.sum((outputs[t] - targets[t].reshape(-1, 1)) ** 2)
        return loss / len(outputs)
```
