# 머신러닝 알고리즘 상세 구현 및 수식 정리

## 1. 선형 회귀 (Linear Regression)

### 1.1 수학적 기초

**가설 함수 (Hypothesis Function)**
```
h(x) = θ₀ + θ₁x₁ + θ₂x₂ + ... + θₙxₙ = θᵀx
```

**비용 함수 (Cost Function) - MSE**
```
J(θ) = (1/2m) Σᵢ₌₁ᵐ (h(xᵢ) - yᵢ)²
```

**경사하강법 업데이트 규칙**
```
θⱼ := θⱼ - α ∂J(θ)/∂θⱼ
∂J(θ)/∂θⱼ = (1/m) Σᵢ₌₁ᵐ (h(xᵢ) - yᵢ)xᵢⱼ
```

### 1.2 Python 구현

```python
import numpy as np
import matplotlib.pyplot as plt

class LinearRegression:
    def __init__(self, learning_rate=0.01, max_iter=1000):
        self.learning_rate = learning_rate
        self.max_iter = max_iter
        self.cost_history = []
    
    def fit(self, X, y):
        """
        선형 회귀 모델 학습
        
        Args:
            X: 특성 행렬 (m x n)
            y: 타겟 벡터 (m x 1)
        """
        # 편향 항 추가 (intercept term)
        m, n = X.shape
        self.X = np.c_[np.ones((m, 1)), X]  # bias term 추가
        self.y = y.reshape(-1, 1)
        
        # 가중치 초기화
        self.theta = np.random.normal(0, 0.01, (n + 1, 1))
        
        # 경사하강법
        for i in range(self.max_iter):
            # 예측값 계산
            predictions = self.X.dot(self.theta)
            
            # 비용 계산
            cost = self._compute_cost(predictions)
            self.cost_history.append(cost)
            
            # 기울기 계산
            gradients = (1/m) * self.X.T.dot(predictions - self.y)
            
            # 가중치 업데이트
            self.theta -= self.learning_rate * gradients
            
            # 수렴 체크
            if i > 0 and abs(self.cost_history[-2] - self.cost_history[-1]) < 1e-6:
                print(f"수렴 완료: {i+1} iterations")
                break
    
    def _compute_cost(self, predictions):
        """MSE 비용 함수 계산"""
        m = len(predictions)
        return (1/(2*m)) * np.sum((predictions - self.y)**2)
    
    def predict(self, X):
        """예측 수행"""
        X_with_bias = np.c_[np.ones((X.shape[0], 1)), X]
        return X_with_bias.dot(self.theta)
    
    def plot_cost_history(self):
        """비용 함수 변화 그래프"""
        plt.figure(figsize=(10, 6))
        plt.plot(self.cost_history)
        plt.title('Cost Function Over Iterations')
        plt.xlabel('Iterations')
        plt.ylabel('Cost')
        plt.grid(True)
        plt.show()

# 사용 예시
np.random.seed(42)
X = 2 * np.random.rand(100, 1)
y = 4 + 3 * X + np.random.randn(100, 1)

model = LinearRegression(learning_rate=0.1, max_iter=1000)
model.fit(X, y)
predictions = model.predict(X)

print(f"학습된 파라미터: θ₀ = {model.theta[0][0]:.3f}, θ₁ = {model.theta[1][0]:.3f}")
```

### 1.3 정규방정식 (Normal Equation) 구현

```python
class LinearRegressionNormalEquation:
    def fit(self, X, y):
        """
        정규방정식을 사용한 해석적 해법
        θ = (XᵀX)⁻¹Xᵀy
        """
        X_with_bias = np.c_[np.ones((X.shape[0], 1)), X]
        self.theta = np.linalg.inv(X_with_bias.T.dot(X_with_bias)).dot(X_with_bias.T).dot(y)
    
    def predict(self, X):
        X_with_bias = np.c_[np.ones((X.shape[0], 1)), X]
        return X_with_bias.dot(self.theta)
```

## 2. 로지스틱 회귀 (Logistic Regression)

### 2.1 수학적 기초

**시그모이드 함수**
```
σ(z) = 1 / (1 + e⁻ᶻ)
```

**가설 함수**
```
h(x) = σ(θᵀx) = 1 / (1 + e⁻θᵀˣ)
```

**비용 함수 (Log-Likelihood)**
```
J(θ) = -(1/m) Σᵢ₌₁ᵐ [yᵢlog(h(xᵢ)) + (1-yᵢ)log(1-h(xᵢ))]
```

**기울기**
```
∂J(θ)/∂θⱼ = (1/m) Σᵢ₌₁ᵐ (h(xᵢ) - yᵢ)xᵢⱼ
```

### 2.2 Python 구현

```python
class LogisticRegression:
    def __init__(self, learning_rate=0.01, max_iter=1000):
        self.learning_rate = learning_rate
        self.max_iter = max_iter
        self.cost_history = []
    
    def sigmoid(self, z):
        """시그모이드 활성화 함수"""
        # 수치적 안정성을 위한 clipping
        z = np.clip(z, -500, 500)
        return 1 / (1 + np.exp(-z))
    
    def fit(self, X, y):
        """로지스틱 회귀 모델 학습"""
        m, n = X.shape
        self.X = np.c_[np.ones((m, 1)), X]  # bias term 추가
        self.y = y.reshape(-1, 1)
        
        # 가중치 초기화
        self.theta = np.random.normal(0, 0.01, (n + 1, 1))
        
        # 경사하강법
        for i in range(self.max_iter):
            # 예측 확률 계산
            z = self.X.dot(self.theta)
            predictions = self.sigmoid(z)
            
            # 비용 계산
            cost = self._compute_cost(predictions)
            self.cost_history.append(cost)
            
            # 기울기 계산
            gradients = (1/m) * self.X.T.dot(predictions - self.y)
            
            # 가중치 업데이트
            self.theta -= self.learning_rate * gradients
            
            # 수렴 체크
            if i > 0 and abs(self.cost_history[-2] - self.cost_history[-1]) < 1e-6:
                print(f"수렴 완료: {i+1} iterations")
                break
    
    def _compute_cost(self, predictions):
        """로그 우도 비용 함수"""
        m = len(predictions)
        # 수치적 안정성을 위한 epsilon 추가
        epsilon = 1e-15
        predictions = np.clip(predictions, epsilon, 1 - epsilon)
        
        cost = -(1/m) * np.sum(
            self.y * np.log(predictions) + (1 - self.y) * np.log(1 - predictions)
        )
        return cost
    
    def predict_proba(self, X):
        """확률 예측"""
        X_with_bias = np.c_[np.ones((X.shape[0], 1)), X]
        return self.sigmoid(X_with_bias.dot(self.theta))
    
    def predict(self, X, threshold=0.5):
        """이진 분류 예측"""
        probabilities = self.predict_proba(X)
        return (probabilities >= threshold).astype(int)

# 사용 예시 - 이진 분류 데이터 생성
np.random.seed(42)
m = 100
X = np.random.randn(m, 2)
y = (X[:, 0] + X[:, 1] > 0).astype(int)

model = LogisticRegression(learning_rate=0.1, max_iter=1000)
model.fit(X, y)

# 정확도 계산
predictions = model.predict(X)
accuracy = np.mean(predictions.flatten() == y) * 100
print(f"훈련 정확도: {accuracy:.2f}%")
```

## 3. 서포트 벡터 머신 (SVM)

### 3.1 수학적 기초

**마진 최대화 문제**
```
minimize: (1/2)||w||²
subject to: yᵢ(wᵀxᵢ + b) ≥ 1 for all i
```

**소프트 마진 (Soft Margin)**
```
minimize: (1/2)||w||² + C Σᵢ₌₁ᵐ ξᵢ
subject to: yᵢ(wᵀxᵢ + b) ≥ 1 - ξᵢ, ξᵢ ≥ 0
```

**듀얼 문제**
```
maximize: Σᵢ₌₁ᵐ αᵢ - (1/2) Σᵢ₌₁ᵐ Σⱼ₌₁ᵐ αᵢαⱼyᵢyⱼxᵢᵀxⱼ
subject to: 0 ≤ αᵢ ≤ C, Σᵢ₌₁ᵐ αᵢyᵢ = 0
```

### 3.2 간단한 SVM 구현 (SMO 알고리즘)

```python
class SimpleSVM:
    def __init__(self, C=1.0, max_iter=1000, tol=1e-3):
        self.C = C
        self.max_iter = max_iter
        self.tol = tol
    
    def fit(self, X, y):
        """SVM 모델 학습 (단순화된 SMO)"""
        self.X = X
        self.y = y
        m, n = X.shape
        
        # 라그랑주 승수 초기화
        self.alphas = np.zeros(m)
        self.b = 0
        
        # SMO 알고리즘 (단순화)
        for iter_num in range(self.max_iter):
            num_changed_alphas = 0
            
            for i in range(m):
                # 예측값 계산
                Ei = self._predict_single(X[i]) - y[i]
                
                # KKT 조건 확인
                if ((y[i] * Ei < -self.tol and self.alphas[i] < self.C) or
                    (y[i] * Ei > self.tol and self.alphas[i] > 0)):
                    
                    # 두 번째 알파 선택 (랜덤)
                    j = np.random.randint(0, m)
                    while j == i:
                        j = np.random.randint(0, m)
                    
                    Ej = self._predict_single(X[j]) - y[j]
                    
                    # 알파 업데이트
                    alpha_i_old = self.alphas[i]
                    alpha_j_old = self.alphas[j]
                    
                    # L, H 계산
                    if y[i] != y[j]:
                        L = max(0, self.alphas[j] - self.alphas[i])
                        H = min(self.C, self.C + self.alphas[j] - self.alphas[i])
                    else:
                        L = max(0, self.alphas[i] + self.alphas[j] - self.C)
                        H = min(self.C, self.alphas[i] + self.alphas[j])
                    
                    if L == H:
                        continue
                    
                    # eta 계산
                    eta = 2 * X[i].dot(X[j]) - X[i].dot(X[i]) - X[j].dot(X[j])
                    if eta >= 0:
                        continue
                    
                    # 새로운 alpha_j 계산
                    self.alphas[j] -= (y[j] * (Ei - Ej)) / eta
                    self.alphas[j] = np.clip(self.alphas[j], L, H)
                    
                    if abs(self.alphas[j] - alpha_j_old) < 1e-5:
                        continue
                    
                    # alpha_i 업데이트
                    self.alphas[i] += y[i] * y[j] * (alpha_j_old - self.alphas[j])
                    
                    # b 업데이트
                    b1 = (self.b - Ei - y[i] * (self.alphas[i] - alpha_i_old) * X[i].dot(X[i]) -
                          y[j] * (self.alphas[j] - alpha_j_old) * X[i].dot(X[j]))
                    
                    b2 = (self.b - Ej - y[i] * (self.alphas[i] - alpha_i_old) * X[i].dot(X[j]) -
                          y[j] * (self.alphas[j] - alpha_j_old) * X[j].dot(X[j]))
                    
                    if 0 < self.alphas[i] < self.C:
                        self.b = b1
                    elif 0 < self.alphas[j] < self.C:
                        self.b = b2
                    else:
                        self.b = (b1 + b2) / 2
                    
                    num_changed_alphas += 1
            
            if num_changed_alphas == 0:
                break
        
        # 서포트 벡터 찾기
        self.support_vector_idx = self.alphas > 1e-5
        self.support_vectors = X[self.support_vector_idx]
        self.support_vector_labels = y[self.support_vector_idx]
        self.support_vector_alphas = self.alphas[self.support_vector_idx]
        
        print(f"서포트 벡터 개수: {len(self.support_vectors)}")
    
    def _predict_single(self, x):
        """단일 샘플에 대한 예측"""
        return np.sum(self.alphas * self.y * self.X.dot(x)) + self.b
    
    def predict(self, X):
        """예측 수행"""
        predictions = []
        for x in X:
            pred = self._predict_single(x)
            predictions.append(1 if pred >= 0 else -1)
        return np.array(predictions)
```

## 4. k-최근접 이웃 (k-NN)

### 4.1 수학적 기초

**유클리드 거리**
```
d(x, x') = √(Σᵢ₌₁ⁿ (xᵢ - x'ᵢ)²)
```

**맨하탄 거리**
```
d(x, x') = Σᵢ₌₁ⁿ |xᵢ - x'ᵢ|
```

**민코프스키 거리**
```
d(x, x') = (Σᵢ₌₁ⁿ |xᵢ - x'ᵢ|ᵖ)^(1/p)
```

### 4.2 Python 구현

```python
from collections import Counter

class KNearestNeighbors:
    def __init__(self, k=3, distance_metric='euclidean'):
        self.k = k
        self.distance_metric = distance_metric
    
    def fit(self, X, y):
        """학습 데이터 저장 (lazy learning)"""
        self.X_train = X
        self.y_train = y
    
    def _calculate_distance(self, x1, x2):
        """거리 계산"""
        if self.distance_metric == 'euclidean':
            return np.sqrt(np.sum((x1 - x2) ** 2))
        elif self.distance_metric == 'manhattan':
            return np.sum(np.abs(x1 - x2))
        elif self.distance_metric == 'chebyshev':
            return np.max(np.abs(x1 - x2))
        else:
            raise ValueError("지원하지 않는 거리 메트릭입니다.")
    
    def _get_neighbors(self, x):
        """k개의 최근접 이웃 찾기"""
        distances = []
        for i, x_train in enumerate(self.X_train):
            dist = self._calculate_distance(x, x_train)
            distances.append((dist, i))
        
        # 거리 기준 정렬
        distances.sort(key=lambda x: x[0])
        
        # k개의 최근접 이웃 인덱스 반환
        neighbors = []
        for i in range(self.k):
            neighbors.append(distances[i][1])
        
        return neighbors
    
    def predict(self, X):
        """예측 수행"""
        predictions = []
        for x in X:
            neighbors = self._get_neighbors(x)
            neighbor_labels = [self.y_train[i] for i in neighbors]
            
            # 분류: 다수결 투표
            if hasattr(self.y_train[0], '__iter__') and not isinstance(self.y_train[0], str):
                # 회귀의 경우
                prediction = np.mean(neighbor_labels)
            else:
                # 분류의 경우
                prediction = Counter(neighbor_labels).most_common(1)[0][0]
            
            predictions.append(prediction)
        
        return np.array(predictions)
    
    def predict_proba(self, X):
        """분류 확률 예측"""
        probabilities = []
        unique_classes = np.unique(self.y_train)
        
        for x in X:
            neighbors = self._get_neighbors(x)
            neighbor_labels = [self.y_train[i] for i in neighbors]
            
            class_counts = Counter(neighbor_labels)
            prob_dict = {}
            
            for class_label in unique_classes:
                prob_dict[class_label] = class_counts.get(class_label, 0) / self.k
            
            probabilities.append([prob_dict[class_label] for class_label in unique_classes])
        
        return np.array(probabilities), unique_classes

# 거리 가중 k-NN 구현
class WeightedKNN(KNearestNeighbors):
    def predict(self, X):
        """거리 가중 예측"""
        predictions = []
        for x in X:
            distances = []
            for i, x_train in enumerate(self.X_train):
                dist = self._calculate_distance(x, x_train)
                distances.append((dist, i, self.y_train[i]))
            
            # 거리 기준 정렬
            distances.sort(key=lambda x: x[0])
            
            # 거리 가중 투표
            if isinstance(self.y_train[0], (int, float)) and not isinstance(self.y_train[0], bool):
                # 회귀
                weighted_sum = 0
                weight_sum = 0
                for j in range(self.k):
                    dist, idx, label = distances[j]
                    weight = 1 / (dist + 1e-8)  # 0으로 나누기 방지
                    weighted_sum += weight * label
                    weight_sum += weight
                prediction = weighted_sum / weight_sum
            else:
                # 분류
                class_weights = {}
                for j in range(self.k):
                    dist, idx, label = distances[j]
                    weight = 1 / (dist + 1e-8)
                    if label in class_weights:
                        class_weights[label] += weight
                    else:
                        class_weights[label] = weight
                
                prediction = max(class_weights, key=class_weights.get)
            
            predictions.append(prediction)
        
        return np.array(predictions)
```

## 5. 나이브 베이즈 (Naive Bayes)

### 5.1 수학적 기초

**베이즈 정리**
```
P(C|X) = P(X|C) × P(C) / P(X)
```

**나이브 베이즈 가정 (독립성)**
```
P(X|C) = P(x₁|C) × P(x₂|C) × ... × P(xₙ|C)
```

**최대 사후 확률 (MAP)**
```
ĉ = argmax P(C|X) = argmax P(X|C) × P(C)
```

### 5.2 가우시안 나이브 베이즈 구현

```python
class GaussianNaiveBayes:
    def fit(self, X, y):
        """가우시안 나이브 베이즈 학습"""
        self.classes = np.unique(y)
        self.parameters = {}
        
        for class_label in self.classes:
            # 클래스별 데이터 분리
            class_data = X[y == class_label]
            
            self.parameters[class_label] = {
                'prior': len(class_data) / len(X),  # 사전 확률
                'mean': np.mean(class_data, axis=0),  # 평균
                'var': np.var(class_data, axis=0) + 1e-9  # 분산 (수치적 안정성)
            }
    
    def _gaussian_pdf(self, x, mean, var):
        """가우시안 확률 밀도 함수"""
        coefficient = 1 / np.sqrt(2 * np.pi * var)
        exponent = np.exp(-0.5 * ((x - mean) ** 2) / var)
        return coefficient * exponent
    
    def _predict_single(self, x):
        """단일 샘플 예측"""
        posteriors = {}
        
        for class_label in self.classes:
            params = self.parameters[class_label]
            prior = params['prior']
            
            # 가능도 계산 (독립성 가정)
            likelihood = np.prod(
                self._gaussian_pdf(x, params['mean'], params['var'])
            )
            
            # 사후 확률 (정규화 상수 제외)
            posteriors[class_label] = prior * likelihood
        
        # 최대 사후 확률 클래스 선택
        return max(posteriors, key=posteriors.get), posteriors
    
    def predict(self, X):
        """예측 수행"""
        predictions = []
        for x in X:
            pred, _ = self._predict_single(x)
            predictions.append(pred)
        return np.array(predictions)
    
    def predict_proba(self, X):
        """확률 예측"""
        probabilities = []
        for x in X:
            _, posteriors = self._predict_single(x)
            
            # 정규화
            total = sum(posteriors.values())
            normalized_probs = {k: v/total for k, v in posteriors.items()}
            
            prob_list = [normalized_probs[class_label] for class_label in self.classes]
            probabilities.append(prob_list)
        
        return np.array(probabilities)

# 멀티노미얼 나이브 베이즈 (텍스트 분류용)
class MultinomialNaiveBayes:
    def __init__(self, alpha=1.0):
        self.alpha = alpha  # 라플라스 스무딩
    
    def fit(self, X, y):
        """멀티노미얼 나이브 베이즈 학습"""
        self.classes = np.unique(y)
        self.parameters = {}
        
        for class_label in self.classes:
            class_mask = (y == class_label)
            class_data = X[class_mask]
            
            # 클래스 사전 확률
            prior = np.sum(class_mask) / len(y)
            
            # 특성별 조건부 확률 (라플라스 스무딩 적용)
            feature_counts = np.sum(class_data, axis=0)
            total_count = np.sum(feature_counts)
            
            # P(feature|class) with Laplace smoothing
            feature_probs = (feature_counts + self.alpha) / (total_count + self.alpha * X.shape[1])
            
            self.parameters[class_label] = {
                'prior': prior,
                'feature_log_probs': np.log(feature_probs)
            }
    
    def predict(self, X):
        """예측 수행"""
        log_probs = self._calculate_log_probs(X)
        return self.classes[np.argmax(log_probs, axis=1)]
    
    def _calculate_log_probs(self, X):
        """로그 확률 계산"""
        log_probs = []
        
        for class_label in self.classes:
            params = self.parameters[class_label]
            
            # 로그 사전 확률
            log_prior = np.log(params['prior'])
            
            # 로그 가능도 (각 특성의 로그 확률 합)
            log_likelihood = X.dot(params['feature_log_probs'])
            
            # 로그 사후 확률
            log_posterior = log_prior + log_likelihood
            log_probs.append(log_posterior)
        
        return np.array(log_probs).T
```

## 6. 의사결정 트리 (Decision Tree)

### 6.1 수학적 기초

**지니 불순도 (Gini Impurity)**
```
Gini(S) = 1 - Σᶜᵢ₌₁ pᵢ²
```

**엔트로피 (Entropy)**
```
Entropy(S) = -Σᶜᵢ₌₁ pᵢ log₂(pᵢ)
```

**정보 이득 (Information Gain)**
```
IG(S, A) = Entropy(S) - Σᵥ∈Values(A) |Sᵥ|/|S| × Entropy(Sᵥ)
```

### 6.2 Python 구현

```python
from collections import Counter

class DecisionTreeNode:
    def __init__(self):
        self.feature_idx = None
        self.threshold = None
        self.left = None
        self.right = None
        self.prediction = None
        self.gini = None

class DecisionTreeClassifier:
    def __init__(self, max_depth=None, min_samples_split=2, min_samples_leaf=1):
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.min_samples_leaf = min_samples_leaf
    
    def fit(self, X, y):
        """의사결정 트리 학습"""
        self.n_features = X.shape[1]
        self.n_classes = len(np.unique(y))
        self.root = self._build_tree(X, y, depth=0)
    
    def _gini_impurity(self, y):
        """지니 불순도 계산"""
        if len(y) == 0:
            return 0
        
        class_counts = Counter(y)
        impurity = 1.0
        
        for count in class_counts.values():
            prob = count / len(y)
            impurity -= prob ** 2
        
        return impurity
    
    def _entropy(self, y):
        """엔트로피 계산"""
        if len(y) == 0:
            return 0
        
        class_counts = Counter(y)
        entropy = 0.0
        
        for count in class_counts.values():
            prob = count / len(y)
            if prob > 0:
                entropy -= prob * np.log2(prob)
        
        return entropy
    
    def _best_split(self, X, y):
        """최적 분할 찾기"""
        best_gini = float('inf')
        best_feature = None
        best_threshold = None
        
        current_gini = self._gini_impurity(y)
        
        for feature_idx in range(self.n_features):
            thresholds = np.unique(X[:, feature_idx])
            
            for threshold in thresholds:
                # 분할
                left_mask = X[:, feature_idx] <= threshold
                right_mask = ~left_mask
                
                if np.sum(left_mask) < self.min_samples_leaf or \
                   np.sum(right_mask) < self.min_samples_leaf:
                    continue
                
                # 가중 평균 지니 불순도
                left_gini = self._gini_impurity(y[left_mask])
                right_gini = self._gini_impurity(y[right_mask])
                
                n_left, n_right = np.sum(left_mask), np.sum(right_mask)
                weighted_gini = (n_left * left_gini + n_right * right_gini) / len(y)
                
                if weighted_gini < best_gini:
                    best_gini = weighted_gini
                    best_feature = feature_idx
                    best_threshold = threshold
        
        return best_feature, best_threshold, best_gini
    
    def _build_tree(self, X, y, depth):
        """재귀적으로 트리 구축"""
        node = DecisionTreeNode()
        node.gini = self._gini_impurity(y)
        
        # 다수 클래스를 기본 예측값으로 설정
        node.prediction = Counter(y).most_common(1)[0][0]
        
        # 종료 조건 확인
        if (self.max_depth is not None and depth >= self.max_depth) or \
           len(np.unique(y)) == 1 or \
           len(y) < self.min_samples_split:
            return node
        
        # 최적 분할 찾기
        feature_idx, threshold, gini = self._best_split(X, y)
        
        if feature_idx is None:  # 더 이상 분할할 수 없음
            return node
        
        # 노드 정보 저장
        node.feature_idx = feature_idx
        node.threshold = threshold
        
        # 자식 노드 생성
        left_mask = X[:, feature_idx] <= threshold
        right_mask = ~left_mask
        
        node.left = self._build_tree(X[left_mask], y[left_mask], depth + 1)
        node.right = self._build_tree(X[right_mask], y[right_mask], depth + 1)
        
        return node
    
    def _predict_single(self, x, node):
        """단일 샘플 예측"""
        if node.prediction is not None and node.left is None and node.right is None:
            return node.prediction
        
        if x[node.feature_idx] <= node.threshold:
            return self._predict_single(x, node.left)
        else:
            return self._predict_single(x, node.right)
    
    def predict(self, X):
        """예측 수행"""
        predictions = []
        for x in X:
            pred = self._predict_single(x, self.root)
            predictions.append(pred)
        return np.array(predictions)
    
    def print_tree(self, node=None, depth=0):
        """트리 구조 출력"""
        if node is None:
            node = self.root
        
        indent = "  " * depth
        
        if node.left is None and node.right is None:
            print(f"{indent}Predict: {node.prediction} (Gini: {node.gini:.3f})")
        else:
            print(f"{indent}Feature_{node.feature_idx} <= {node.threshold:.3f} (Gini: {node.gini:.3f})")
            if node.left:
                print(f"{indent}├─ True:")
                self.print_tree(node.left, depth + 1)
            if node.right:
                print(f"{indent}└─ False:")
                self.print_tree(node.right, depth + 1)

# 사용 예시

X, y = make_classification(n_samples=100, n_features=4, n_classes=3, 
                          n_informative=3, n_redundant=1, random_state=42)

print(f"의사결정 트리 정확도: {accuracy:.2f}%")
print("\n트리 구조:")
tree.print_tree()
```
