# 앙상블 및 고급 머신러닝 알고리즘 구현

## 1. 랜덤 포레스트 (Random Forest)

### 1.1 수학적 기초

**배깅 (Bootstrap Aggregating)**
```
- 복원 추출로 B개의 부트스트랩 샘플 생성
- 각 샘플로 개별 모델 학습
- 최종 예측: 다수결(분류) 또는 평균(회귀)
```

**특성 무작위성 (Feature Randomness)**
```
- 각 노드에서 m = √p 개의 특성만 고려 (p: 전체 특성 수)
- 과적합 방지 및 다양성 증가
```

### 1.2 Python 구현

```python
import numpy as np
from collections import Counter
import random

class RandomForestClassifier:
    def __init__(self, n_estimators=100, max_depth=None, min_samples_split=2, 
                 max_features='sqrt', bootstrap=True, random_state=None):
        self.n_estimators = n_estimators
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.max_features = max_features
        self.bootstrap = bootstrap
        self.random_state = random_state
        self.trees = []
        self.feature_indices = []
    
    def _bootstrap_sample(self, X, y):
        """부트스트랩 샘플링"""
        n_samples = X.shape[0]
        indices = np.random.choice(n_samples, size=n_samples, replace=True)
        return X[indices], y[indices]
    
    def _get_feature_subset(self, n_features):
        """특성 부분집합 선택"""
        if self.max_features == 'sqrt':
            max_features = int(np.sqrt(n_features))
        elif self.max_features == 'log2':
            max_features = int(np.log2(n_features))
        elif isinstance(self.max_features, int):
            max_features = self.max_features
        else:
            max_features = n_features
        
        return np.random.choice(n_features, size=min(max_features, n_features), replace=False)
    
    def fit(self, X, y):
        """랜덤 포레스트 학습"""
        if self.random_state:
            np.random.seed(self.random_state)
            random.seed(self.random_state)
        
        self.n_classes = len(np.unique(y))
        self.n_features = X.shape[1]
        
        for i in range(self.n_estimators):
            # 부트스트랩 샘플링
            if self.bootstrap:
                X_sample, y_sample = self._bootstrap_sample(X, y)
            else:
                X_sample, y_sample = X, y
            
            # 특성 부분집합 선택
            feature_indices = self._get_feature_subset(self.n_features)
            self.feature_indices.append(feature_indices)
            
            # 의사결정 트리 생성 및 학습
            tree = DecisionTreeClassifier(
                max_depth=self.max_depth,
                min_samples_split=self.min_samples_split
            )
            tree.fit(X_sample[:, feature_indices], y_sample)
            self.trees.append(tree)
    
    def predict(self, X):
        """앙상블 예측"""
        predictions = []
        
        # 각 트리의 예측 수집
        for i, tree in enumerate(self.trees):
            feature_indices = self.feature_indices[i]
            tree_pred = tree.predict(X[:, feature_indices])
            predictions.append(tree_pred)
        
        # 다수결 투표
        ensemble_predictions = []
        for i in range(X.shape[0]):
            votes = [pred[i] for pred in predictions]
            majority_vote = Counter(votes).most_common(1)[0][0]
            ensemble_predictions.append(majority_vote)
        
        return np.array(ensemble_predictions)
    
    def predict_proba(self, X):
        """확률 예측"""
        all_predictions = []
        
        for i, tree in enumerate(self.trees):
            feature_indices = self.feature_indices[i]
            tree_pred = tree.predict(X[:, feature_indices])
            all_predictions.append(tree_pred)
        
        # 각 클래스별 확률 계산
        probabilities = []
        unique_classes = np.unique(np.concatenate(all_predictions))
        
        for i in range(X.shape[0]):
            votes = [pred[i] for pred in all_predictions]
            vote_counts = Counter(votes)
            
            sample_probs = []
            for class_label in unique_classes:
                prob = vote_counts.get(class_label, 0) / self.n_estimators
                sample_probs.append(prob)
            
            probabilities.append(sample_probs)
        
        return np.array(probabilities), unique_classes
    
    def feature_importance(self):
        """특성 중요도 계산 (단순화된 버전)"""
        importances = np.zeros(self.n_features)
        
        for i, tree in enumerate(self.trees):
            feature_indices = self.feature_indices[i]
            # 각 트리에서 사용된 특성에 동일한 가중치 부여 (단순화)
            for idx in feature_indices:
                importances[idx] += 1
        
        # 정규화
        importances = importances / np.sum(importances)
        return importances
```

## 2. 그래디언트 부스팅 (Gradient Boosting)

### 2.1 수학적 기초

**손실 함수의 음의 기울기**
```
residual = -∂L(y, F(x))/∂F(x)
```

**부스팅 업데이트**
```
F_m(x) = F_{m-1}(x) + α × h_m(x)
α: 학습률, h_m: m번째 약한 학습기
```

**제곱 손실의 경우**
```
residual = y - F_{m-1}(x)
```

### 2.2 Python 구현

```python
class GradientBoostingClassifier:
    def __init__(self, n_estimators=100, learning_rate=0.1, max_depth=3, random_state=None):
        self.n_estimators = n_estimators
        self.learning_rate = learning_rate
        self.max_depth = max_depth
        self.random_state = random_state
        self.estimators = []
        self.init_prediction = None
    
    def _sigmoid(self, x):
        """시그모이드 함수"""
        x = np.clip(x, -500, 500)  # 수치적 안정성
        return 1 / (1 + np.exp(-x))
    
    def _log_odds(self, p):
        """로그 오즈 계산"""
        p = np.clip(p, 1e-15, 1 - 1e-15)  # 수치적 안정성
        return np.log(p / (1 - p))
    
    def fit(self, X, y):
        """그래디언트 부스팅 학습"""
        if self.random_state:
            np.random.seed(self.random_state)
        
        # 이진 분류로 변환 (단순화)
        self.classes = np.unique(y)
        if len(self.classes) != 2:
            raise ValueError("이 구현은 이진 분류만 지원합니다.")
        
        # y를 0, 1로 변환
        y_binary = (y == self.classes[1]).astype(int)
        
        # 초기 예측값 설정 (로그 오즈)
        p_init = np.mean(y_binary)
        self.init_prediction = self._log_odds(p_init)
        
        # 현재 예측값 초기화
        F = np.full(X.shape[0], self.init_prediction)
        
        for m in range(self.n_estimators):
            # 확률 계산
            p = self._sigmoid(F)
            
            # 잔차 계산 (음의 기울기)
            residuals = y_binary - p
            
            # 의사결정 트리로 잔차 학습
            tree = DecisionTreeRegressor(max_depth=self.max_depth)
            tree.fit(X, residuals)
            
            # 예측값 업데이트
            tree_predictions = tree.predict(X)
            F += self.learning_rate * tree_predictions
            
            self.estimators.append(tree)
    
    def predict_proba(self, X):
        """확률 예측"""
        # 초기 예측값으로 시작
        F = np.full(X.shape[0], self.init_prediction)
        
        # 각 트리의 예측값 누적
        for tree in self.estimators:
            F += self.learning_rate * tree.predict(X)
        
        # 시그모이드 적용하여 확률 계산
        p1 = self._sigmoid(F)
        p0 = 1 - p1
        
        return np.column_stack([p0, p1])
    
    def predict(self, X):
        """클래스 예측"""
        probas = self.predict_proba(X)
        predictions = np.where(probas[:, 1] > 0.5, self.classes[1], self.classes[0])
        return predictions

# 회귀용 의사결정 트리 (간단한 구현)
class DecisionTreeRegressor:
    def __init__(self, max_depth=None, min_samples_split=2):
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
    
    def fit(self, X, y):
        """회귀 트리 학습"""
        self.root = self._build_tree(X, y, depth=0)
    
    def _mse(self, y):
        """평균 제곱 오차 계산"""
        if len(y) == 0:
            return 0
        return np.mean((y - np.mean(y)) ** 2)
    
    def _best_split_regression(self, X, y):
        """회귀를 위한 최적 분할"""
        best_mse = float('inf')
        best_feature = None
        best_threshold = None
        
        for feature_idx in range(X.shape[1]):
            thresholds = np.unique(X[:, feature_idx])
            
            for threshold in thresholds:
                left_mask = X[:, feature_idx] <= threshold
                right_mask = ~left_mask
                
                if np.sum(left_mask) == 0 or np.sum(right_mask) == 0:
                    continue
                
                left_mse = self._mse(y[left_mask])
                right_mse = self._mse(y[right_mask])
                
                weighted_mse = (np.sum(left_mask) * left_mse + np.sum(right_mask) * right_mse) / len(y)
                
                if weighted_mse < best_mse:
                    best_mse = weighted_mse
                    best_feature = feature_idx
                    best_threshold = threshold
        
        return best_feature, best_threshold
    
    def _build_tree(self, X, y, depth):
        """회귀 트리 구축"""
        # 리프 노드 조건
        if (self.max_depth is not None and depth >= self.max_depth) or \
           len(y) < self.min_samples_split or \
           len(np.unique(y)) == 1:
            return {'prediction': np.mean(y)}
        
        feature_idx, threshold = self._best_split_regression(X, y)
        
        if feature_idx is None:
            return {'prediction': np.mean(y)}
        
        left_mask = X[:, feature_idx] <= threshold
        right_mask = ~left_mask
        
        return {
            'feature_idx': feature_idx,
            'threshold': threshold,
            'left': self._build_tree(X[left_mask], y[left_mask], depth + 1),
            'right': self._build_tree(X[right_mask], y[right_mask], depth + 1)
        }
    
    def predict(self, X):
        """예측 수행"""
        predictions = []
        for x in X:
            pred = self._predict_single(x, self.root)
            predictions.append(pred)
        return np.array(predictions)
    
    def _predict_single(self, x, node):
        """단일 샘플 예측"""
        if 'prediction' in node:
            return node['prediction']
        
        if x[node['feature_idx']] <= node['threshold']:
            return self._predict_single(x, node['left'])
        else:
            return self._predict_single(x, node['right'])
```

## 3. XGBoost 스타일 구현

### 3.1 수학적 기초

**목적 함수 (Objective Function)**
```
Obj = Σᵢ₌₁ⁿ L(yᵢ, ŷᵢ) + Σₖ₌₁ᴷ Ω(fₖ)
L: 손실 함수, Ω: 정규화 항
```

**2차 테일러 근사**
```
Obj ≈ Σᵢ₌₁ⁿ [L(yᵢ, ŷᵢ⁽ᵗ⁻¹⁾) + gᵢfₜ(xᵢ) + ½hᵢfₜ²(xᵢ)] + Ω(fₜ)
gᵢ = ∂L/∂ŷᵢ⁽ᵗ⁻¹⁾, hᵢ = ∂²L/∂ŷᵢ⁽ᵗ⁻¹⁾²
```

**최적 가중치**
```
w* = -G/(H + λ)
G: 기울기 합, H: 헤시안 합
```

### 3.2 단순화된 XGBoost 구현

```python
class SimpleXGBoost:
    def __init__(self, n_estimators=100, learning_rate=0.3, max_depth=6, reg_lambda=1.0):
        self.n_estimators = n_estimators
        self.learning_rate = learning_rate
        self.max_depth = max_depth
        self.reg_lambda = reg_lambda
        self.trees = []
        self.base_prediction = 0.5
    
    def _sigmoid(self, x):
        """시그모이드 함수"""
        x = np.clip(x, -500, 500)
        return 1 / (1 + np.exp(-x))
    
    def _log_loss_gradient(self, y_true, y_pred):
        """로그 손실의 1차 미분 (기울기)"""
        p = self._sigmoid(y_pred)
        return p - y_true
    
    def _log_loss_hessian(self, y_pred):
        """로그 손실의 2차 미분 (헤시안)"""
        p = self._sigmoid(y_pred)
        return p * (1 - p)
    
    def fit(self, X, y):
        """XGBoost 스타일 학습"""
        # 초기 예측값
        predictions = np.full(X.shape[0], np.log(self.base_prediction / (1 - self.base_prediction)))
        
        for i in range(self.n_estimators):
            # 기울기와 헤시안 계산
            gradients = self._log_loss_gradient(y, predictions)
            hessians = self._log_loss_hessian(predictions)
            
            # 트리 학습 (기울기와 헤시안 기반)
            tree = XGBoostTree(max_depth=self.max_depth, reg_lambda=self.reg_lambda)
            tree.fit(X, gradients, hessians)
            
            # 예측값 업데이트
            tree_predictions = tree.predict(X)
            predictions += self.learning_rate * tree_predictions
            
            self.trees.append(tree)
    
    def predict_proba(self, X):
        """확률 예측"""
        predictions = np.full(X.shape[0], np.log(self.base_prediction / (1 - self.base_prediction)))
        
        for tree in self.trees:
            predictions += self.learning_rate * tree.predict(X)
        
        probabilities = self._sigmoid(predictions)
        return np.column_stack([1 - probabilities, probabilities])
    
    def predict(self, X):
        """클래스 예측"""
        probas = self.predict_proba(X)
        return (probas[:, 1] > 0.5).astype(int)

class XGBoostTree:
    def __init__(self, max_depth=6, reg_lambda=1.0, min_child_weight=1):
        self.max_depth = max_depth
        self.reg_lambda = reg_lambda
        self.min_child_weight = min_child_weight
    
    def fit(self, X, gradients, hessians):
        """기울기와 헤시안을 사용한 트리 학습"""
        self.root = self._build_tree(X, gradients, hessians, depth=0)
    
    def _calculate_gain(self, left_g, left_h, right_g, right_h, parent_g, parent_h):
        """정보 이득 계산"""
        def calculate_score(G, H):
            return G**2 / (H + self.reg_lambda)
        
        left_score = calculate_score(left_g, left_h)
        right_score = calculate_score(right_g, right_h)
        parent_score = calculate_score(parent_g, parent_h)
        
        return 0.5 * (left_score + right_score - parent_score)
    
    def _find_best_split(self, X, gradients, hessians):
        """최적 분할 찾기"""
        best_gain = -float('inf')
        best_feature = None
        best_threshold = None
        
        parent_g = np.sum(gradients)
        parent_h = np.sum(hessians)
        
        for feature_idx in range(X.shape[1]):
            thresholds = np.unique(X[:, feature_idx])
            
            for threshold in thresholds:
                left_mask = X[:, feature_idx] <= threshold
                right_mask = ~left_mask
                
                if np.sum(left_mask) == 0 or np.sum(right_mask) == 0:
                    continue
                
                left_g = np.sum(gradients[left_mask])
                left_h = np.sum(hessians[left_mask])
                right_g = np.sum(gradients[right_mask])
                right_h = np.sum(hessians[right_mask])
                
                # min_child_weight 조건 확인
                if left_h < self.min_child_weight or right_h < self.min_child_weight:
                    continue
                
                gain = self._calculate_gain(left_g, left_h, right_g, right_h, parent_g, parent_h)
                
                if gain > best_gain:
                    best_gain = gain
                    best_feature = feature_idx
                    best_threshold = threshold
        
        return best_feature, best_threshold, best_gain
    
    def _build_tree(self, X, gradients, hessians, depth):
        """재귀적 트리 구축"""
        # 리프 노드 가중치 계산
        G = np.sum(gradients)
        H = np.sum(hessians)
        weight = -G / (H + self.reg_lambda)
        
        # 종료 조건
        if depth >= self.max_depth or len(gradients) < 2:
            return {'weight': weight}
        
        # 최적 분할 찾기
        feature_idx, threshold, gain = self._find_best_split(X, gradients, hessians)
        
        if feature_idx is None or gain <= 0:
            return {'weight': weight}
        
        # 분할
        left_mask = X[:, feature_idx] <= threshold
        right_mask = ~left_mask
        
        left_child = self._build_tree(
            X[left_mask], gradients[left_mask], hessians[left_mask], depth + 1
        )
        right_child = self._build_tree(
            X[right_mask], gradients[right_mask], hessians[right_mask], depth + 1
        )
        
        return {
            'feature_idx': feature_idx,
            'threshold': threshold,
            'left': left_child,
            'right': right_child
        }
    
    def predict(self, X):
        """예측 수행"""
        predictions = []
        for x in X:
            pred = self._predict_single(x, self.root)
            predictions.append(pred)
        return np.array(predictions)
    
    def _predict_single(self, x, node):
        """단일 샘플 예측"""
        if 'weight' in node:
            return node['weight']
        
        if x[node['feature_idx']] <= node['threshold']:
            return self._predict_single(x, node['left'])
        else:
            return self._predict_single(x, node['right'])
```

## 4. AdaBoost (Adaptive Boosting)

### 4.1 수학적 기초

**샘플 가중치 초기화**
```
D₁(i) = 1/m for all i
```

**약한 학습기 오류율**
```
ε_t = Σᵢ₌₁ᵐ D_t(i) × I(h_t(xᵢ) ≠ yᵢ)
```

**학습기 가중치**
```
α_t = ½ ln((1 - ε_t) / ε_t)
```

**샘플 가중치 업데이트**
```
D_{t+1}(i) = D_t(i) × exp(-α_t × yᵢ × h_t(xᵢ)) / Z_t
```

### 4.2 Python 구현

```python
class AdaBoostClassifier:
    def __init__(self, n_estimators=50, learning_rate=1.0):
        self.n_estimators = n_estimators
        self.learning_rate = learning_rate
        self.estimators = []
        self.estimator_weights = []
        self.estimator_errors = []
    
    def fit(self, X, y):
        """AdaBoost 학습"""
        m, n = X.shape
        
        # 이진 분류를 위해 -1, 1로 변환
        y_encoded = np.where(y == np.unique(y)[0], -1, 1)
        
        # 샘플 가중치 초기화
        sample_weights = np.ones(m) / m
        
        for t in range(self.n_estimators):
            # 가중치 기반 약한 학습기 학습
            weak_learner = DecisionStump()
            weak_learner.fit(X, y_encoded, sample_weights)
            
            # 예측 수행
            predictions = weak_learner.predict(X)
            
            # 가중 오류율 계산
            incorrect = predictions != y_encoded
            error_rate = np.sum(sample_weights * incorrect)
            
            # 학습 종료 조건
            if error_rate == 0.0:
                self.estimators.append(weak_learner)
                self.estimator_weights.append(1.0)
                break
            
            if error_rate >= 0.5:
                continue
            
            # 학습기 가중치 계산
            alpha = self.learning_rate * 0.5 * np.log((1 - error_rate) / error_rate)
            
            # 샘플 가중치 업데이트
            sample_weights *= np.exp(-alpha * y_encoded * predictions)
            sample_weights /= np.sum(sample_weights)  # 정규화
            
            # 저장
            self.estimators.append(weak_learner)
            self.estimator_weights.append(alpha)
            self.estimator_errors.append(error_rate)
    
    def predict(self, X):
        """앙상블 예측"""
        if not self.estimators:
            raise ValueError("모델이 학습되지 않았습니다.")
        
        # 가중 투표
        ensemble_predictions = np.zeros(X.shape[0])
        
        for estimator, weight in zip(self.estimators, self.estimator_weights):
            predictions = estimator.predict(X)
            ensemble_predictions += weight * predictions
        
        # 부호 기준으로 최종 분류
        return np.where(ensemble_predictions >= 0, 1, -1)
    
    def decision_function(self, X):
        """결정 함수값 반환"""
        ensemble_predictions = np.zeros(X.shape[0])
        
        for estimator, weight in zip(self.estimators, self.estimator_weights):
            predictions = estimator.predict(X)
            ensemble_predictions += weight * predictions
        
        return ensemble_predictions

class DecisionStump:
    """의사결정 그루터기 (깊이 1인 의사결정 트리)"""
    
    def __init__(self):
        self.feature_idx = None
        self.threshold = None
        self.polarity = 1
    
    def fit(self, X, y, sample_weights):
        """가중치를 고려한 최적 분할 찾기"""
        m, n = X.shape
        min_error = float('inf')
        
        for feature_idx in range(n):
            feature_values = X[:, feature_idx]
            thresholds = np.unique(feature_values)
            
            for threshold in thresholds:
                for polarity in [-1, 1]:
                    # 예측 수행
                    predictions = np.ones(m)
                    if polarity == 1:
                        predictions[feature_values < threshold] = -1
                    else:
                        predictions[feature_values >= threshold] = -1
                    
                    # 가중 오류율 계산
                    misclassified = predictions != y
                    error = np.sum(sample_weights * misclassified)
                    
                    if error < min_error:
                        min_error = error
                        self.feature_idx = feature_idx
                        self.threshold = threshold
                        self.polarity = polarity
    
    def predict(self, X):
        """예측 수행"""
        m = X.shape[0]
        predictions = np.ones(m)
        
        if self.polarity == 1:
            predictions[X[:, self.feature_idx] < self.threshold] = -1
        else:
            predictions[X[:, self.feature_idx] >= self.threshold] = -1
        
        return predictions
```

## 5. 투표 분류기 (Voting Classifier)

### 5.1 수학적 기초

**하드 투표 (Hard Voting)**
```
ŷ = mode{h₁(x), h₂(x), ..., hₘ(x)}
```

**소프트 투표 (Soft Voting)**
```
ŷ = argmax Σⱼ₌₁ᵐ pⱼ,ᵢ
pⱼ,ᵢ: j번째 분류기가 클래스 i에 할당한 확률
```

### 5.2 Python 구현

```python
class VotingClassifier:
    def __init__(self, estimators, voting='hard'):
        self.estimators = estimators  # [(name, estimator), ...]
        self.voting = voting
        self.fitted_estimators = {}
    
    def fit(self, X, y):
        """모든 기본 분류기 학습"""
        for name, estimator in self.estimators:
            estimator.fit(X, y)
            self.fitted_estimators[name] = estimator
    
    def predict(self, X):
        """앙상블 예측"""
        if self.voting == 'hard':
            return self._hard_voting(X)
        else:
            return self._soft_voting(X)
    
    def _hard_voting(self, X):
        """하드 투표"""
        predictions = []
        
        for name, estimator in self.fitted_estimators.items():
            pred = estimator.predict(X)
            predictions.append(pred)
        
        predictions = np.array(predictions).T
        
        # 다수결 투표
        final_predictions = []
        for pred_row in predictions:
            vote_count = Counter(pred_row)
            majority_vote = vote_count.most_common(1)[0][0]
            final_predictions.append(majority_vote)
        
        return np.array(final_predictions)
    
    def _soft_voting(self, X):
        """소프트 투표"""
        all_probas = []
        
        for name, estimator in self.fitted_estimators.items():
            if hasattr(estimator, 'predict_proba'):
                probas = estimator.predict_proba(X)
                all_probas.append(probas)
            else:
                raise ValueError(f"{name} 분류기가 predict_proba를 지원하지 않습니다.")
        
        # 확률 평균
        avg_probas = np.mean(all_probas, axis=0)
        return np.argmax(avg_probas, axis=1)

# 스태킹 앙상블 (Stacking)
class StackingClassifier:
    def __init__(self, base_estimators, meta_estimator, cv=3):
        self.base_estimators = base_estimators
        self.meta_estimator = meta_estimator
        self.cv = cv
        self.fitted_base_estimators = {}
    
    def fit(self, X, y):
        """스태킹 앙상블 학습"""
        from sklearn.model_selection import cross_val_predict
        
        # 1단계: 기본 분류기들의 교차 검증 예측
        meta_features = []
        
        for name, estimator in self.base_estimators:
            # 교차 검증으로 예측 생성
            cv_predictions = cross_val_predict(estimator, X, y, cv=self.cv, method='predict_proba')
            meta_features.append(cv_predictions)
            
            # 전체 데이터로 재학습
            estimator.fit(X, y)
            self.fitted_base_estimators[name] = estimator
        
        # 메타 특성 행렬 생성
        meta_X = np.column_stack([pred.max(axis=1) for pred in meta_features])
        
        # 2단계: 메타 분류기 학습
        self.meta_estimator.fit(meta_X, y)
    
    def predict(self, X):
        """스태킹 예측"""
        # 기본 분류기들의 예측
        base_predictions = []
        for name, estimator in self.fitted_base_estimators.items():
            pred_proba = estimator.predict_proba(X)
            base_predictions.append(pred_proba.max(axis=1))
        
        # 메타 특성 생성
        meta_X = np.column_stack(base_predictions)
        
        # 메타 분류기로 최종 예측
        return self.meta_estimator.predict(meta_X)
```
