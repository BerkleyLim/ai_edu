# 최적화 알고리즘 및 손실함수 상세 구현

## 1. 손실함수 (Loss Functions)

### 1.1 회귀 문제 손실함수

#### 평균 제곱 오차 (Mean Squared Error)

**수식**
```
MSE = (1/n) Σᵢ₌₁ⁿ (yᵢ - ŷᵢ)²
∂MSE/∂ŷ = (2/n) Σᵢ₌₁ⁿ (ŷᵢ - yᵢ)
```

**Python 구현**
```python
import numpy as np

class MeanSquaredError:
    def __init__(self):
        self.name = "MSE"
    
    def forward(self, y_true, y_pred):
        """
        순전파: 손실 계산
        
        Args:
            y_true: 실제 값 (n_samples,)
            y_pred: 예측 값 (n_samples,)
        
        Returns:
            loss: MSE 손실값
        """
        self.y_true = y_true
        self.y_pred = y_pred
        n = len(y_true)
        
        loss = np.mean((y_true - y_pred) ** 2)
        return loss
    
    def backward(self):
        """
        역전파: 기울기 계산
        
        Returns:
            gradient: 예측값에 대한 기울기
        """
        n = len(self.y_true)
        gradient = (2 / n) * (self.y_pred - self.y_true)
        return gradient

class MeanAbsoluteError:
    def __init__(self):
        self.name = "MAE"
    
    def forward(self, y_true, y_pred):
        """평균 절대 오차"""
        self.y_true = y_true
        self.y_pred = y_pred
        
        loss = np.mean(np.abs(y_true - y_pred))
        return loss
    
    def backward(self):
        """MAE 기울기"""
        n = len(self.y_true)
        gradient = (1 / n) * np.sign(self.y_pred - self.y_true)
        return gradient

class HuberLoss:
    def __init__(self, delta=1.0):
        """
        Huber 손실 (이상치에 강건)
        
        Args:
            delta: 임계값
        """
        self.delta = delta
        self.name = "Huber"
    
    def forward(self, y_true, y_pred):
        """Huber 손실 계산"""
        self.y_true = y_true
        self.y_pred = y_pred
        self.residual = y_true - y_pred
        
        # |r| <= δ인 경우: 0.5 * r²
        # |r| > δ인 경우: δ * (|r| - 0.5 * δ)
        abs_residual = np.abs(self.residual)
        quadratic = np.minimum(abs_residual, self.delta)
        linear = abs_residual - quadratic
        
        loss = np.mean(0.5 * quadratic ** 2 + self.delta * linear)
        return loss
    
    def backward(self):
        """Huber 손실 기울기"""
        n = len(self.y_true)
        abs_residual = np.abs(self.residual)
        
        # |r| <= δ인 경우: -r, |r| > δ인 경우: -δ * sign(r)
        gradient = np.where(
            abs_residual <= self.delta,
            -self.residual,
            -self.delta * np.sign(self.residual)
        )
        
        return gradient / n
```

### 1.2 분류 문제 손실함수

#### 교차 엔트로피 (Cross Entropy)

**수식**
```
Binary Cross Entropy:
BCE = -(1/n) Σᵢ₌₁ⁿ [yᵢlog(pᵢ) + (1-yᵢ)log(1-pᵢ)]

Categorical Cross Entropy:
CCE = -(1/n) Σᵢ₌₁ⁿ Σⱼ₌₁ᶜ yᵢⱼ log(pᵢⱼ)
```

**Python 구현**
```python
class BinaryCrossEntropy:
    def __init__(self):
        self.name = "Binary CrossEntropy"
    
    def forward(self, y_true, y_pred):
        """
        이진 교차 엔트로피 손실
        
        Args:
            y_true: 실제 라벨 (n_samples,) - 0 또는 1
            y_pred: 예측 확률 (n_samples,) - 0과 1 사이
        """
        self.y_true = y_true
        self.y_pred = y_pred
        
        # 수치적 안정성을 위한 클리핑
        y_pred_clipped = np.clip(y_pred, 1e-15, 1 - 1e-15)
        self.y_pred_clipped = y_pred_clipped
        
        # BCE 계산
        loss = -np.mean(
            y_true * np.log(y_pred_clipped) + 
            (1 - y_true) * np.log(1 - y_pred_clipped)
        )
        
        return loss
    
    def backward(self):
        """BCE 기울기"""
        n = len(self.y_true)
        
        # ∂BCE/∂p = -y/p + (1-y)/(1-p)
        gradient = -(self.y_true / self.y_pred_clipped - 
                    (1 - self.y_true) / (1 - self.y_pred_clipped)) / n
        
        return gradient

class CategoricalCrossEntropy:
    def __init__(self):
        self.name = "Categorical CrossEntropy"
    
    def forward(self, y_true, y_pred):
        """
        다범주 교차 엔트로피 손실
        
        Args:
            y_true: 원-핫 인코딩된 실제 라벨 (n_samples, n_classes)
            y_pred: 예측 확률 (n_samples, n_classes)
        """
        self.y_true = y_true
        self.y_pred = y_pred
        
        # 수치적 안정성
        y_pred_clipped = np.clip(y_pred, 1e-15, 1 - 1e-15)
        self.y_pred_clipped = y_pred_clipped
        
        # CCE 계산
        loss = -np.mean(np.sum(y_true * np.log(y_pred_clipped), axis=1))
        
        return loss
    
    def backward(self):
        """CCE 기울기"""
        n = self.y_true.shape[0]
        gradient = -self.y_true / (self.y_pred_clipped * n)
        
        return gradient

class SparseCategoricalCrossEntropy:
    def __init__(self):
        self.name = "Sparse Categorical CrossEntropy"
    
    def forward(self, y_true, y_pred):
        """
        희소 다범주 교차 엔트로피 (정수 라벨 사용)
        
        Args:
            y_true: 정수 라벨 (n_samples,)
            y_pred: 예측 확률 (n_samples, n_classes)
        """
        self.y_true = y_true.astype(int)
        self.y_pred = y_pred
        
        # 수치적 안정성
        y_pred_clipped = np.clip(y_pred, 1e-15, 1 - 1e-15)
        self.y_pred_clipped = y_pred_clipped
        
        # 각 샘플의 정답 클래스 확률 추출
        correct_confidences = y_pred_clipped[np.arange(len(y_true)), y_true]
        loss = -np.mean(np.log(correct_confidences))
        
        return loss
    
    def backward(self):
        """Sparse CCE 기울기"""
        n = len(self.y_true)
        gradient = np.copy(self.y_pred_clipped)
        
        # 정답 클래스 위치에만 기울기 적용
        gradient[np.arange(n), self.y_true] -= 1
        gradient /= n
        
        return gradient

class FocalLoss:
    def __init__(self, alpha=1.0, gamma=2.0):
        """
        Focal Loss (불균형 데이터용)
        
        Args:
            alpha: 가중치 인수
            gamma: 포커싱 인수
        """
        self.alpha = alpha
        self.gamma = gamma
        self.name = "Focal Loss"
    
    def forward(self, y_true, y_pred):
        """Focal Loss 계산"""
        self.y_true = y_true
        self.y_pred = y_pred
        
        # 수치적 안정성
        y_pred_clipped = np.clip(y_pred, 1e-15, 1 - 1e-15)
        self.y_pred_clipped = y_pred_clipped
        
        # p_t 계산
        p_t = np.where(y_true == 1, y_pred_clipped, 1 - y_pred_clipped)
        
        # α_t 계산
        alpha_t = np.where(y_true == 1, self.alpha, 1 - self.alpha)
        
        # Focal Loss: -α_t * (1-p_t)^γ * log(p_t)
        focal_weight = alpha_t * ((1 - p_t) ** self.gamma)
        loss = -np.mean(focal_weight * np.log(p_t))
        
        # 역전파를 위한 저장
        self.p_t = p_t
        self.alpha_t = alpha_t
        self.focal_weight = focal_weight
        
        return loss
    
    def backward(self):
        """Focal Loss 기울기"""
        n = len(self.y_true)
        
        # 복잡한 미분식 구현
        term1 = self.focal_weight / self.p_t
        term2 = self.alpha_t * self.gamma * ((1 - self.p_t) ** (self.gamma - 1)) * np.log(self.p_t)
        
        gradient_pt = -(term1 + term2) / n
        
        # p_t에서 y_pred로 변환
        gradient = np.where(self.y_true == 1, gradient_pt, -gradient_pt)
        
        return gradient
```

## 2. 최적화 알고리즘 (Optimization Algorithms)

### 2.1 기본 경사하강법

**수식**
```
θ_{t+1} = θ_t - α ∇J(θ_t)
α: 학습률
∇J(θ_t): 비용함수의 기울기
```

**Python 구현**
```python
class SGD:
    def __init__(self, learning_rate=0.01):
        """
        확률적 경사하강법
        
        Args:
            learning_rate: 학습률
        """
        self.learning_rate = learning_rate
        self.name = "SGD"
    
    def update(self, params, grads):
        """
        매개변수 업데이트
        
        Args:
            params: 매개변수 딕셔너리 {'W1': w1, 'b1': b1, ...}
            grads: 기울기 딕셔너리 {'dW1': dw1, 'db1': db1, ...}
        """
        for key in params:
            params[key] -= self.learning_rate * grads['d' + key]
        
        return params

class SGDMomentum:
    def __init__(self, learning_rate=0.01, momentum=0.9):
        """
        모멘텀 SGD
        
        Args:
            learning_rate: 학습률
            momentum: 모멘텀 계수
        """
        self.learning_rate = learning_rate
        self.momentum = momentum
        self.velocities = {}
        self.name = "SGD with Momentum"
    
    def update(self, params, grads):
        """모멘텀을 사용한 매개변수 업데이트"""
        for key in params:
            # 속도 초기화
            if key not in self.velocities:
                self.velocities[key] = np.zeros_like(params[key])
            
            # v_t = β * v_{t-1} + (1-β) * ∇J(θ)
            self.velocities[key] = (self.momentum * self.velocities[key] + 
                                   (1 - self.momentum) * grads['d' + key])
            
            # θ_{t+1} = θ_t - α * v_t
            params[key] -= self.learning_rate * self.velocities[key]
        
        return params

class AdaGrad:
    def __init__(self, learning_rate=0.01, epsilon=1e-8):
        """
        AdaGrad 최적화
        
        Args:
            learning_rate: 초기 학습률
            epsilon: 수치적 안정성을 위한 작은 값
        """
        self.learning_rate = learning_rate
        self.epsilon = epsilon
        self.accumulated_grads = {}
        self.name = "AdaGrad"
    
    def update(self, params, grads):
        """AdaGrad 업데이트"""
        for key in params:
            # 누적 기울기 제곱 초기화
            if key not in self.accumulated_grads:
                self.accumulated_grads[key] = np.zeros_like(params[key])
            
            # G_t = G_{t-1} + (∇J(θ))²
            self.accumulated_grads[key] += grads['d' + key] ** 2
            
            # θ_{t+1} = θ_t - α/√(G_t + ε) * ∇J(θ)
            adjusted_lr = self.learning_rate / (np.sqrt(self.accumulated_grads[key]) + self.epsilon)
            params[key] -= adjusted_lr * grads['d' + key]
        
        return params

class RMSprop:
    def __init__(self, learning_rate=0.001, decay_rate=0.9, epsilon=1e-8):
        """
        RMSprop 최적화
        
        Args:
            learning_rate: 학습률
            decay_rate: 감쇠율
            epsilon: 수치적 안정성을 위한 작은 값
        """
        self.learning_rate = learning_rate
        self.decay_rate = decay_rate
        self.epsilon = epsilon
        self.accumulated_grads = {}
        self.name = "RMSprop"
    
    def update(self, params, grads):
        """RMSprop 업데이트"""
        for key in params:
            # 누적 기울기 제곱 초기화
            if key not in self.accumulated_grads:
                self.accumulated_grads[key] = np.zeros_like(params[key])
            
            # E[g²]_t = β * E[g²]_{t-1} + (1-β) * (∇J(θ))²
            self.accumulated_grads[key] = (self.decay_rate * self.accumulated_grads[key] + 
                                         (1 - self.decay_rate) * grads['d' + key] ** 2)
            
            # θ_{t+1} = θ_t - α/√(E[g²]_t + ε) * ∇J(θ)
            adjusted_lr = self.learning_rate / (np.sqrt(self.accumulated_grads[key]) + self.epsilon)
            params[key] -= adjusted_lr * grads['d' + key]
        
        return params

class Adam:
    def __init__(self, learning_rate=0.001, beta1=0.9, beta2=0.999, epsilon=1e-8):
        """
        Adam 최적화
        
        Args:
            learning_rate: 학습률
            beta1: 1차 모멘트 지수 감쇠율
            beta2: 2차 모멘트 지수 감쇠율
            epsilon: 수치적 안정성을 위한 작은 값
        """
        self.learning_rate = learning_rate
        self.beta1 = beta1
        self.beta2 = beta2
        self.epsilon = epsilon
        self.m = {}  # 1차 모멘트
        self.v = {}  # 2차 모멘트
        self.t = 0   # 시간 단계
        self.name = "Adam"
    
    def update(self, params, grads):
        """Adam 업데이트"""
        self.t += 1
        
        for key in params:
            # 모멘트 초기화
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])
            
            # 1차 모멘트 (지수 이동 평균)
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * grads['d' + key]
            
            # 2차 모멘트 (지수 이동 평균)
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * (grads['d' + key] ** 2)
            
            # 편향 보정
            m_corrected = self.m[key] / (1 - self.beta1 ** self.t)
            v_corrected = self.v[key] / (1 - self.beta2 ** self.t)
            
            # 매개변수 업데이트
            params[key] -= self.learning_rate * m_corrected / (np.sqrt(v_corrected) + self.epsilon)
        
        return params

class AdamW:
    def __init__(self, learning_rate=0.001, beta1=0.9, beta2=0.999, epsilon=1e-8, weight_decay=0.01):
        """
        AdamW (가중치 감쇠가 포함된 Adam)
        
        Args:
            weight_decay: 가중치 감쇠 계수
        """
        self.learning_rate = learning_rate
        self.beta1 = beta1
        self.beta2 = beta2
        self.epsilon = epsilon
        self.weight_decay = weight_decay
        self.m = {}
        self.v = {}
        self.t = 0
        self.name = "AdamW"
    
    def update(self, params, grads):
        """AdamW 업데이트"""
        self.t += 1
        
        for key in params:
            # 모멘트 초기화
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])
            
            # 1차 모멘트
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * grads['d' + key]
            
            # 2차 모멘트
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * (grads['d' + key] ** 2)
            
            # 편향 보정
            m_corrected = self.m[key] / (1 - self.beta1 ** self.t)
            v_corrected = self.v[key] / (1 - self.beta2 ** self.t)
            
            # 가중치 감쇠를 포함한 업데이트
            params[key] -= self.learning_rate * (
                m_corrected / (np.sqrt(v_corrected) + self.epsilon) + 
                self.weight_decay * params[key]
            )
        
        return params
```

class Nadam:
    def __init__(self, learning_rate=0.002, beta1=0.9, beta2=0.999, epsilon=1e-8):
        """
        Nadam (Nesterov-accelerated Adam)
        
        Args:
            learning_rate: 학습률
            beta1: 1차 모멘트 지수 감쇠율
            beta2: 2차 모멘트 지수 감쇠율
            epsilon: 수치적 안정성
        """
        self.learning_rate = learning_rate
        self.beta1 = beta1
        self.beta2 = beta2
        self.epsilon = epsilon
        self.m = {}
        self.v = {}
        self.t = 0
        self.name = "Nadam"
    
    def update(self, params, grads):
        """Nadam 업데이트"""
        self.t += 1
        
        for key in params:
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])
            
            # 모멘트 업데이트
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * grads['d' + key]
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * (grads['d' + key] ** 2)
            
            # 편향 보정
            m_corrected = self.m[key] / (1 - self.beta1 ** self.t)
            v_corrected = self.v[key] / (1 - self.beta2 ** self.t)
            
            # Nesterov 가속 적용
            m_nesterov = self.beta1 * m_corrected + (1 - self.beta1) * grads['d' + key] / (1 - self.beta1 ** self.t)
            
            # 매개변수 업데이트
            params[key] -= self.learning_rate * m_nesterov / (np.sqrt(v_corrected) + self.epsilon)
        
        return params

class LAMB:
    def __init__(self, learning_rate=0.001, beta1=0.9, beta2=0.999, epsilon=1e-6, weight_decay=0.01):
        """
        LAMB (Layer-wise Adaptive Moments optimizer for Large Batch training)
        
        대용량 배치 훈련에 최적화된 옵티마이저
        """
        self.learning_rate = learning_rate
        self.beta1 = beta1
        self.beta2 = beta2
        self.epsilon = epsilon
        self.weight_decay = weight_decay
        self.m = {}
        self.v = {}
        self.t = 0
        self.name = "LAMB"
    
    def update(self, params, grads):
        """LAMB 업데이트"""
        self.t += 1
        
        for key in params:
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])
            
            # 모멘트 업데이트
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * grads['d' + key]
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * (grads['d' + key] ** 2)
            
            # 편향 보정
            m_corrected = self.m[key] / (1 - self.beta1 ** self.t)
            v_corrected = self.v[key] / (1 - self.beta2 ** self.t)
            
            # Adam 스텝 계산
            adam_step = m_corrected / (np.sqrt(v_corrected) + self.epsilon)
            
            # 가중치 감쇠 추가
            if self.weight_decay > 0:
                adam_step += self.weight_decay * params[key]
            
            # Layer-wise adaptation
            param_norm = np.linalg.norm(params[key])
            adam_norm = np.linalg.norm(adam_step)
            
            if param_norm > 0 and adam_norm > 0:
                trust_ratio = param_norm / adam_norm
                trust_ratio = min(trust_ratio, 10)  # 클리핑
            else:
                trust_ratio = 1
            
            # 매개변수 업데이트
            params[key] -= self.learning_rate * trust_ratio * adam_step
        
        return params

# 학습률 스케줄링
class LearningRateScheduler:
    def __init__(self, optimizer, schedule_type='constant', **kwargs):
        """
        학습률 스케줄러
        
        Args:
            optimizer: 최적화 객체
            schedule_type: 스케줄 타입 ('constant', 'step', 'exponential', 'cosine')
        """
        self.optimizer = optimizer
        self.schedule_type = schedule_type
        self.initial_lr = optimizer.learning_rate
        self.step_count = 0
        
        # 스케줄별 파라미터
        self.step_size = kwargs.get('step_size', 100)
        self.gamma = kwargs.get('gamma', 0.1)
        self.T_max = kwargs.get('T_max', 1000)
        self.eta_min = kwargs.get('eta_min', 0)
    
    def step(self):
        """학습률 업데이트"""
        self.step_count += 1
        
        if self.schedule_type == 'step':
            # 스텝 감소
            if self.step_count % self.step_size == 0:
                self.optimizer.learning_rate *= self.gamma
        
        elif self.schedule_type == 'exponential':
            # 지수 감소
            self.optimizer.learning_rate = self.initial_lr * (self.gamma ** (self.step_count / self.step_size))
        
        elif self.schedule_type == 'cosine':
            # 코사인 어닐링
            self.optimizer.learning_rate = self.eta_min + (self.initial_lr - self.eta_min) * \
                                          (1 + np.cos(np.pi * self.step_count / self.T_max)) / 2
        
        # 'constant'인 경우 변화 없음
        
        return self.optimizer.learning_rate

# 최적화 알고리즘 비교 예시
class OptimizationComparison:
    def __init__(self, X, y, model_class):
        """
        최적화 알고리즘 비교
        
        Args:
            X: 입력 데이터
            y: 타겟 데이터
            model_class: 모델 클래스
        """
        self.X = X
        self.y = y
        self.model_class = model_class
        self.results = {}
    
    def compare_optimizers(self, optimizers, epochs=100):
        """여러 최적화 알고리즘 비교"""
        for opt in optimizers:
            print(f"\n{opt.name} 최적화 시작...")
            
            # 모델 초기화
            model = self.model_class()
            losses = []
            
            for epoch in range(epochs):
                # 순전파
                predictions = model.forward(self.X)
                loss = model.compute_loss(predictions, self.y)
                losses.append(loss)
                
                # 역전파
                gradients = model.backward()
                
                # 매개변수 업데이트
                model.parameters = opt.update(model.parameters, gradients)
                
                if epoch % 20 == 0:
                    print(f"Epoch {epoch}: Loss = {loss:.6f}")
            
            self.results[opt.name] = {
                'losses': losses,
                'final_loss': losses[-1],
                'model': model
            }
            
            print(f"{opt.name} 최종 손실: {losses[-1]:.6f}")
    
    def plot_comparison(self):
        """결과 시각화"""
        plt.figure(figsize=(12, 8))
        
        for name, result in self.results.items():
            plt.plot(result['losses'], label=f"{name} (최종: {result['final_loss']:.4f})")
        
        plt.xlabel('Epochs')
        plt.ylabel('Loss')
        plt.title('Optimization Algorithms Comparison')
        plt.legend()
        plt.grid(True)
        plt.yscale('log')
        plt.show()
